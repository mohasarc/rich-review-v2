284   private async waitForStartupAuthorization(): Promise<{
285     readonly lease: DaemonStartupLease;
286     readonly record: DaemonRecord;
287   }> {
288     const deadline = this.clock.wallNowMs() + this.policy.values.startup.coordinationGraceMs;
289     while (this.clock.wallNowMs() <= deadline) {
290       const record = this.options.registry.readInstance(
291         this.options.identity,
292         this.options.coordinates.instanceId,
293       );
294       if (
295         record?.state === "starting" &&
296         (record.pid === 0 || record.pid === process.pid) &&
297         record.processToken === this.options.coordinates.processToken
298       ) {
299         const lease = this.options.registry.claimStartupForDaemon(
300           this.options.identity,
301           this.options.coordinates.instanceId,
302           this.options.coordinates.processToken,
303           process.pid,
304         );
305         if (lease === undefined) {
306           await this.pause();
307           continue;
308         }
309         this.options.registry.writeStartingIfStartupOwner(this.options.identity, {
310           ...record,
311           pid: process.pid,
312         });
313         const adoptedRecord = this.options.registry.readInstance(
314           this.options.identity,
315           this.options.coordinates.instanceId,
316         );
317         if (adoptedRecord?.pid !== process.pid) {
318           lease.release();
319           await this.pause();
320           continue;
321         }
322         if (
323           this.options.registry.startupOwnerMatchesProcess(this.options.identity, adoptedRecord)
324         ) {
325           return { lease, record: adoptedRecord };
326         }
327         lease.release();
328       }
329       await this.pause();
330     }
331     throw new Error("Daemon process did not receive startup authorization");
332   }

350   private async handle(
351     request: DaemonRequest,
352     send: DaemonServerSend,
353   ): Promise<DaemonResponse | void> {
354     if (request.kind === "identify") return this.identify(request);
355     if (request.kind === "terminate" || request.kind === "kill") {
356       return this.terminate(request);
357     }
358     if (
359       request.protocolVersion !== DAEMON_PROTOCOL_VERSION ||
360       request.instanceId !== this.options.coordinates.instanceId
361     ) {
362       throw new Error("Daemon request does not match protocol or instance");
363     }
364     if (request.kind === "ping") return this.pong();
365     if (request.kind === "execute") return this.acceptExecution(request, send);
366     if (
367       request.kind === "execution-status" ||
368       request.kind === "result-fetch" ||
369       request.kind === "result-ack"
370     ) {
371       if (request.processToken !== this.options.coordinates.processToken) {
372         throw new Error("Daemon execution request does not match process instance");
373       }
374     }
375     if (request.kind === "result-fetch") {
376       await this.deliverySession.fetch(request, send);
377       return;
378     }
379     if (request.kind === "result-ack") {
380       return this.deliverySession.acknowledge(request);
381     }
382     if (request.kind === "execution-status") {
383       return {
384         kind: "execution-status",
385         instanceId: this.options.coordinates.instanceId,
386         processToken: this.options.coordinates.processToken,
387         requestId: request.requestId,
388         status: this.acceptedExecutionSession.status(request.requestId),
389       };
390     }
391     this.beginGracefulShutdown();
392     await this.acceptedExecutionSession.drain();
393     await this.deliverySession.waitForCompletionAcknowledgements();
394     setTimeout(() => void this.shutdown("graceful"), 0);
395     return { kind: "stopped", instanceId: this.options.coordinates.instanceId };
396   }
