71   async launch(): Promise<void> {
72     const lease = this.registry.acquireStartup(this.identity, {
73       identityKey: this.identity.identityKey,
74       instanceId: this.instanceId,
75       processToken: this.processToken,
76       ownerPid: process.pid,
77       ownerKind: "launcher",
78       heartbeatAt: Date.now(),
79     });
80     if (lease === undefined) throw new Error("Built process entry startup lease unavailable");
81     const startingRecord = this.startingRecord(0);
82     if (!this.registry.writeStartingIfStartupOwner(this.identity, startingRecord)) {
83       throw new Error("Built process entry starting record unavailable");
84     }
85     const daemonProcess = await new NodeDaemonProcessLauncher(
86       this.productVersion,
87       this.executorModuleUrl,
88       this.policy,
89     ).launch(this.identity, this.instanceId, this.processToken);
90     this.daemonProcess = daemonProcess;
91     void daemonProcess.exited.then((exit) => {
92       this.observedExit = exit;
93     });
94     const transferred = lease.transferToDaemon(daemonProcess.pid, this.processToken);
95     if (
96       !transferred &&
97       !this.registry.daemonOwnsStartupProcess(
98         this.identity,
99         this.instanceId,
100         this.processToken,
101         daemonProcess.pid,
102       )
103     ) {
104       throw new Error("Built process entry startup ownership unavailable");
105     }
106     if (
107       !this.registry.writeStartingIfStartupOwner(
108         this.identity,
109         this.startingRecord(daemonProcess.pid),
110       )
111     ) {
112       throw new Error("Built process entry process record unavailable");
113     }

116   async waitForReady(): Promise<DaemonRecord> {
117     for (let attempt = 0; attempt < 400; attempt += 1) {
118       const record = this.registry.readInstance(this.identity, this.instanceId);
119       if (record?.state === "ready") return record;
120       if (this.observedExit !== undefined) {
121         throw new Error(

124       }
125       await delay(5);
126     }
127     throw new Error("Built process entry did not publish readiness");
