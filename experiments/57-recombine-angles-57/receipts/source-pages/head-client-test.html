442 async function outputText(result: DaemonExecutorExecutionResult): Promise<string> {
443   const chunks: Uint8Array[] = [];
444   for await (const outputRecord of result.output.records()) chunks.push(outputRecord.bytes);
445   await result.output.dispose();
446   return Buffer.concat(chunks).toString();
