161   private send(
162     key: string,
163     request: DaemonNavigationWorkerRequest,
164     appendOutput: ((record: DaemonSequencedOutputRecord) => Promise<void>) | undefined = undefined,
165   ): Promise<DaemonNavigationWorkerResponse> {
166     if (this.exit !== undefined)
167       return Promise.reject(new Error("Daemon navigation worker exited"));
168     if (this.pending.has(key) || this.completed.has(key)) {
169       return Promise.reject(new Error(`Duplicate daemon navigation worker request: ${key}`));
170     }
171     const response = new Promise<DaemonNavigationWorkerResponse>((resolve, reject) => {
172       this.pending.set(key, {
173         resolve,
174         reject,
175         appendOutput:
176           appendOutput ?? (() => Promise.reject(new Error("Worker output sink is unavailable"))),
177         nextSequence: 0,
178         chunkInFlight: false,
179       });
180     });
181     this.worker.postMessage(DaemonNavigationWorkerProtocol.request(request));
182     return response;
183   }
184 
185   private async receive(value: unknown): Promise<void> {
186     let response: DaemonNavigationWorkerResponse;
187     try {
188       response = DaemonNavigationWorkerProtocol.response(value, this.maximumChunkRawBytes);
189     } catch (error) {
190       this.failCommunication(error);
191       return;
192     }
193     if (response.generation !== this.generation) return;
194     if (response.kind === "output-chunk") {
195       await this.receiveOutputChunk(response);
196       return;
197     }
198     const key = NodeDaemonNavigationWorker.responseKey(response);
199     if (this.completed.has(key)) {
200       this.failCommunication(new Error(`Duplicate daemon navigation worker response: ${key}`));
201       return;
202     }
203     const pending = this.pending.get(key);
204     if (pending === undefined) {
205       this.failCommunication(new Error(`Uncorrelated daemon navigation worker response: ${key}`));
206       return;
207     }
208     this.pending.delete(key);
209     this.completed.add(key);
210     if (response.kind === "failed") {
211       pending.reject(new Error(`Daemon navigation worker ${response.failureCode} failure`));
212       return;
213     }
214     if (response.kind === "closed") this.closeAcknowledged = true;
215     pending.resolve(response);
216   }

218   private async receiveOutputChunk(
219     response: Extract<DaemonNavigationWorkerResponse, { kind: "output-chunk" }>,
220   ): Promise<void> {
221     const key = `execute:${response.requestId}`;
222     const pending = this.pending.get(key);
223     if (
224       pending === undefined ||
225       pending.chunkInFlight ||
226       response.sequence !== pending.nextSequence
227     ) {
228       this.failCommunication(new Error(`Unexpected daemon navigation worker output: ${key}`));
229       return;
230     }
231     pending.chunkInFlight = true;
232     try {
233       const record: DaemonSequencedOutputRecord = {
234         sequence: response.sequence,
235         stream: response.stream,
236         bytes: response.bytes,
237       };
238       await pending.appendOutput(record);
239       pending.nextSequence += 1;
240       pending.chunkInFlight = false;
241       const acknowledgement: DaemonNavigationWorkerRequest = {
242         kind: "output-ack",
243         generation: this.generation,
244         requestId: response.requestId,
245         sequence: response.sequence,
246       };
247       this.worker.postMessage(DaemonNavigationWorkerProtocol.request(acknowledgement));
248     } catch (error) {
249       this.failCommunication(error);
250     }
251   }

253   private failCommunication(error: unknown): void {
254     if (this.exit !== undefined) return;
255     const failure = error instanceof Error ? error : new Error(String(error));
256     this.communicationFailure = failure;
257     this.communicationFailureCode =
258       typeof error === "object" &&
259       error !== null &&
260       "code" in error &&
261       typeof error.code === "string"
262         ? error.code
263         : undefined;
264     void this.worker.terminate();
265   }
266 
267   private finishExit(): void {
268     if (this.exit !== undefined) return;
269     const communicationErrorName = this.communicationFailure?.name;
270     const communicationFailureIdentity = this.communicationFailureCode ?? communicationErrorName;
271     const cause = communicationFailureIdentity
272       ? communicationFailureIdentity === "ERR_WORKER_OUT_OF_MEMORY"
273         ? "out-of-memory"
274         : "error"
275       : this.terminating
276         ? "terminated"
277         : this.closeAcknowledged
278           ? "closed"
279           : "error";
280     this.exit = {
281       generation: this.generation,
282       cause,
283       ...(communicationFailureIdentity === undefined
284         ? {}
285         : { errorName: communicationFailureIdentity }),
286     };
287     const failure = new DaemonNavigationWorkerExitedError(
288       this.exit,
289       this.communicationFailure?.message,
290     );
291     for (const pending of this.pending.values()) pending.reject(failure);
292     this.pending.clear();
293     this.resolveExited(this.exit);
294   }

296   private static responseKey(response: DaemonNavigationWorkerResponse): string {
297     if (response.kind === "ready") return "initialize";
298     if (response.kind === "result") return `execute:${response.requestId}`;
299     if (response.kind === "heap") return `release-transient:${response.operationId}`;
300     if (response.kind === "closed") return "close";
301     if (response.kind === "failed" && response.operationId !== undefined) {
302       return `release-transient:${response.operationId}`;
303     }
304     return response.requestId === undefined ? "initialize" : `execute:${response.requestId}`;
305   }
