196 class RecordingDaemonSocketClient implements DaemonSocketClient {
197   readonly calls: { readonly endpoint: string; readonly timeoutMs?: number }[] = [];
198   private nextConnection = 0;
199 
200   constructor(private readonly connections: readonly DaemonSocketConnection[]) {}
201 
202   connect(endpoint: string, timeoutMs?: number): Promise<DaemonSocketConnection> {
203     this.calls.push({ endpoint, ...(timeoutMs === undefined ? {} : { timeoutMs }) });
204     const connection = this.connections[this.nextConnection];
205     this.nextConnection += 1;
206     if (connection === undefined) return Promise.reject(new Error("Unexpected daemon connection"));
207     return Promise.resolve(connection);
208   }
209 }

211 class ScriptedDaemonSocketConnection implements DaemonSocketConnection {
212   readonly writes: Uint8Array[] = [];
213   disableTimeoutCount = 0;
214   endCount = 0;
215   destroyCount = 0;
216   readonly incoming: AsyncIterable<Uint8Array>;
217 
218   constructor(bytes: readonly Uint8Array[], terminalError?: Error) {
219     this.incoming = ScriptedDaemonSocketConnection.stream(bytes, terminalError);
220   }
221 
222   write(frame: Uint8Array): void {
223     this.writes.push(frame);
224   }
225 
226   disableTimeout(): void {
227     this.disableTimeoutCount += 1;
228   }
229 
230   end(): void {
231     this.endCount += 1;
232   }
233 
234   destroy(): void {
235     this.destroyCount += 1;
236   }
237 
238   private static async *stream(
239     bytes: readonly Uint8Array[],
240     terminalError?: Error,
241   ): AsyncIterable<Uint8Array> {
242     yield* bytes;
243     if (terminalError !== undefined) throw terminalError;
244   }
