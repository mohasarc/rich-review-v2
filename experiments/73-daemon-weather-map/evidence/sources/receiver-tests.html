180 class FakeCapture implements DaemonOutputCapture {
181   readonly appended: DaemonSequencedOutputRecord[] = [];
182   readonly finishExitCodes: number[] = [];
183   readonly resultOutput = new FakeOutput();
184   readonly result: DaemonExecutorExecutionResult = { exitCode: 0, output: this.resultOutput };
185   disposeCalls = 0;
186 
187   constructor(
188     private readonly capturedSummary: DaemonCapturedOutputSummary,
189     private readonly appendOperation: () => Promise<void> = () => Promise.resolve(),
190   ) {}
191 
192   async append(record: DaemonSequencedOutputRecord): Promise<void> {
193     await this.appendOperation();
194     this.appended.push({ ...record, bytes: Buffer.from(record.bytes) });
195   }
196 
197   finish(exitCode: number): Promise<DaemonCapturedOutput> {
198     this.finishExitCodes.push(exitCode);
199     return Promise.resolve({
200       result: this.result,
201       summary: this.capturedSummary,
202     });
203   }
204 
205   dispose(): Promise<void> {
206     this.disposeCalls += 1;
207     return Promise.resolve();
208   }
209 }

264 function chunk(sequence: number, stream: "stdout" | "stderr" = "stdout"): DaemonResultChunk {
265   return {
266     transferId: "transfer",
267     requestId: "request",
268     offset: sequence,
269     sequence,
270     stream,
271     bytes: Buffer.from(String.fromCharCode(97 + sequence)),
272   };
273 }

275 function summary(records: readonly DaemonResultChunk[]): DaemonCapturedOutputSummary {
276   const hash = createHash("sha256");
277   let rawBytes = 0;
278   for (const record of records) {
279     const encoded = Buffer.alloc(5 + record.bytes.byteLength);
280     encoded.writeUInt8(record.stream === "stdout" ? 0 : 1, 0);
281     encoded.writeUInt32BE(record.bytes.byteLength, 1);
282     Buffer.from(record.bytes).copy(encoded, 5);
283     hash.update(encoded);
284     rawBytes += record.bytes.byteLength;
285   }
286   return { rawBytes, recordCount: records.length, sha256: hash.digest("hex") };
