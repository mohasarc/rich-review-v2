167   async execute(request: DaemonClientExecuteRequest): Promise<DaemonClientExecuteResult> {
168     if (!this.options.daemonEnabled) return this.executeLocally(request, "cold");
169     const identity = DaemonWorkspaceIdentity.from(
170       request.workspaceRoot,
171       this.options.stateDirectory,
172     );
173     const route = await this.routing.decide(
174       new DaemonRoutingContextState(
175         identity,
176         this.options.productVersion,
177         () => this.registry.read(identity),
178         (record) => this.observer.observe(record),
179         (record) => this.registry.removeIfProcess(identity, record.instanceId, record.processToken),
180       ),
181     );
182     if (route.kind === "warm") return this.executeWarm(route.record, request);
183     if ((route.kind === "cold" && route.reason === "absent") || route.kind === "fallback") {
184       DaemonClientRuntime.triggerIndependently(this.coordinator, identity);
185     }
186     return this.executeLocally(request, route.kind === "fallback" ? "fallback" : "cold");
187   }

209   private async executeWarm(
210     record: DaemonRecord,
211     request: DaemonClientExecuteRequest,
212   ): Promise<DaemonClientExecuteResult> {
213     let receipt: Awaited<ReturnType<DaemonClientTransport["execute"]>>;
214     try {
215       receipt = await this.routingTransport.execute(record.endpoint, {
216         kind: "execute",
217         protocolVersion: DAEMON_PROTOCOL_VERSION,
218         instanceId: record.instanceId,
219         processToken: record.processToken,
220         requestId: randomUUID(),
221         commandName: request.commandName,
222         request: DaemonClientRuntime.executorRequest(request, "warm"),
223       });
224     } catch (error) {
225       if (error instanceof DaemonTransportError && error.retrySafe) {
226         return this.executeLocally(request, "fallback");
227       }
228       return { mode: "warm", result: DaemonControlledResult.acceptedRequestDidNotComplete() };
229     }
230     try {
231       const completion = await receipt.completion;
232       if (completion.status === "failed") {
233         return { mode: "warm", result: DaemonClientRuntime.controlledFailure(completion.code) };
234       }
235       if (!DaemonClientRuntime.isCompleteResult(completion.result)) {
236         await DaemonClientRuntime.disposeMalformedOutput(completion.result);
237         return { mode: "warm", result: DaemonControlledResult.acceptedRequestDidNotComplete() };
238       }
239       return { mode: "warm", result: completion.result };
240     } catch {
241       return { mode: "warm", result: DaemonControlledResult.acceptedRequestDidNotComplete() };
242     }
243   }

245   private async executeLocally(
246     request: DaemonClientExecuteRequest,
247     mode: "cold" | "fallback",
248   ): Promise<DaemonClientExecuteResult> {
249     const executor = await this.options.executorFactory({
250       stateDirectory: this.options.stateDirectory,
251       productVersion: this.options.productVersion,
252       sampleResources: () => undefined,
253     });
254     const result = await executor.execute(DaemonClientRuntime.executorRequest(request, mode));
255     return { mode, result };
256   }

291   private static controlledFailure(
292     code: DaemonExecutionFailureCode,
293   ): DaemonExecutorExecutionResult {
294     if (code === "controlled-resource") return DaemonControlledResult.workspaceCapacityExceeded();
295     if (code === "response-capacity") return DaemonControlledResult.responseCapacityExceeded();
