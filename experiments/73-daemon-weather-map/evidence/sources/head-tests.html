1483 async function rawExecutionServer(
1484   servers: Server[],
1485   sockets: Socket[],
1486   directories: string[],
1487   connected: (socket: Socket) => void,
1488 ): Promise<string> {
1489   const endpoint = executionEndpoint(directories);
1490   const server = createServer((socket) => {
1491     sockets.push(socket);
1492     socket.on("error", (error: NodeJS.ErrnoException) => {
1493       if (error.code === "EPIPE" || error.code === "ECONNRESET") return;
1494       throw error;
1495     });
1496     connected(socket);
1497   });
1498   servers.push(server);
1499   await new Promise<void>((resolve, reject) => {
1500     server.once("error", reject);
1501     server.listen(endpoint, resolve);
1502   });
1503   return endpoint;
1504 }

1522 async function settleWithin<T>(operation: Promise<T>, milliseconds: number): Promise<T> {
1523   let timeout: NodeJS.Timeout | undefined;
1524   const expired = new Promise<never>((_resolve, reject) => {
1525     timeout = setTimeout(() => reject(new Error("Operation did not settle")), milliseconds);
1526   });
1527   try {
1528     return await Promise.race([operation, expired]);
1529   } finally {
1530     if (timeout !== undefined) clearTimeout(timeout);
1531   }
1532 }
