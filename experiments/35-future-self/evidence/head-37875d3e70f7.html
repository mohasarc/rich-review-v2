173class AcceptedExecutionHarness {
174 readonly events: string[] = [];

177 readonly workerExecute = vi.fn(

179 _requestId: string,
180 _request: unknown,
181 output: { append(record: DaemonSequencedOutputRecord): Promise<void> },
182 ) => {
183 await output.append({ sequence: 0, stream: "stdout", bytes: new Uint8Array([1]) });
184 this.events.push("worker-observed-append");
185 if (this.options.workerFailure !== undefined) throw this.options.workerFailure;
186 return {
187 kind: "result" as const,
188 generation: 7,
189 requestId: _requestId,
190 result: { exitCode: 0 },
191 refresh: { added: 0, changed: 0, removed: 0, unchanged: 1 },
192 durations: { freshnessMs: 1, navigationMs: 2, renderMs: 3, outputMs: 4 },
193 resources: {
194 workerHeapUsedBytes: 11,
195 peakWorkerHeapUsedBytes: 13,
196 workerHeapLimitBytes: 20,
197 },
198 };
199 },
200 );

206 constructor(private readonly options: AcceptedExecutionHarnessOptions = {}) {
207 const ledger = new AcceptedRequestLedger({ wallNowMs: () => this.wallNowMs() });
208 this.ledger = ledger;

210 const completion: DaemonCompletionWriter = {
211 append: async (record) => {
212 this.events.push(`output-appended:${record.sequence}`);
213 },
214 finish: async (exitCode) => {
215 this.events.push(`completion-finished:${exitCode}`);

217 transferId: "transfer",
218 requestId: "request",
219 instanceId: "instance",
220 exitCode,
221 rawBytes: 1,
222 recordCount: 1,
223 sha256: "hash",
224 };
225 },
226 dispose: async () => {

228 if (this.options.completionDisposeFailure !== undefined) {
229 throw this.options.completionDisposeFailure;
230 }
231 },
232 };
233 const trace: DaemonOperationTrace = {
234 accepted: (queuePosition, generation) =>
235 this.events.push(`trace-accepted:${queuePosition}:${generation}`),
236 turnStarted: (generation) => this.events.push(`turn-started:${generation}`),
237 workerCompleted: () => this.events.push("worker-completed"),
238 spooled: () => undefined,
239 executionTerminated: (outcome) => this.events.push(`execution-${outcome}`),
240 clientDisconnected: () => undefined,

242 deliveryTerminated: () => undefined,

246 trace.accepted(queuePosition, generation);

250 this.events.push("completion-created");

255 return Promise.resolve();

263 snapshot: { generation: 7 },
264 workerHeapReported: (generation: number, used: number, limit: number, peak: number) => {
265 this.events.push(`worker-heap-reported:${generation}:${used}:${limit}:${peak}`);
266 },
267 sampleAtTurnBoundary: async () => {
268 this.events.push("resource-sampled");
269 if (this.options.resourceSampleFailure !== undefined) {
270 throw this.options.resourceSampleFailure;
271 }
272 },
273 } as unknown as DaemonResourceSupervisor;
274 const processLifecycle: AcceptedExecutionProcessLifecycle = {
275 shutdownSnapshot: () => ({ started: false }),

280 workspaceDeletedAfterDelivery: async () => {
281 this.events.push("workspace-deleted-after-delivery");
282 },
283 };

293 resourceSupervisor,
294 processLifecycle,
295 lifetime,
296 diagnostics: { record: (event) => this.diagnostics.push(event) },
297 clock: {

299 monotonicNowMs: () => this.monotonicNow++,
300 },
301 });
302 }
