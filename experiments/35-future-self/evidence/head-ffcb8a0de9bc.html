73 async stop(workspaceRoot: string): Promise<DaemonStopResult> {
74 const stopStartedAt = this.clock.wallNowMs();
75 const deadline = stopStartedAt + this.stopTimeoutMs;
76 const forceWaitMs = Math.min(
77 this.policy.shutdown.forcedTerminationReserveMaximumMs,
78 Math.floor(this.stopTimeoutMs / 2),
79 );
80 const gracefulDeadline = deadline - forceWaitMs;
81 const identity = DaemonWorkspaceIdentity.from(workspaceRoot, this.stateDirectory);
82 const record = this.registry.read(identity);
83 if (record === undefined) {
84 return { status: "not-running", workspaceRoot };
85 }
86 if (record.state === "starting") return this.stopStarting(identity, record, deadline);
87 const observation = await this.observer.observeIdentity(record);
88 if (observation.kind === "exited") {
89 this.registry.removeIfProcess(identity, record.instanceId, record.processToken);
90 return { status: "not-running", workspaceRoot };
91 }
92 if (observation.kind === "unresponsive" || observation.kind === "starting") {
93 throw new Error(`Daemon process ${record.pid} is live but unresponsive`);
94 }
95
96 const stopRequest = this.transport
97 .request(record.endpoint, {
98 kind: "stop",
99 protocolVersion: DAEMON_PROTOCOL_VERSION,
100 instanceId: record.instanceId,
101 })
102 .then(() => true)
103 .catch(() => false);
104 const acknowledged = await Promise.race([
105 stopRequest,
106 this.pause(Math.max(0, gracefulDeadline - this.clock.wallNowMs())).then(() => false),
107 ]);
108 if (acknowledged && (await this.waitForExit(record.pid, gracefulDeadline))) {
109 this.registry.removeIfProcess(identity, record.instanceId, record.processToken);
110 return { status: "stopped", workspaceRoot, pid: record.pid };
111 }
112
113 const killed = await this.killIdentified(record, deadline);
114 if (!killed || !(await this.waitForIdentifiedExit(record, deadline))) {
115 throw new Error(`Daemon process ${record.pid} did not exit after authenticated kill`);
116 }
117 this.registry.removeIfProcess(identity, record.instanceId, record.processToken);
118 return { status: "killed", workspaceRoot, pid: record.pid };
119 }

121 private async statusForRecord(record: DaemonRecord): Promise<RunningDaemonStatus | undefined> {
122 const identity = DaemonWorkspaceIdentity.from(record.workspaceRoot, this.stateDirectory);
123 if (record.state === "starting") {
124 if (record.pid > 0) {
125 const observation = await this.observer.observe(record);
126 if (observation.kind === "exited") {
127 this.removeStartingOwnership(identity, record);
128 return undefined;
129 }
130 if (observation.kind === "starting") return this.startingStatus(record);
131 if (observation.kind === "responsive" && observation.pong.activity === undefined) {
132 return this.startingStatus(record);
133 }
134 return this.statusForObservation(observation);
135 }
136 const ownerForRecord = this.registry.startupOwnerForRecordCredentials(identity, record);
137 const ownerForInstance = this.registry.startupOwnerForInstance(identity, record.instanceId);
138 const armedLaunchIsWithinGrace =
139 ownerForRecord !== undefined && this.registry.startupOwnerIsWithinGrace(ownerForRecord);
140 if (
141 armedLaunchIsWithinGrace ||
142 (ownerForInstance !== undefined &&
143 this.processTerminator.isAlive(ownerForInstance.ownerPid) &&
144 this.registry.startupOwnerIsWithinGrace(ownerForInstance))
145 ) {
146 return this.startingStatus(record);
147 }
148 if (ownerForInstance !== undefined) {
149 if (!this.registry.removeStartupLockIfOwner(identity, ownerForInstance)) {
150 const renewedOwner = this.registry.startupOwnerForInstance(identity, record.instanceId);
151 if (
152 renewedOwner !== undefined &&
153 this.processTerminator.isAlive(renewedOwner.ownerPid) &&
154 this.registry.startupOwnerIsWithinGrace(renewedOwner)
155 ) {
156 return this.startingStatus(record);
157 }
158 return undefined;
159 }
160 }
161 this.registry.removeIfProcess(identity, record.instanceId, record.processToken);
162 return undefined;
163 }
164
165 return this.statusForObservation(await this.observer.observe(record));
166 }

168 private async stopStarting(
169 identity: DaemonWorkspaceIdentity,
170 record: DaemonRecord,
171 deadline: number,
172 ): Promise<DaemonStopResult> {
173 if (record.pid <= 0) {
174 if (this.registry.startupOwnerForRecordCredentials(identity, record) !== undefined) {
175 return this.waitForClaimedProcessAndStop(identity, record, deadline);
176 }
177 this.registry.removeStartupLockIfInstance(identity, record.instanceId);
178 this.registry.removeIfProcess(identity, record.instanceId, record.processToken);
179 return { status: "not-running", workspaceRoot: record.workspaceRoot };
180 }
181
182 if (!this.processTerminator.isAlive(record.pid)) {
183 this.removeStartingOwnership(identity, record);
184 return { status: "not-running", workspaceRoot: record.workspaceRoot };
185 }
186 if (!this.registry.startupOwnerMatchesProcess(identity, record)) {
187 throw new Error(`Daemon process ${record.pid} has no authenticated startup ownership`);
188 }
189 const terminated = await this.terminateStartingProcess(record.pid, deadline);
190 if (!terminated || this.processTerminator.isAlive(record.pid)) {
191 throw new Error(`Daemon process ${record.pid} did not exit after authenticated stop`);
192 }
193 this.registry.removeStartupLockIfProcess(identity, record);
194 this.registry.removeIfProcess(identity, record.instanceId, record.processToken);
195 return { status: "stopped", workspaceRoot: record.workspaceRoot, pid: record.pid };
196 }

198 private async waitForClaimedProcessAndStop(
199 identity: DaemonWorkspaceIdentity,
200 claimedRecord: DaemonRecord,
201 deadline: number,
202 ): Promise<DaemonStopResult> {
203 while (this.clock.wallNowMs() <= deadline) {
204 const record = this.registry.readStoredInstance(identity, claimedRecord.instanceId);
205 if (record === undefined || record.processToken !== claimedRecord.processToken) {
206 return { status: "not-running", workspaceRoot: claimedRecord.workspaceRoot };
207 }
208 if (record.pid > 0) return this.stopStarting(identity, record, deadline);
209 if (this.registry.startupOwnerForRecordCredentials(identity, claimedRecord) === undefined) {
210 return { status: "not-running", workspaceRoot: claimedRecord.workspaceRoot };
211 }
212 await this.pause(this.pollIntervalMs);
213 }
214 throw new Error("Daemon launch did not publish its process before authenticated stop");
215 }

217 private async terminateStartingProcess(pid: number, deadline: number): Promise<boolean> {
218 let timeout: NodeJS.Timeout | undefined;
219 const deadlineReached = new Promise<false>((resolve) => {
220 timeout = setTimeout(() => resolve(false), Math.max(0, deadline - this.clock.wallNowMs()));
221 });
222 try {
223 return await Promise.race([
224 this.processTerminator.terminate(pid).then(() => true),
225 deadlineReached,
226 ]);
227 } finally {
228 if (timeout !== undefined) clearTimeout(timeout);
229 }
230 }

232 private removeStartingOwnership(identity: DaemonWorkspaceIdentity, record: DaemonRecord): void {
233 const storedRecord = this.registry.readStoredInstance(identity, record.instanceId);
234 if (
235 storedRecord?.processToken !== record.processToken ||
236 storedRecord.pid !== record.pid ||
237 storedRecord.startedAt !== record.startedAt
238 ) {
239 return;
240 }
241 if (this.registry.startupOwnerMatchesProcess(identity, record)) {
242 this.registry.removeStartupLockIfProcess(identity, record);
243 }
244 this.registry.removeIfProcess(identity, record.instanceId, record.processToken);

257 private statusForObservation(observation: DaemonObservation): RunningDaemonStatus | undefined {
258 const record = observation.record;
259 if (observation.kind === "exited") {
260 const identity = DaemonWorkspaceIdentity.from(record.workspaceRoot, this.stateDirectory);
261 this.registry.removeIfProcess(identity, record.instanceId, record.processToken);
262 return undefined;
263 }
264 if (observation.kind !== "responsive") return this.unresponsiveStatus(record);
265 if (observation.pong.activity !== undefined) {
266 return this.statusFromActivity(record, observation.pong.activity);
267 }
268 const lastNavigationAt = observation.pong.lastNavigationAt ?? record.lastNavigationAt;
269 const fileCount = observation.pong.fileCount ?? record.fileCount;
270 const memoryBytes = observation.pong.memoryBytes ?? record.memoryBytes;
271 if (observation.pong.state === "busy") {
272 if (
273 observation.pong.currentCommand === undefined ||
274 observation.pong.currentCommandElapsedMs === undefined ||
275 observation.pong.queued === undefined ||
276 memoryBytes === undefined
277 ) {
278 return this.unresponsiveStatus(record);
279 }
280 return {
281 workspaceRoot: record.workspaceRoot,
282 state: "busy",
283 pid: record.pid,
284 uptimeMs: Math.max(0, this.clock.wallNowMs() - record.startedAt),
285 command: DaemonRuntimeValues.isCommandName(observation.pong.currentCommand)
286 ? observation.pong.currentCommand
287 : "unknown",
288 elapsedMs: observation.pong.currentCommandElapsedMs,
289 queued: observation.pong.queued,
290 memoryBytes,
291 };
292 }
293 if (fileCount === undefined || memoryBytes === undefined) {
294 return this.unresponsiveStatus(record);
295 }
296 return {
297 workspaceRoot: record.workspaceRoot,
298 state: "ready",
299 pid: record.pid,
300 uptimeMs: Math.max(0, this.clock.wallNowMs() - record.startedAt),
301 fileCount,
302 memoryBytes,
303 ...(lastNavigationAt === undefined
304 ? {}
305 : { lastRequestAgoMs: Math.max(0, this.clock.wallNowMs() - lastNavigationAt) }),
306 };
307 }

324 state: "starting",

327 startupElapsedMs: activity.startupElapsedMs,
328 memoryBytes: activity.processRssBytes,
329 };
330 }
331 const uptimeMs = activity.startupElapsedMs;
332 if (activity.lifecycle === "busy" && activity.current !== undefined) {
333 return {
334 state: "busy",
335 workspaceRoot: record.workspaceRoot,
336 pid: record.pid,
337 uptimeMs,

339 elapsedMs: activity.current.elapsedMs,
340 queued: activity.queued,
341 memoryBytes: activity.processRssBytes,
342 };
343 }

345 const detail = activity.lifecycle === "draining" ? "draining" : activity.recoveryDetail;
346 if (detail === undefined) return this.unresponsiveStatus(record);
347 return {
348 state: "recovering",

350 pid: record.pid,

354 memoryBytes: activity.processRssBytes,
355 };
356 }

359 state: "ready",
360 workspaceRoot: record.workspaceRoot,

363 fileCount: activity.fileCount,

372 try {

375 kind: "kill",

387 private async waitForIdentifiedExit(record: DaemonRecord, deadline: number): Promise<boolean> {
388 while (this.clock.wallNowMs() <= deadline) {
389 if (!(await this.identifies(record)) && !this.processTerminator.isAlive(record.pid)) {
390 return true;
391 }
392 await this.pause(this.pollIntervalMs);
393 }
394 return !(await this.identifies(record)) && !this.processTerminator.isAlive(record.pid);
395 }

414 private async waitForExit(pid: number, deadline: number): Promise<boolean> {
415 while (this.clock.wallNowMs() <= deadline) {
416 if (!this.processTerminator.isAlive(pid)) return true;
417 await this.pause(this.pollIntervalMs);
418 }
419 return !this.processTerminator.isAlive(pid);
