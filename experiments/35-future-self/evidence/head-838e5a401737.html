57export class DaemonWorkerGenerationManager {

84 if (this.startOperation !== undefined) return this.startOperation;
85 const generation = this.currentGeneration;
86 generation.ready = this.startGeneration(generation);
87 this.startOperation = this.waitForReadyGeneration(generation).then((report) => {
88 this.publishProtocolReady(generation, report);

91 return this.startOperation;

94 activateReadiness(): void {
95 if (this.protocolReadyGeneration !== this.currentGeneration) {
96 throw new Error("Navigation worker protocol readiness is unavailable");
97 }
98 this.readinessActivated = true;
99 this.workerReady = true;

103 requestId: string,
104 request: DaemonWorkerExecuteRequest,
105 output: DaemonOutputSink,

115 response.kind !== "result" ||
116 response.requestId !== requestId ||
117 response.generation !== generation.worker.generation

120 }
121 return response;
122 });
123 }

126 if (this.replacementOperation !== undefined) return this.replacementOperation;
127 const operation = this.replaceGeneration(cause);

134 }

137 const response = await this.currentGeneration.worker.releaseTransientResources();
138 if (response.kind !== "heap") {
139 throw new Error("Navigation worker did not report heap usage");

141 this.options.onDiagnostic({

143 workerGeneration: response.generation,

150 close(): Promise<void> {
151 if (this.closeOperation !== undefined) return this.closeOperation;
152 this.stopping = true;
153 this.workerReady = false;

156 }

158 terminate(): Promise<void> {
159 if (this.terminationOperation !== undefined) return this.terminationOperation;
160 this.stopping = true;
161 this.workerReady = false;
162 this.terminationOperation = this.currentGeneration.worker.terminate();
163 return this.terminationOperation;
164 }

166 private async replaceGeneration(
167 cause: DaemonWorkerReplacementCause,
168 ): Promise<DaemonWorkerReadyReport> {
169 const previous = this.currentGeneration;
170 if (cause !== "worker-exit") this.options.onActiveResourceInterruption(cause);
171 this.workerReady = false;
172 this.protocolReadyGeneration = undefined;
173 const nextWorker = this.options.createWorker(previous.worker.generation + 1);
174 if (nextWorker.generation !== previous.worker.generation + 1) {
175 throw new Error("Navigation worker factory returned the wrong generation");
176 }
177 const next = this.createGeneration(nextWorker);
178 this.currentGeneration = next;
179 next.ready = this.startGeneration(next);
180 await previous.worker.terminate().catch(() => undefined);
181 const report = await next.ready;
182 this.publishProtocolReady(next, report);
183 this.options.onDiagnostic({
184 kind: "worker-replaced",
185 cause,
186 previousWorkerGeneration: previous.worker.generation,
187 workerGeneration: next.worker.generation,
188 fileCount: report.fileCount,
189 ...report.startupDurations,
190 });
191 return report;
192 }

194 private startGeneration(generation: DaemonWorkerGeneration): Promise<DaemonWorkerReadyReport> {
195 return generation.worker
196 .start(this.options.workspaceRoot)
197 .then((response): DaemonWorkerReadyReport => {
198 if (response.kind !== "ready" || response.generation !== generation.worker.generation) {
199 throw new Error("Navigation worker did not become ready");
200 }
201 return response;
202 });
203 }

205 private async waitForReadyGeneration(
206 generation: DaemonWorkerGeneration,
207 ): Promise<DaemonWorkerReadyReport> {
208 try {
209 if (generation.ready === undefined) throw new Error("Navigation worker is not starting");
210 return await generation.ready;
211 } catch (error) {
212 if (!(error instanceof DaemonNavigationWorkerExitedError)) throw error;
213 await generation.worker.exited;
214 const recovery = this.recoveryOperation;
215 if (recovery === undefined) throw error;
216 await recovery;
217 const replacement = this.currentGeneration;
218 if (replacement === generation || replacement.ready === undefined) throw error;
219 return replacement.ready;
220 }
221 }

223 private createGeneration(worker: DaemonNavigationWorker): DaemonWorkerGeneration {
224 const generation: DaemonWorkerGeneration = { worker, ready: undefined };
225 void worker.exited.then((exit) => this.observeExit(generation, exit));
226 return generation;
227 }
228
229 private observeExit(generation: DaemonWorkerGeneration, exit: DaemonNavigationWorkerExit): void {
230 if (this.stopping || generation !== this.currentGeneration) return;
231 const recovery = this.options.exitRecovery.recover(exit);
232 this.recoveryOperation = recovery;
233 void recovery.catch(() => undefined);
234 }

236 private publishProtocolReady(
237 generation: DaemonWorkerGeneration,
238 report: DaemonWorkerReadyReport,
239 ): void {
240 if (generation !== this.currentGeneration) return;
241 this.protocolReadyGeneration = generation;
242 this.fileCount = report.fileCount;
243 if (this.readinessActivated) this.workerReady = true;
244 }

246 private clearReplacement(operation: Promise<DaemonWorkerReadyReport>): void {
247 if (this.replacementOperation === operation) this.replacementOperation = undefined;
