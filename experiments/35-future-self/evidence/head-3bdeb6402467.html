104 private async sampleWithinBoundary(
105 runAtBoundary: (operation: () => Promise<void>) => Promise<void>,
106 ): Promise<void> {
107 if (this.currentState === "draining" || this.currentState === "stopped") return;
108 this.captureUsage();
109 const policy = this.options.policy;
110 if (this.currentProcessRssBytes >= policy.hardProcessRssBytes) {
111 await this.replace("hard-pressure");
112 return;
113 }
114 if (this.currentProcessRssBytes <= policy.resumeProcessRssBytes) {
115 this.admissionPaused = false;
116 this.shedCompleted = false;
117 this.currentState = "ready";
118 return;
119 }
120 if (this.currentProcessRssBytes < policy.softProcessRssBytes) return;
121 this.admissionPaused = true;
122 this.currentState = "shedding";
123 if (this.shedCompleted) return;

174 this.workerHeapUsedBytes = undefined;

179 this.admissionPaused = false;
