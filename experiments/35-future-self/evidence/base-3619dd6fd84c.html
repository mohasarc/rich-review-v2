377class GatedWritable extends Writable {
378 private readonly callbacks: Array<(error?: Error | null) => void> = [];
379
380 constructor(
381 private readonly writes: string[],
382 private readonly name: string,
383 ) {
384 super({ highWaterMark: 1 });
385 }
386
387 get pendingCount(): number {
388 return this.callbacks.length;
389 }
390
391 release(): void {
392 this.callbacks.shift()?.();
393 }
394
395 fail(error: Error): void {
396 this.destroy(error);
397 }
398
399 override _write(
400 chunk: Buffer,
401 _encoding: BufferEncoding,
402 callback: (error?: Error | null) => void,
403 ): void {
404 this.writes.push(`${this.name}:${chunk.toString()}`);
405 this.callbacks.push(callback);
406 }
407}
