742class ControllerTransport {
743 readonly live = new Set<string>();
744 readonly killed: string[] = [];
745 onStop: (() => void) | undefined;
746 onKill: (() => void) | undefined;
747
748 constructor(private readonly registry: DaemonRegistry) {}
749
750 execute(): Promise<never> {
751 return Promise.reject(new Error("Registry observation must not execute"));
752 }
753
754 async request(_endpoint: string, request: DaemonRequest): Promise<DaemonResponse> {
755 const records = this.registry.list();
756 const record = records.find((candidate) => candidate.instanceId === request.instanceId);
757 if (record === undefined || !this.live.has(request.instanceId)) throw new Error("unreachable");
758 if (request.kind === "identify") {
759 return {
760 kind: "identity",
761 instanceId: record.instanceId,
762 processToken: record.processToken,
763 pid: record.pid,
764 startedAt: record.startedAt,
765 };
766 }
767 if (request.kind === "ping") {
768 return {
769 kind: "pong",
770 protocolVersion: DAEMON_PROTOCOL_VERSION,
771 instanceId: record.instanceId,
772 symnavVersion: record.symnavVersion,
773 startedAt: record.startedAt,
774 fileCount: record.fileCount ?? 0,
775 memoryBytes: 1234,
776 ...(record.lastNavigationAt === undefined
777 ? {}
778 : { lastNavigationAt: record.lastNavigationAt }),
779 };
780 }
781 if (request.kind === "stop") {
782 this.onStop?.();
783 return { kind: "stopped", instanceId: record.instanceId };
784 }
785 if (request.kind === "kill") {
786 this.onKill?.();
787 this.killed.push(record.instanceId);
788 return {
789 kind: "killing",
790 instanceId: record.instanceId,
791 processToken: record.processToken,
792 };
793 }
794 throw new Error(`Unsupported controller request ${request.kind}`);
795 }
796
797 async removeUnavailableEndpoint(): Promise<boolean> {
798 return true;
799 }
800}

848function waitForProcess(child: ChildProcess): Promise<void> {
849 if (child.exitCode !== null) {
850 return child.exitCode === 0
851 ? Promise.resolve()
852 : Promise.reject(new Error(`Registry cleaner exited with code ${String(child.exitCode)}`));
853 }
854 if (child.signalCode !== null) {
855 return Promise.reject(
856 new Error(`Registry cleaner exited with signal ${String(child.signalCode)}`),
857 );
858 }
859 return new Promise((resolve, reject) => {
860 child.once("error", reject);
861 child.once("exit", (code) => {
862 if (code === 0) resolve();
863 else reject(new Error(`Registry cleaner exited with code ${String(code)}`));
864 });
865 });
866}

868async function waitUntil(predicate: () => boolean): Promise<void> {
869 const deadline = Date.now() + 5_000;
870 while (Date.now() <= deadline) {
871 if (predicate()) return;
872 await new Promise((resolve) => setTimeout(resolve, 2));
873 }
874 throw new Error("Timed out waiting for registry race");
875}
