557class DeliverySend {
558 readonly frames: DaemonServerMessage[] = [];
559 readonly resultEndReceived: Promise<void>;
560 private readonly closeListeners = new Set<() => void>();
561 private resolveResultEndReceived!: () => void;
562 readonly send: DaemonServerSend;
563
564 private constructor(onMessage?: (message: DaemonServerMessage) => void | Promise<void>) {
565 this.resultEndReceived = new Promise((resolve) => {
566 this.resolveResultEndReceived = resolve;
567 });
568 this.send = Object.assign(
569 async (message: DaemonServerMessage) => {
570 await onMessage?.(message);
571 this.frames.push(message);
572 if ("kind" in message && message.kind === "result-end") {
573 this.resolveResultEndReceived();
574 }
575 },
576 {
577 onClose: (listener: () => void) => {
578 this.closeListeners.add(listener);
579 return () => this.closeListeners.delete(listener);
580 },
581 },
582 );
583 }
584
585 static create(onMessage?: (message: DaemonServerMessage) => void | Promise<void>): DeliverySend {
586 return new DeliverySend(onMessage);
587 }
588
589 close(): void {
590 for (const listener of this.closeListeners) listener();
591 }

593 static frameKind(frame: DaemonServerMessage): string {
594 return "kind" in frame ? frame.kind : "chunk";
