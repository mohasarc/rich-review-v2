29class DaemonNavigationWorkerEntry {
30 private executor: DaemonExecutor | undefined;
31 private activeHeapMonitor: WorkerHeapHighWater | undefined;
32 private tail: Promise<void> = Promise.resolve();
33 private readonly outputAcknowledgements = new Map<string, () => void>();
34 private readonly policy: DaemonPolicy;
35 private readonly clock = new NodeDaemonClock();
36
37 constructor(
38 private readonly port: NonNullable<typeof parentPort>,
39 private readonly data: DaemonWorkerData,
40 ) {
41 this.policy = DaemonPolicy.fromSerialized(data.policy);
42 }

44 run(): void {
45 this.port.on("message", (value: unknown) => {
46 if (
47 typeof value === "object" &&
48 value !== null &&
49 "kind" in value &&
50 value.kind === "output-ack"
51 ) {
52 try {
53 const request = DaemonNavigationWorkerProtocol.request(value);
54 if (request.kind !== "output-ack") throw new Error("Invalid output acknowledgement");
55 this.acknowledgeOutput(request.requestId, request.sequence);
56 return;
57 } catch (error) {
58 this.fail("protocol", error);
59 return;
60 }
61 }
62 this.tail = this.tail.then(() => this.handle(value));
63 });
64 }
65
66 private async handle(value: unknown): Promise<void> {
67 let request: DaemonNavigationWorkerRequest;
68 try {
69 request = DaemonNavigationWorkerProtocol.request(value);
70 } catch (error) {
71 this.fail("protocol", error);
72 return;
73 }
74 if (request.generation !== this.data.generation) return;
75 if (request.kind === "initialize") {
76 await this.initialize(request.workspaceRoot);
77 return;
78 }
79 if (request.kind === "execute") {
80 await this.execute(request);
81 return;
82 }
83 if (request.kind === "release-transient") {
84 await this.releaseTransientResources(request.operationId);
85 return;
86 }
87 await this.close();
88 }

90 private async initialize(workspaceRoot: string): Promise<void> {
91 const startedAt = this.clock.monotonicNowMs();
92 try {
93 this.executor = await DaemonExecutorModuleLoader.load(this.data.executorModuleUrl, {
94 stateDirectory: this.data.stateDirectory,
95 productVersion: this.data.productVersion,
96 sampleResources: () => this.activeHeapMonitor?.sample(),
97 });
98 const initialized = await this.executor.initialize(workspaceRoot);
99 const totalMs = this.clock.monotonicNowMs() - startedAt;
100 this.send({
101 kind: "ready",
102 generation: this.data.generation,
103 fileCount: initialized.fileCount,
104 refresh: DaemonWorkerDiagnosticProjection.refresh(initialized.diagnostics),
105 startupDurations: {
106 ...DaemonWorkerDiagnosticProjection.startupDurations(initialized.diagnostics),
107 totalMs,
108 },
109 ...(initialized.diagnostics === undefined ? {} : { diagnostics: initialized.diagnostics }),
110 });
111 } catch (error) {
112 this.fail("initialization", error);
113 }
114 }

116 private async execute(request: Extract<DaemonNavigationWorkerRequest, { kind: "execute" }>) {
117 const heapMonitor = new WorkerHeapHighWater(
118 this.policy.values.resources.workerHeapSampleIntervalMs,
119 );
120 this.activeHeapMonitor = heapMonitor;
121 try {
122 if (this.executor === undefined) throw new Error("Navigation worker is not ready");
123 const result = await this.executor.execute(request.request);
124 heapMonitor.sample();
125 const outputStartedAt = this.clock.monotonicNowMs();
126 let sequence = 0;
127 for await (const record of result.output.records()) {
128 sequence = await this.sendOutput(request.requestId, sequence, record);
129 heapMonitor.sample();
130 }
131 await result.output.dispose();
132 const outputMs = this.clock.monotonicNowMs() - outputStartedAt;
133 const resources = heapMonitor.finish();
134 this.send({
135 kind: "result",
136 generation: this.data.generation,
137 requestId: request.requestId,
138 result: { exitCode: result.exitCode },
139 refresh: DaemonWorkerDiagnosticProjection.refresh(result.diagnostics),
140 durations: {
141 ...DaemonWorkerDiagnosticProjection.commandDurations(result.diagnostics),
142 outputMs,
143 },
144 ...(result.diagnostics === undefined ? {} : { diagnostics: result.diagnostics }),
145 resources,
146 });
147 } catch (error) {
148 this.fail("execution", error, { requestId: request.requestId });
149 } finally {
150 heapMonitor.close();
151 this.activeHeapMonitor = undefined;
152 }
153 }

155 private async releaseTransientResources(operationId: string): Promise<void> {
156 try {
157 await this.executor?.releaseTransientResources();
158 const heap = getHeapStatistics();
159 this.send({
160 kind: "heap",
161 generation: this.data.generation,
162 operationId,
163 usedHeapBytes: heap.used_heap_size,
164 heapLimitBytes: heap.heap_size_limit,
165 });
166 } catch (error) {
167 this.fail("resource", error, { operationId });
168 }
169 }

171 private async close(): Promise<void> {
172 await this.executor?.releaseTransientResources();
173 this.send({ kind: "closed", generation: this.data.generation });
174 this.port.close();
175 }

196 response,
197 this.policy.values.output.maximumChunkRawBytes,

199 if (validated.kind === "output-chunk") {

206 private sendOutput(
207 requestId: string,
208 initialSequence: number,
209 record: DaemonOutputRecord,
210 ): Promise<number> {
211 return this.sendOutputChunks(requestId, initialSequence, record);

214 private async sendOutputChunks(
215 requestId: string,
216 initialSequence: number,
217 record: DaemonOutputRecord,
218 ): Promise<number> {
219 let sequence = initialSequence;
220 if (record.bytes.byteLength === 0) {
221 await this.sendOutputChunk(requestId, sequence, record.stream, new Uint8Array());
222 return sequence + 1;
223 }
224 for (
225 let offset = 0;
226 offset < record.bytes.byteLength;
227 offset += this.policy.values.output.maximumChunkRawBytes
228 ) {
229 const bytes = Uint8Array.from(
230 record.bytes.subarray(offset, offset + this.policy.values.output.maximumChunkRawBytes),
231 );
232 await this.sendOutputChunk(requestId, sequence, record.stream, bytes);
233 sequence += 1;
234 }
235 return sequence;
236 }

238 private sendOutputChunk(
239 requestId: string,
240 sequence: number,
241 stream: DaemonOutputRecord["stream"],
242 bytes: Uint8Array,
243 ): Promise<void> {
244 const key = `${requestId}:${sequence}`;
245 if (this.outputAcknowledgements.has(key)) {
246 return Promise.reject(new Error("Duplicate worker output sequence"));
247 }
248 const acknowledgement = new Promise<void>((resolve) => {
249 this.outputAcknowledgements.set(key, resolve);
250 });
251 this.send({
252 kind: "output-chunk",
253 generation: this.data.generation,
254 requestId,
255 sequence,
256 stream,
257 bytes,
258 });
259 return acknowledgement;
260 }
261
262 private acknowledgeOutput(requestId: string, sequence: number): void {
263 const key = `${requestId}:${sequence}`;
264 const resolve = this.outputAcknowledgements.get(key);
265 if (resolve === undefined) throw new Error("Unexpected worker output acknowledgement");
266 this.outputAcknowledgements.delete(key);
267 resolve();
268 }

271class DaemonWorkerDiagnosticProjection {
272 private static readonly emptyRefresh: DaemonRefreshSummary = {
273 added: 0,
274 changed: 0,
275 removed: 0,
276 unchanged: 0,
277 };
278
279 static refresh(diagnostics: DaemonDiagnostics | undefined): DaemonRefreshSummary {
280 const refresh = this.record(diagnostics?.refresh);
281 if (refresh === undefined) return this.emptyRefresh;
282 return {
283 added: this.count(refresh.added),
284 changed: this.count(refresh.changed),
285 removed: this.count(refresh.removed),
286 unchanged: this.count(refresh.unchanged),
287 };
288 }
289
290 static startupDurations(diagnostics: DaemonDiagnostics | undefined): {
291 readonly discoveryMs: number;
292 readonly indexingMs: number;
293 } {
294 const durations = this.record(diagnostics?.durations);
295 return {
296 discoveryMs: this.duration(durations?.discoveryMs),
297 indexingMs: this.duration(durations?.indexingMs),
298 };
299 }
300
301 static commandDurations(
302 diagnostics: DaemonDiagnostics | undefined,
303 ): Omit<WorkerCommandDurations, "outputMs"> {
304 const durations = this.record(diagnostics?.durations);
305 return {
306 freshnessMs: this.duration(durations?.freshnessMs),
307 navigationMs: this.duration(durations?.navigationMs),
308 renderMs: this.duration(durations?.renderMs),
309 };
310 }
311
312 private static record(value: unknown): Readonly<Record<string, unknown>> | undefined {
313 return typeof value === "object" && value !== null && !Array.isArray(value)
314 ? (value as Readonly<Record<string, unknown>>)
315 : undefined;
316 }
317
318 private static count(value: unknown): number {
319 return Number.isSafeInteger(value) && Number(value) >= 0 ? Number(value) : 0;
320 }
321
322 private static duration(value: unknown): number {
323 return typeof value === "number" && Number.isFinite(value) && value >= 0 ? value : 0;
324 }

327class WorkerHeapHighWater {
328 private currentUsedBytes = 0;
329 private peakUsedBytes = 0;
330 private heapLimitBytes = 0;
331 private readonly timer: ReturnType<typeof setInterval>;
332
333 constructor(sampleIntervalMs: number) {
334 this.sample();
335 this.timer = setInterval(() => this.sample(), sampleIntervalMs);
336 this.timer.unref?.();
337 }
338
339 sample(): void {
340 const heap = getHeapStatistics();
341 this.currentUsedBytes = heap.used_heap_size;
342 this.peakUsedBytes = Math.max(this.peakUsedBytes, this.currentUsedBytes);
343 this.heapLimitBytes = heap.heap_size_limit;
344 }
345
346 finish(): {
347 readonly workerHeapUsedBytes: number;
348 readonly peakWorkerHeapUsedBytes: number;
349 readonly workerHeapLimitBytes: number;
350 } {
351 this.sample();
352 this.close();
353 return {
354 workerHeapUsedBytes: this.currentUsedBytes,
355 peakWorkerHeapUsedBytes: this.peakUsedBytes,
356 workerHeapLimitBytes: this.heapLimitBytes,
357 };
358 }
359
360 close(): void {
361 clearInterval(this.timer);
362 }
363}
