98 async trigger(identity: DaemonWorkspaceIdentity): Promise<DaemonWarmupTriggerResult> {
99 const readyRecord = await this.validatedReadyRecord(identity);
100 if (readyRecord?.symnavVersion === this.launcher.symnavVersion) {
101 return {
102 status: "ready",
103 instanceId: readyRecord.instanceId,
104 pid: readyRecord.pid,
105 };
106 }
107
108 const instanceId = this.nextInstanceId();
109 const processToken = randomUUID();
110 const lease = this.registry.acquireStartup(identity, {
111 identityKey: identity.identityKey,
112 instanceId,
113 processToken,
114 ownerPid: process.pid,
115 ownerKind: "launcher",
116 heartbeatAt: this.now(),
117 });
118 if (lease === undefined) return this.observeElectedWarmup(identity);
119 try {
120 const currentRecord = await this.validatedReadyRecord(identity);
121 if (currentRecord?.symnavVersion === this.launcher.symnavVersion) {
122 lease.release();
123 return {
124 status: "ready",
125 instanceId: currentRecord.instanceId,
126 pid: currentRecord.pid,
127 };
128 }
129 const storedRecord = this.registry.readStored(identity);
130 if (storedRecord !== undefined) await this.replaceStoredRecord(identity, storedRecord);
131 return await this.launch(identity, instanceId, processToken, lease);
132 } catch (error) {
133 lease.release();
134 throw error;
135 }

138 async waitUntilReady(
139 identity: DaemonWorkspaceIdentity,
140 expectedInstanceId?: string,
141 ): Promise<DaemonStartResult> {
142 let missingOwner: { readonly instanceId: string; readonly firstObservedAt: number } | undefined;
143 while (true) {
144 const record = this.registry.read(identity);
145 if (record?.state === "ready" && record.symnavVersion === this.launcher.symnavVersion) {
146 const validated = await this.validatedChildRecord(identity);
147 if (validated?.instanceId === record.instanceId) {
148 await this.probeExecution(validated);
149 return this.launchedInstances.has(record.instanceId)
150 ? {
151 status: "ready",
152 workspaceRoot: record.workspaceRoot,
153 fileCount: record.fileCount ?? 0,
154 loadDurationMs: (record.readyAt ?? this.now()) - record.startedAt,
155 }
156 : this.alreadyRunning(record);
157 }
158 }
159 const storedRecord = this.registry.readStored(identity);
160 const observedInstanceId =
161 storedRecord?.instanceId ?? this.launchedInstanceIdsByIdentity.get(identity.identityKey);
162 const launchedExit =
163 observedInstanceId === undefined ? undefined : this.launchedExits.get(observedInstanceId);
164 if (launchedExit !== undefined) throw new DaemonChildExitError(launchedExit);
165 const owner = this.registry.startupOwner(identity);
166 if (owner !== undefined) missingOwner = undefined;
167 if (
168 storedRecord?.state === "ready" &&
169 storedRecord.symnavVersion === this.launcher.symnavVersion
170 ) {
171 await this.pause();
172 continue;
173 }
174 if (storedRecord?.state === "starting") {
175 if (owner !== undefined && this.startupOwnerIsAbandoned(identity, owner)) {
176 if (this.cleanupAbandonedStartup(identity, owner)) {
177 throw new DaemonWarmupLostError("Daemon child exited before readiness");
178 }
179 await this.pause();
180 continue;
181 }
182 const daemonProcess = this.launchedProcesses.get(storedRecord.instanceId);
183 if (daemonProcess !== undefined) {
184 const childExit = await Promise.race([
185 this.pause().then(() => undefined),
186 daemonProcess.exited,
187 ]);
188 if (childExit !== undefined) throw new DaemonChildExitError(childExit);
189 continue;
190 }
191 if (owner !== undefined) {
192 await this.pause();
193 continue;
194 }
195 if (missingOwner?.instanceId !== storedRecord.instanceId) {
196 missingOwner = {
197 instanceId: storedRecord.instanceId,
198 firstObservedAt: this.now(),
199 };
200 await this.pause();
201 continue;
202 }
203 if (this.now() - missingOwner.firstObservedAt <= this.coordinationGraceMs) {
204 await this.pause();
205 continue;
206 }
207 }
208 if (owner !== undefined) {
209 await this.pause();
210 continue;
211 }
212 if (expectedInstanceId !== undefined) {
213 throw new DaemonWarmupLostError(
214 `Daemon startup ${expectedInstanceId} ended before readiness`,
215 );
216 }
217 throw new Error("Daemon startup failed before readiness");
218 }

221 private async launch(
222 identity: DaemonWorkspaceIdentity,
223 instanceId: string,
224 processToken: string,
225 lease: NonNullable<ReturnType<DaemonRegistry["acquireStartup"]>>,
226 ): Promise<DaemonWarmupTriggerResult> {
227 const startedAt = this.now();
228 const startingRecord: DaemonRecord = {
229 schemaVersion: DAEMON_RECORD_SCHEMA_VERSION,
230 protocolVersion: DAEMON_PROTOCOL_VERSION,
231 symnavVersion: this.launcher.symnavVersion,
232 workspaceRoot: identity.workspaceRoot,
233 workspaceKey: identity.workspaceKey,
234 stateKey: identity.stateKey,
235 identityKey: identity.identityKey,
236 instanceId,
237 processToken,
238 endpoint: identity.endpoint(instanceId),
239 pid: 0,
240 state: "starting",
241 startedAt,
242 memoryCapBytes: this.launcher.memoryCapBytes,
243 };
244 if (!this.registry.writeStartingIfStartupOwner(identity, startingRecord)) {
245 throw new Error("Daemon startup ownership changed before process launch");
246 }
247 let daemonProcess: DaemonProcess | undefined;
248 try {
249 daemonProcess = await this.launcher.launch(identity, instanceId, processToken);
250 const transferred = lease.transferToDaemon(daemonProcess.pid, processToken);
251 const daemonOwner = this.registry.startupOwner(identity);
252 if (
253 !transferred &&
254 !(
255 daemonOwner?.identityKey === identity.identityKey &&
256 daemonOwner.instanceId === instanceId &&
257 daemonOwner.processToken === processToken &&
258 daemonOwner.ownerKind === "daemon" &&
259 daemonOwner.ownerPid === daemonProcess.pid
260 )
261 ) {
262 throw new Error("Daemon startup ownership changed after process launch");
263 }
264 if (
265 !this.registry.writeStartingIfStartupOwner(identity, {
266 ...startingRecord,
267 pid: daemonProcess.pid,
268 })
269 ) {
270 throw new Error("Daemon startup ownership changed after process launch");
271 }
272 this.launchedInstances.add(instanceId);
273 this.launchedProcesses.set(instanceId, daemonProcess);
274 this.launchedInstanceIdsByIdentity.set(identity.identityKey, instanceId);
275 this.observeLaunchedProcess(identity, instanceId, processToken, daemonProcess);
276 return {
277 status: "launched",
278 instanceId,
279 pid: daemonProcess.pid,
280 };
281 } catch (error) {
282 if (error instanceof DaemonChildExitError) {
283 this.cleanupLaunchedProcess(identity, instanceId, processToken);
284 throw error;
285 }
286 if (daemonProcess !== undefined) {
287 try {
288 await daemonProcess.terminate();
289 } catch (terminationError) {
290 if (terminationError instanceof DaemonProcessTerminationError) throw terminationError;
291 throw new DaemonProcessTerminationError(String(terminationError));
292 }
293 }
294 this.cleanupLaunchedProcess(identity, instanceId, processToken);
295 throw error;
296 }

362 private cleanupAbandonedStartup(identity: DaemonWorkspaceIdentity, owner: StartupOwner): boolean {
363 const record = this.registry.readStoredInstance(identity, owner.instanceId);
364 if (
365 record !== undefined &&
366 ((owner.processToken.length > 0 && record.processToken !== owner.processToken) ||
367 (owner.ownerKind === "daemon" && record.pid !== owner.ownerPid))
368 ) {
369 return false;
370 }
371 if (!this.registry.removeStartupLockIfOwner(identity, owner)) return false;
372 if (record !== undefined) {

378 private startupOwnerIsAbandoned(identity: DaemonWorkspaceIdentity, owner: StartupOwner): boolean {
379 const record = this.registry.readStoredInstance(identity, owner.instanceId);
380 if (record !== undefined && record.pid > 0) {

383 if (owner.ownerKind === "daemon") return !this.processTerminator.isAlive(owner.ownerPid);

401 private async validatedReadyRecord(
402 identity: DaemonWorkspaceIdentity,
403 ): Promise<DaemonRecord | undefined> {
404 const record = this.registry.read(identity);
405 if (record?.state !== "ready") return undefined;
406 const observation = await this.observer.observe(record);
407 if (observation.kind === "responsive") {
408 this.registry.removeStartupLockIfProcess(identity, record);
409 return record;
410 }
411 if (observation.kind === "exited") {
412 this.registry.removeIfProcess(identity, record.instanceId, record.processToken);
413 return undefined;
414 }
415 if (observation.kind === "incompatible" || observation.kind === "corrupt") return undefined;
416 throw new DaemonOwnedButUnresponsiveError(
417 `Daemon process ${record.pid} is live but unresponsive; ownership was retained`,
418 );
419 }

421 private async replaceStoredRecord(
422 identity: DaemonWorkspaceIdentity,
423 record: DaemonRecord,
424 ): Promise<void> {
425 const observation = await this.observer.observe(record);
426 if (observation.kind === "exited") {
427 this.registry.removeIfProcess(identity, record.instanceId, record.processToken);
428 return;
429 }
430 if (observation.kind === "starting" || observation.kind === "unresponsive") {
431 throw new DaemonOwnedButUnresponsiveError(
432 `Daemon process ${record.pid} is live but unresponsive; ownership was retained`,
433 );
434 }
435 await this.transport.request(record.endpoint, {
436 kind: "terminate",
437 instanceId: record.instanceId,
438 processToken: record.processToken,
439 });
440 await this.waitForProcessExitAndEndpointRelease(record);
441 this.registry.removeIfProcess(identity, record.instanceId, record.processToken);
442 }

444 private async waitForProcessExitAndEndpointRelease(record: DaemonRecord): Promise<void> {
445 const waitStartedAt = this.now();
446 while (this.now() - waitStartedAt <= this.terminationTimeoutMs) {
447 const endpointReleased = !(await this.identifiesRecordedProcess(record));
448 const processExited = !this.processTerminator.isAlive(record.pid);
449 if (endpointReleased && processExited) return;
450 await this.pause();
451 }
452 throw new DaemonProcessTerminationError(
453 `Daemon process ${record.pid} did not exit and release its endpoint`,
454 );
455 }
