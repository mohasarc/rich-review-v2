83 start(requestId: string, command: DaemonCommandName): DaemonOperationTrace {
84 const acceptedAt = this.clock.monotonicNowMs();
85 let turnStartedAt: number | undefined;
86 let deliveryStartedAt: number | undefined;
87 let executionTerminal = false;
88 let deliveryTerminal = false;
89 return {
90 accepted: (queueDepth, generation) => {
91 this.logger.record({
92 kind: "request-accepted",
93 requestId,
94 command,
95 queueDepth,
96 workerGeneration: generation,
97 });
98 },
99 turnStarted: (generation) => {
100 turnStartedAt = this.clock.monotonicNowMs();
101 this.logger.record({
102 kind: "turn-started",
103 requestId,
104 queueWaitMs: DaemonOperationObserver.elapsed(acceptedAt, turnStartedAt),
105 workerGeneration: generation,
106 });
107 },
108 workerCompleted: (durations, refresh) => {
109 this.logger.record({ kind: "worker-completed", requestId, ...durations, ...refresh });
110 },
111 spooled: (manifest, durationMs) => {
112 this.logger.record({
113 kind: "response-spooled",
114 requestId,
115 rawBytes: manifest.rawBytes,
116 recordCount: manifest.recordCount,
117 spoolMs: Math.max(0, durationMs),
118 });
119 },
120 executionTerminated: (outcome) => {
121 if (executionTerminal) return;
122 executionTerminal = true;
123 deliveryStartedAt = this.clock.monotonicNowMs();
124 const resources = this.resources?.snapshot;
125 this.logger.record({
126 kind: "execution-terminal",
127 requestId,
128 outcome,
129 serviceMs: DaemonOperationObserver.elapsed(
130 turnStartedAt ?? deliveryStartedAt,
131 deliveryStartedAt,
132 ),
133 ...(resources === undefined
134 ? {}
135 : {
136 processRssBytes: resources.processRssBytes,
137 peakProcessRssBytes: resources.peakProcessRssBytes,
138 ...(resources.peakWorkerHeapUsedBytes === undefined
139 ? {}
140 : { peakWorkerHeapUsedBytes: resources.peakWorkerHeapUsedBytes }),
141 ...(resources.workerHeapUsedBytes === undefined
142 ? {}
143 : { workerHeapUsedBytes: resources.workerHeapUsedBytes }),
144 spoolBytes: resources.spoolBytes,
145 }),
146 });
147 },
148 clientDisconnected: () => {
149 this.logger.record({ kind: "client-disconnected", requestId });
150 },
151 reattached: () => {
152 this.reattached(requestId);
153 },
154 deliveryTerminated: (outcome) => {
155 if (deliveryTerminal) return;
156 deliveryTerminal = true;
157 const completedAt = this.clock.monotonicNowMs();
158 this.deliveryTerminated(
159 requestId,
160 outcome,
161 DaemonOperationObserver.elapsed(deliveryStartedAt ?? completedAt, completedAt),
162 );
163 },
164 };
165 }

167 private static elapsed(startedAt: number, completedAt: number): number {
168 return Math.max(0, completedAt - startedAt);
