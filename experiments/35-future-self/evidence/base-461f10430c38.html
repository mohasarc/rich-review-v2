222class FakeSocket extends EventEmitter {
223 readonly writes: Uint8Array[] = [];
224 readonly writeResults: boolean[] = [];
225 destroyed = false;
226 pauseCount = 0;
227 resumeCount = 0;
228 endCount = 0;
229 destroyCount = 0;
230 readonly timeoutMilliseconds: number[] = [];
231 private timeoutListener: (() => void) | undefined;
232 writeError: Error | undefined;
233
234 setTimeout(milliseconds: number, listener?: () => void): this {
235 this.timeoutMilliseconds.push(milliseconds);
236 this.timeoutListener = listener;
237 return this;
238 }
239
240 timeout(): void {
241 this.timeoutListener?.();
242 }
243
244 pause(): this {
245 this.pauseCount += 1;
246 return this;
247 }
248
249 resume(): this {
250 this.resumeCount += 1;
251 return this;
252 }
253
254 write(frame: Uint8Array): boolean {
255 if (this.writeError !== undefined) throw this.writeError;
256 this.writes.push(Uint8Array.from(frame));
257 return this.writeResults.shift() ?? true;
258 }
259
260 end(): this {
261 this.endCount += 1;
262 return this;
263 }
264
265 destroy(): this {
266 this.destroyed = true;
267 this.destroyCount += 1;
268 return this;
269 }
