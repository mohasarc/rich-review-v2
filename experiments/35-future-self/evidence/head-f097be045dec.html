340class ControlledNavigationWorker implements DaemonNavigationWorker {
341 readonly exited: Promise<DaemonNavigationWorkerExit>;
342 readonly operations: string[] = [];
343 execution:
344 | {
345 readonly requestId: string;
346 readonly commandName: Parameters<DaemonNavigationWorker["execute"]>[1];
347 readonly request: Parameters<DaemonNavigationWorker["execute"]>[2];
348 readonly output: DaemonOutputSink;
349 }
350 | undefined;
351 executionResponse: DaemonNavigationWorkerResponse | undefined;
352 resourceResponse: DaemonNavigationWorkerResponse | undefined;
353 closeOperation: Promise<void> = Promise.resolve();
354 terminateOperation: Promise<void> = Promise.resolve();
355 private resolveExited!: (exit: DaemonNavigationWorkerExit) => void;
356 private resolveReady!: (response: DaemonNavigationWorkerResponse) => void;
357 private rejectReady!: (error: Error) => void;
358 private readonly ready: Promise<DaemonNavigationWorkerResponse>;
359
360 constructor(
361 readonly generation: number,
362 private readonly sharedOperations: string[] = [],
363 ) {
364 this.exited = new Promise((resolve) => {
365 this.resolveExited = resolve;
366 });
367 this.ready = new Promise((resolve, reject) => {
368 this.resolveReady = resolve;
369 this.rejectReady = reject;
370 });
371 }
372
373 start(workspaceRoot: string): Promise<DaemonNavigationWorkerResponse> {
374 this.operations.push(`start:${this.generation}`);
375 this.sharedOperations.push(`start:${this.generation}`);
376 expect(workspaceRoot).toBe("/workspace");
377 return this.ready;
378 }
379
380 execute(
381 requestId: string,
382 commandName: Parameters<DaemonNavigationWorker["execute"]>[1],
383 request: Parameters<DaemonNavigationWorker["execute"]>[2],
384 output: DaemonOutputSink,
385 ): Promise<DaemonNavigationWorkerResponse> {
386 this.execution = { requestId, commandName, request, output };
387 if (this.executionResponse === undefined) {
388 return Promise.reject(new Error("Execution response is unavailable"));
389 }
390 return Promise.resolve(this.executionResponse);
391 }
392
393 releaseTransientResources(): Promise<DaemonNavigationWorkerResponse> {
394 this.operations.push(`release:${this.generation}`);
395 if (this.resourceResponse === undefined) {
396 return Promise.reject(new Error("Resource response is unavailable"));
397 }
398 return Promise.resolve(this.resourceResponse);
399 }
400
401 drainAndClose(): Promise<void> {
402 this.operations.push(`close:${this.generation}`);
403 this.sharedOperations.push(`close:${this.generation}`);
404 return this.closeOperation;
405 }
406
407 terminate(): Promise<void> {
408 this.operations.push(`terminate:${this.generation}`);
409 this.sharedOperations.push(`terminate:${this.generation}`);
410 return this.terminateOperation;
411 }
412
413 completeReady(fileCount: number): void {
414 this.complete({
415 kind: "ready",
416 generation: this.generation,
417 fileCount,
418 refresh: { added: fileCount, changed: 0, removed: 0, unchanged: 0 },
419 startupDurations: { discoveryMs: 1, indexingMs: 2, totalMs: 3 },
420 });
421 }
422
423 complete(response: DaemonNavigationWorkerResponse): void {
424 this.resolveReady(response);
425 }
426
427 rejectStart(error: Error): void {
428 this.rejectReady(error);
429 }
430
431 exit(exit: DaemonNavigationWorkerExit): void {
432 this.resolveExited(exit);
433 }
434}

453 let resolve!: () => void;
