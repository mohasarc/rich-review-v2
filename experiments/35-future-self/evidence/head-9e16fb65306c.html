713 });
714 }
715
716 override drainAndClose(): Promise<void> {
717 this.resolveDrainStarted();
718 return this.drain;
719 }

721 override async terminate(): Promise<void> {
722 this.terminateCount += 1;
723 this.resolveDrain();
724 await super.terminate();
725 }
726}
