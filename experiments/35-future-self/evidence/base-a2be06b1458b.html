200function policy(overrides: Partial<CapturePolicy> = {}): CapturePolicy {
201 return {
202 maximumChunkRawBytes: 64 * 1024,

215function temporaryDirectory(directories: string[]): string {
216 const directory = mkdtempSync(join(tmpdir(), "symnav-client-capture-"));
217 directories.push(directory);
218 return directory;
219}

221function digest(records: readonly DaemonSequencedOutputRecord[]): string {
222 const hash = createHash("sha256");
223 for (const record of records) hash.update(encode(record).subarray(4));
224 return hash.digest("hex");
225}
226
227function encode(record: DaemonSequencedOutputRecord): Buffer {
228 const header = Buffer.alloc(9);
229 header.writeUInt32BE(record.sequence, 0);
230 header.writeUInt8(record.stream === "stdout" ? 0 : 1, 4);
231 header.writeUInt32BE(record.bytes.byteLength, 5);
232 return Buffer.concat([header, Buffer.from(record.bytes)]);
233}
