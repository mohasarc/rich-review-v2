183class FakeCapture implements DaemonOutputCapture {
184 readonly appended: DaemonSequencedOutputRecord[] = [];
185 readonly finishExitCodes: number[] = [];
186 readonly resultOutput = new FakeOutput();
187 readonly result: DaemonExecutorExecutionResult = { exitCode: 0, output: this.resultOutput };
188 disposeCalls = 0;
189
190 constructor(
191 private readonly capturedSummary: DaemonCapturedOutputSummary,
192 private readonly appendOperation: () => Promise<void> = () => Promise.resolve(),
193 ) {}
194
195 async append(record: DaemonSequencedOutputRecord): Promise<void> {
196 await this.appendOperation();
197 this.appended.push({ ...record, bytes: Buffer.from(record.bytes) });
198 }
199
200 finish(exitCode: number): Promise<DaemonCapturedOutput> {
201 this.finishExitCodes.push(exitCode);
202 return Promise.resolve({
203 result: this.result,
204 summary: this.capturedSummary,
205 });
206 }
207
208 dispose(): Promise<void> {
209 this.disposeCalls += 1;
210 return Promise.resolve();
211 }
212}

278function summary(records: readonly DaemonResultChunk[]): DaemonCapturedOutputSummary {
279 const hash = createHash("sha256");
280 let rawBytes = 0;
281 for (const record of records) {
282 const encoded = Buffer.alloc(5 + record.bytes.byteLength);
283 encoded.writeUInt8(record.stream === "stdout" ? 0 : 1, 0);
284 encoded.writeUInt32BE(record.bytes.byteLength, 1);
285 Buffer.from(record.bytes).copy(encoded, 5);
286 hash.update(encoded);
287 rawBytes += record.bytes.byteLength;
288 }
289 return { rawBytes, recordCount: records.length, sha256: hash.digest("hex") };
