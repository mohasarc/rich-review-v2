396class BlockingLogStorage implements DaemonLogStorage {
397 readonly appendStarted: Promise<void>;
398 private resolveAppendStarted!: () => void;
399 private releaseAppend!: () => void;
400 private readonly gate: Promise<void>;
401 private readonly lines: string[] = [];
402
403 constructor() {
404 this.appendStarted = new Promise((resolve) => {
405 this.resolveAppendStarted = resolve;
406 });
407 this.gate = new Promise((resolve) => {
408 this.releaseAppend = resolve;
409 });
410 }
411
412 prepare(): Promise<void> {
413 return Promise.resolve();
414 }
415
416 size(): Promise<number> {
417 return Promise.resolve(this.lines.reduce((total, line) => total + Buffer.byteLength(line), 0));
418 }
419
420 async append(_path: string, line: string): Promise<void> {
421 this.resolveAppendStarted();
422 await this.gate;
423 this.lines.push(line);
424 }
425
426 move(): Promise<void> {
427 return Promise.resolve();
428 }
429
430 remove(): Promise<void> {
431 return Promise.resolve();
432 }
433
434 sync(): Promise<void> {
435 return Promise.resolve();
436 }
437
438 release(): void {
439 this.releaseAppend();
440 }
441
442 events(): readonly Record<string, unknown>[] {
443 return this.lines.map((line) => JSON.parse(line) as Record<string, unknown>);
444 }
445}
