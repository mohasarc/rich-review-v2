17class CompletionSpoolRecordCodec {
18 static encode(record: CommandOutputRecord): Buffer {
19 const header = Buffer.alloc(RECORD_HEADER_BYTES);
20 header.writeUInt32BE(record.sequence, 0);
21 header.writeUInt8(record.stream === "stdout" ? 0 : 1, 4);
22 header.writeUInt32BE(record.bytes.byteLength, 5);
23 return Buffer.concat([header, Buffer.from(record.bytes)]);
24 }

26 static async *decodeFile(
27 filePath: string,
28 maximumRecordBytes: number,
29 ): AsyncIterable<CommandOutputRecord> {
30 const noFollow = "O_NOFOLLOW" in constants ? constants.O_NOFOLLOW : 0;
31 const handle = await open(filePath, constants.O_RDONLY | noFollow);
32 try {
33 const metadata = await handle.stat();
34 if (!metadata.isFile()) throw new Error("Command output spool is not a regular file");
35 let position = 0;
36 let expectedSequence = 0;
37 while (position < metadata.size) {
38 const header = Buffer.alloc(RECORD_HEADER_BYTES);
39 await CompletionSpoolRecordCodec.readExact(handle, header, position);
40 position += RECORD_HEADER_BYTES;
41 const sequence = header.readUInt32BE(0);
42 const streamByte = header.readUInt8(4);
43 const length = header.readUInt32BE(5);
44 if (sequence !== expectedSequence || streamByte > 1 || length > maximumRecordBytes) {
45 throw new Error("Corrupt command output");
46 }
47 const bytes = Buffer.alloc(length);
48 await CompletionSpoolRecordCodec.readExact(handle, bytes, position);
49 position += length;
50 expectedSequence += 1;
51 yield { sequence, stream: streamByte === 0 ? "stdout" : "stderr", bytes };
52 }
53 } finally {
54 await handle.close();
55 }
56 }

58 private static async readExact(
59 handle: FileHandle,
60 buffer: Buffer,
61 position: number,
62 ): Promise<void> {
63 let offset = 0;
64 while (offset < buffer.byteLength) {
65 const { bytesRead } = await handle.read(
66 buffer,
67 offset,
68 buffer.byteLength - offset,
69 position + offset,
70 );
71 if (bytesRead === 0) throw new Error("Truncated command output");
72 offset += bytesRead;
73 }

193 async append(record: CommandOutputRecord): Promise<void> {
194 if (this.terminal) throw new Error("Completion spool is already terminal");
195 if (record.sequence !== this.recordCount) throw new Error("Unexpected command output sequence");
196 if (record.bytes.byteLength > this.options.maximumChunkBytes) {
197 throw new Error("Command output record exceeds chunk capacity");
198 }
199 if (this.rawBytes + record.bytes.byteLength > this.options.maximumResultBytes) {
200 await this.failCapacity();
201 }
202 let currentRecordReserved = false;
203 try {
204 this.options.reserve(record.bytes.byteLength);
205 currentRecordReserved = true;
206 const stored: CommandOutputRecord = { ...record, bytes: Buffer.from(record.bytes) };
207 if (
208 this.file === undefined &&
209 this.rawBytes + stored.bytes.byteLength > this.options.inlineBytes
210 ) {
211 await this.spillInlineRecords();
212 }
213 const encoded = CompletionSpoolRecordCodec.encode(stored);
214 if (this.file === undefined) this.inlineRecords.push(stored);
215 else await this.file.write(encoded);
216 this.hash.update(encoded.subarray(4));
217 this.rawBytes += stored.bytes.byteLength;
218 this.recordCount += 1;
219 } catch (error) {
220 if (currentRecordReserved) this.options.release(record.bytes.byteLength);
221 await this.dispose();
222 throw error;
223 }
224 }

333 }

372 this.inlineBytes = policy.inlineRawBytes;

378 async create(requestId: string): Promise<CompletionSpool> {
379 if (this.spools.has(requestId))
380 throw new Error(`Completion spool already exists: ${requestId}`);
381 let completionCounted = false;
382 const spool = new CompletionSpool({
383 directory: this.options.directory,
384 identity: {
385 workspaceKey: this.options.workspaceKey,
386 instanceId: this.options.instanceId,
387 transferId: randomUUID(),
388 },
389 requestId,
390 inlineBytes: this.inlineBytes,
391 maximumResultBytes: this.maximumResultBytes,
392 maximumChunkBytes: this.maximumChunkBytes,
393 reserve: (bytes) => this.reserve(bytes),
394 release: (bytes) => {
395 this.rawBytes -= bytes;
396 },
397 complete: () => {
398 completionCounted = true;
399 this.completionCount += 1;
400 },
401 remove: () => {
402 if (this.spools.get(requestId) !== spool) return;
403 this.spools.delete(requestId);
404 },
405 releaseCompletion: () => {
406 if (completionCounted) this.completionCount -= 1;
407 completionCounted = false;
408 },
409 cleanupComplete: () => {
410 this.pendingCleanup.delete(spool);
411 },
412 storage: this.storage,
413 });
414 this.spools.set(requestId, spool);
415 this.pendingCleanup.add(spool);
416 return spool;
417 }

427 async cleanupInstance(instanceId: string): Promise<void> {
428 if (instanceId !== this.options.instanceId) return;
