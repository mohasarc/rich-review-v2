177class ObserverTransport {
178 readonly requests: DaemonRequest[] = [];
179
180 constructor(private readonly outcomes: readonly (DaemonResponse | Error)[]) {}
181
182 async request(_endpoint: string, request: DaemonRequest): Promise<DaemonResponse> {
183 this.requests.push(request);
184 const outcome = this.outcomes[this.requests.length - 1];
185 if (outcome instanceof Error) throw outcome;
186 if (outcome === undefined) throw new Error("Unexpected observer request");
187 return outcome;
188 }
189}
190
191class ConcurrentObserverTransport extends ObserverTransport {
192 readonly pingRequested: Promise<void>;
193 private resolvePingRequested!: () => void;
194 private readonly identityAllowed: Promise<void>;
195 private resolveIdentityAllowed!: () => void;
196
197 constructor() {
198 super([]);
199 this.pingRequested = new Promise((resolve) => {
200 this.resolvePingRequested = resolve;
201 });
202 this.identityAllowed = new Promise((resolve) => {
203 this.resolveIdentityAllowed = resolve;
204 });
205 }
206
207 override async request(_endpoint: string, request: DaemonRequest): Promise<DaemonResponse> {
208 if (request.kind === "identify") {
209 await this.identityAllowed;
210 return identityResponse();
211 }
212 if (request.kind === "ping") {
213 this.resolvePingRequested();
214 return {
215 kind: "pong",
216 protocolVersion: DAEMON_PROTOCOL_VERSION,
217 instanceId: "instance",
218 symnavVersion: "0.1.0",
219 };
220 }
221 throw new Error("Unexpected observer request");
222 }
223
224 allowIdentity(): void {
225 this.resolveIdentityAllowed();
226 }
227}
