 297   private readConfiguredDirectory(
 298     snapshot: WorkspaceSnapshot,
 299     root: string,
 300     extensions: readonly string[],
 301     excludes: readonly string[] | undefined,
 302     includes: readonly string[] | undefined,
 303     depth: number | undefined,
 304   ): string[] {
 305     const paths = new WorkspacePathDialect(snapshot.root);
 306     const normalizedRoot = paths.normalize(root);
 307     return snapshot.files
 308       .filter((file) => {
 309         const relative = paths.relative(normalizedRoot, file.absolute);
 310         if (!paths.contains(normalizedRoot, file.absolute)) return false;
 311         if (depth !== undefined && relative.split("/").length - 1 > depth) return false;
 312         if (!extensions.some((extension) => file.absolute.endsWith(extension))) return false;
 313         if (
 314           includes &&
 315           includes.length > 0 &&
 316           !includes.some((pattern) =>
 317             TypeScriptProjectGraph.matchesConfiguredPattern(
 318               relative,
 319               pattern,
 320               normalizedRoot,
 321               paths,
 322             ),
 323           )
 324         ) {
 325           return false;
 326         }
 327         return !excludes?.some((pattern) =>

 334   private readWorkspacePackages(
 335     snapshot: WorkspaceSnapshot,
 336     configurations: readonly ParsedTypeScriptConfiguration[],
 337     observedProjectInputs: Map<string, string | null>,
 338   ): readonly WorkspacePackage[] {
 339     const paths = new WorkspacePathDialect(snapshot.root);
 340     const directories = new Map<string, string>([[paths.key(snapshot.root), snapshot.root]]);
 341     for (const configuration of configurations) {
 342       directories.set(paths.key(configuration.directory), configuration.directory);
 343     }
 344     for (const file of snapshot.files) {
 345       let directory = paths.dirname(file.absolute);
 346       while (paths.contains(snapshot.root, directory)) {
 347         directories.set(paths.key(directory), directory);
 348         if (paths.equals(directory, snapshot.root)) break;
 349         directory = paths.dirname(directory);
 350       }
 351     }
 352     const packages: WorkspacePackage[] = [];
 353     for (const directory of directories.values()) {
 354       const path = paths.join(directory, "package.json");
 355       const content = this.captureProjectInput(path, observedProjectInputs);
 356       if (content === undefined) continue;
 357       const value = TypeScriptProjectGraph.parseJson(content);
 358       const name = value?.name;
 359       if (typeof name !== "string" || !value) continue;
 360       const mappings = TypeScriptProjectGraph.packageMappings(value, name).map((mapping) => ({
 361         specifier: mapping.specifier,
 362         target: paths.resolve(directory, mapping.target),
 363       }));
 364       if (mappings.length === 0) continue;
 365       packages.push({ path, content, mappings });
 366     }
 367     return packages;
 368   }

 472   private static matchesGlob(path: string, pattern: string, caseSensitive = true): boolean {
 473     const relativePattern = pattern.replace(/^\.\//, "");
 474     const normalized =
 475       !relativePattern.includes("*") && posix.extname(relativePattern) === ""
 476         ? `${relativePattern.replace(/\/$/, "")}/**/*`
 477         : relativePattern.replace(/\/$/, "/**/*");
 478     let expression = "";
 479     for (let index = 0; index < normalized.length; index += 1) {
 480       const character = normalized[index] as string;
 481       if (character === "*" && normalized[index + 1] === "*") {
 482         if (normalized[index + 2] === "/") {
 483           expression += "(?:.*/)?";
 484           index += 2;
 485         } else {
 486           expression += ".*";
 487           index += 1;
 488         }
 489       } else if (character === "*") {
 490         expression += "[^/]*";
 491       } else {
 492         expression += character.replace(/[|\\{}()[\]^$+?.]/g, "\\$&");
 493       }
 494     }
 495     return new RegExp(`^${expression}$`, caseSensitive ? "" : "i").test(path);
 496   }

 498   private static matchesConfiguredPattern(
 499     path: string,
 500     pattern: string,
 501     root: string,
 502     paths: WorkspacePathDialect,
 503   ): boolean {
 504     const normalizedPattern = paths.normalize(pattern);
 505     const relativePattern = paths.isAbsolute(normalizedPattern)
 506       ? paths.relative(root, normalizedPattern)
 507       : normalizedPattern;
 508     return TypeScriptProjectGraph.matchesGlob(path, relativePattern, paths.caseSensitive);
 509   }

 520   private static packageMappings(
 521     value: Record<string, unknown>,
 522     packageName: string,
 523   ): readonly WorkspacePackageMapping[] {
 524     const exports = value.exports;
 525     const exportMap = TypeScriptProjectGraph.recordValue(exports);
 526     const subpathEntries = Object.entries(exportMap).filter(([subpath]) => subpath.startsWith("."));
 527     if (subpathEntries.length > 0) {
 528       return subpathEntries.flatMap(([subpath, exportTarget]) => {
 529         const target = TypeScriptProjectGraph.stringTarget(exportTarget);
 530         if (!target) return [];
 531         const specifier =
 532           subpath === "." ? packageName : `${packageName}/${subpath.replace(/^\.\//, "")}`;
 533         return [{ specifier, target }];
 534       });
 535     }
 536     const rootTarget =
 537       TypeScriptProjectGraph.stringTarget(exports) ??
 538       TypeScriptProjectGraph.stringTarget(value.types) ??
 539       TypeScriptProjectGraph.stringTarget(value.main);
 540     return rootTarget ? [{ specifier: packageName, target: rootTarget }] : [];
 541   }
 542 
 543   private static stringTarget(value: unknown): string | undefined {
 544     if (typeof value === "string") return value;
 545     const record = TypeScriptProjectGraph.recordValue(value);
 546     if (Object.keys(record).length === 0) return undefined;
 547     return [record.types, record.import, record.default]
 548       .map((candidate) => TypeScriptProjectGraph.stringTarget(candidate))
 549       .find((candidate) => candidate !== undefined);
 550   }

 552   private static recordValue(value: unknown): Record<string, unknown> {
 553     return value !== null && typeof value === "object" && !Array.isArray(value)
 554       ? (value as Record<string, unknown>)
 555       : {};
 556   }

## Context

The [daemon architecture spec](https://github.com/mohasarc/symnav/blob/main/plans/005/daemon-architecture-functional-spec.md) assigns language-neutral project discovery, input invalidation, membership, and graph publication to core. TypeScript retains tsconfig parsing, package mapping, and `ts-morph` project construction without changing refresh or semantic lookup behavior.

## Decisions

- Chose a core abstract base over leaving reusable graph mechanics in TypeScript, because discovery, invalidation, membership, and publication have no TypeScript dependency.
- Chose one fresh collector per rebuild over retaining observations across graphs, because unreachable inputs must stop invalidating while missing inputs must trigger rediscovery when they appear.
- Chose exact active-input validation over an undocumented subclass precondition, because every published input must participate in the observation set that drives invalidation.
- Chose canonical snapshot members over relative-path-only membership, because project preparation must not receive a foreign path or revision identity.
- Chose complete validation followed by one state assignment over incremental mutation, because every candidate failure must preserve the prior graph.
