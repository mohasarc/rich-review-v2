 104 function temporaryWorkspace(directories: string[], name = "workspace"): string {
 105   const parent = mkdtempSync(join(tmpdir(), `symnav-daemon-status-${name}-`));
 106   directories.push(parent);

 108   mkdirSync(join(workspace, ".git"), { recursive: true });

 693 function waitForProcess(child: ChildProcess): Promise<void> {
 694   if (child.exitCode !== null) {
 695     return child.exitCode === 0
 696       ? Promise.resolve()
 697       : Promise.reject(new Error(`Startup publisher exited with code ${String(child.exitCode)}`));
 698   }
 699   if (child.signalCode !== null) {
 700     return Promise.reject(
 701       new Error(`Startup publisher exited with signal ${String(child.signalCode)}`),
 702     );
 703   }
 704   return new Promise((resolve, reject) => {
 705     child.once("error", reject);
 706     child.once("exit", (code) => {
 707       if (code === 0) resolve();
 708       else reject(new Error(`Startup publisher exited with code ${String(code)}`));
 709     });
 710   });
 711 }

 713 function waitForKilledProcess(child: ChildProcess): Promise<void> {
 714   if (child.signalCode !== null || (child.exitCode !== null && child.exitCode !== 0)) {
 715     return Promise.resolve();
 716   }
 717   if (child.exitCode === 0) {
 718     return Promise.reject(new Error("Startup caller did not exit abruptly"));
 719   }
 720   return new Promise((resolve, reject) => {
 721     child.once("error", reject);
 722     child.once("exit", (code, signal) => {
 723       if (signal !== null || (code !== null && code !== 0)) resolve();
 724       else reject(new Error("Startup caller did not exit abruptly"));
 725     });
 726   });
 727 }

 748 async function replay(
 749   output: import("../../../src/command-execution-result.js").CommandOutput,
 750   stream: "stdout" | "stderr",
 751 ): Promise<string> {
 752   const chunks = [];
 753   for await (const record of output.records()) {
 754     if (record.stream === stream) chunks.push(Buffer.from(record.bytes));
 755   }
 756   return Buffer.concat(chunks).toString("utf8");
 757 }

 759 async function waitUntil(predicate: () => boolean, timeoutMs = 5_000): Promise<void> {
 760   const deadline = Date.now() + timeoutMs;
 761   while (Date.now() <= deadline) {
 762     if (predicate()) return;
 763     await new Promise((resolve) => setTimeout(resolve, 10));
 764   }
 765   throw new Error("Timed out waiting for daemon shutdown");
 766 }
