 226 async function startControlledDaemon(

 228   workspaceRoot: string,

 232   const registry = new DaemonRegistry(identity.registryDirectory);
 233   const instanceId = `controlled-${releasePath === undefined ? "forced" : "draining"}`;
 234   const processToken = `${instanceId}-token`;
 235   const readyPath = join(stateDirectory, `${instanceId}-ready`);
 236   const requestStartedPath = join(stateDirectory, `${instanceId}-request`);
 237   const lease = registry.acquireStartup(identity, instanceId);
 238   if (lease === undefined) throw new Error("Expected controlled daemon startup ownership");
 239   const child = spawn(
 240     process.execPath,
 241     [
 242       fileURLToPath(new URL("../../../node_modules/tsx/dist/cli.mjs", import.meta.url)),
 243       fileURLToPath(new URL("../../helpers/workspace-daemon-stuck.ts", import.meta.url)),
 244       workspaceRoot,
 245       stateDirectory,
 246       instanceId,
 247       processToken,
 248       readyPath,
 249       requestStartedPath,
 250       ...(releasePath === undefined ? [] : [releasePath]),
 251     ],
 252     { stdio: "ignore" },
 253   );
 254   await waitUntil(() => existsSync(`${readyPath}.boot`));
 255   const daemonPid = Number(readFileSync(`${readyPath}.boot`, "utf8"));
 256   const record: DaemonRecord = {
 257     schemaVersion: DAEMON_RECORD_SCHEMA_VERSION,
 258     protocolVersion: DAEMON_PROTOCOL_VERSION,
 259     symnavVersion: "test",
 260     workspaceRoot,
 261     workspaceKey: identity.workspaceKey,
 262     stateKey: identity.stateKey,
 263     identityKey: identity.identityKey,
 264     instanceId,
 265     processToken,
 266     endpoint: identity.endpoint(instanceId),
 267     pid: daemonPid,
 268     state: "starting",
 269     startedAt: Date.now(),
 270     memoryCapBytes: Number.MAX_SAFE_INTEGER,
 271   };
 272   if (!registry.writeStartingIfStartupOwner(identity, record)) {
 273     throw new Error("Controlled daemon lost startup ownership");
 274   }

 276   lease.release();
 277   const readyRecord = registry.read(identity);
 278   if (readyRecord?.state !== "ready") throw new Error("Controlled daemon did not become ready");
 279   return { child, record: readyRecord, requestStartedPath };

 282 function runBuiltStop(
 283   cwd: string,
 284   stateDirectory: string,
 285 ): Promise<{ readonly status: number | null; readonly stdout: string; readonly stderr: string }> {
 286   const child = spawn(
 287     process.execPath,
 288     [fileURLToPath(new URL("../../../dist/cli.js", import.meta.url)), "daemon", "stop", "--json"],
 289     {
 290       cwd,
 291       env: { ...process.env, SYMNAV_STATE_DIR: stateDirectory, SYMNAV_TELEMETRY: "0" },
 292       stdio: ["ignore", "pipe", "pipe"],
 293     },
 294   );
 295   let stdout = "";
 296   let stderr = "";
 297   child.stdout?.setEncoding("utf8").on("data", (chunk: string) => {
 298     stdout += chunk;
 299   });
 300   child.stderr?.setEncoding("utf8").on("data", (chunk: string) => {
 301     stderr += chunk;
 302   });
 303   return new Promise((resolve, reject) => {
 304     child.once("error", reject);
 305     child.once("exit", (status) => resolve({ status, stdout, stderr }));
 306   });
 307 }

 309 function waitForProcess(child: ChildProcess): Promise<void> {
 310   if (child.exitCode !== null) {
 311     return child.exitCode === 0
 312       ? Promise.resolve()
 313       : Promise.reject(new Error(`Controlled daemon exited with code ${String(child.exitCode)}`));
 314   }
 315   return new Promise((resolve, reject) => {
 316     child.once("error", reject);
 317     child.once("exit", (code) => {
 318       if (code === 0) resolve();
 319       else reject(new Error(`Controlled daemon exited with code ${String(code)}`));
 320     });
 321   });
 322 }

 324 async function waitUntil(predicate: () => boolean): Promise<void> {
 325   const deadline = Date.now() + 5_000;
 326   while (Date.now() <= deadline) {
 327     if (predicate()) return;
 328     await new Promise((resolve) => setTimeout(resolve, 10));
 329   }
 330   throw new Error("Timed out waiting for controlled daemon state");
 331 }
