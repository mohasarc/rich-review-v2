 181   private candidateIndex(
 182     request: RevisionedBackendPreparationRequest<PreparedDetails>,
 183     preparedFiles: readonly RevisionedBackendPreparedFile<PreparedDetails>[],
 184   ): RevisionedBackendIndex<PreparedDetails> {
 185     const effectiveByRelativePath = new Map(
 186       request.effectiveFiles.map((file) => [file.relative, file]),
 187     );
 188     if (effectiveByRelativePath.size !== request.effectiveFiles.length) {
 189       throw new Error("Revisioned backend preparation has an incomplete final key set");
 190     }
 191     const preparedByRelativePath = new Map<
 192       string,
 193       RevisionedBackendPreparedFile<PreparedDetails>
 194     >();
 195     for (const prepared of preparedFiles) {
 196       if (preparedByRelativePath.has(prepared.file.relative)) {
 197         throw new Error(`Revisioned backend preparation duplicated ${prepared.file.relative}`);
 198       }
 199       const effective = effectiveByRelativePath.get(prepared.file.relative);
 200       if (!effective) {
 201         throw new Error(
 202           `Revisioned backend preparation returned outside path ${prepared.file.relative}`,
 203         );
 204       }
 205       if (!RevisionedBackendState.sameRevision(prepared.file, effective)) {
 206         throw new Error(
 207           `Revisioned backend preparation returned wrong revision for ${prepared.file.relative}`,
 208         );
 209       }
 210       preparedByRelativePath.set(prepared.file.relative, prepared);
 211     }
 212     for (const change of request.changes) {
 213       if (!preparedByRelativePath.has(change.file.relative)) {
 214         throw new Error(
 215           `Revisioned backend preparation omitted changed path ${change.file.relative}`,
 216         );
 217       }
 218     }
 219     const candidateByRelativePath = new Map(this.index.byRelativePath);
 220     for (const removed of request.removedFiles) {
 221       candidateByRelativePath.delete(removed.file.relative);
 222     }
 223     for (const prepared of preparedFiles) {
 224       candidateByRelativePath.set(prepared.file.relative, prepared);
 225     }
 226     if (
 227       candidateByRelativePath.size !== effectiveByRelativePath.size ||
 228       [...candidateByRelativePath.keys()].some((path) => !effectiveByRelativePath.has(path))
 229     ) {
 230       throw new Error("Revisioned backend preparation has an incomplete final key set");
 231     }
 232     return RevisionedBackendState.buildIndex(candidateByRelativePath, this.index);
 233   }

 248   private static buildIndex<PreparedDetails>(
 249     byRelativePath: ReadonlyMap<string, RevisionedBackendPreparedFile<PreparedDetails>>,
 250     previous: RevisionedBackendIndex<PreparedDetails>,
 251   ): RevisionedBackendIndex<PreparedDetails> {
 252     const declarationsByIdentity = new Map<string, SymbolOverviewNode[]>();
 253     const declarationsByRelativePath = new Map<string, readonly SymbolOverviewNode[]>();
 254     const diagnosticsByRelativePath = new Map<string, readonly NavigationDiagnostic[]>();
 255     const relativePathByAbsolute = new Map<string, string>();
 256     for (const prepared of byRelativePath.values()) {
 257       const previousPrepared = previous.byRelativePath.get(prepared.file.relative);
 258       const unchanged = previousPrepared === prepared;
 259       const declarations = unchanged
 260         ? (previous.declarationsByRelativePath.get(prepared.file.relative) ?? [])
 261         : OverviewTree.walkSymbols(prepared.entries.entries);
 262       declarationsByRelativePath.set(prepared.file.relative, declarations);
 263       diagnosticsByRelativePath.set(
 264         prepared.file.relative,
 265         unchanged
 266           ? (previous.diagnosticsByRelativePath.get(prepared.file.relative) ?? [])
 267           : (prepared.entries.diagnostics ?? []),
 268       );
 269       relativePathByAbsolute.set(prepared.file.absolute, prepared.file.relative);
 270       for (const declaration of declarations) {
 271         const key = formatSymbolIdentity(declaration.identity);
 272         const matches = declarationsByIdentity.get(key) ?? [];
 273         matches.push(declaration);
 274         declarationsByIdentity.set(key, matches);
 275       }
 276     }
 277     return {
 278       byRelativePath,
 279       declarationsByIdentity,
 280       declarationsByRelativePath,
 281       diagnosticsByRelativePath,
 282       relativePathByAbsolute,
 283     };
 284   }

 288       current.relative === incoming.relative &&
 289       current.absolute === incoming.absolute &&

 370       for (const declaration of prepared.declarations) {
 371         const identity = DeclarationLocator.identityKey(declaration.identity);
 372         const declarations = declarationsByIdentity.get(identity) ?? [];
 373         declarations.push(declaration);
 374         declarationsByIdentity.set(identity, declarations);
 375       }
 376       declarationsByPosition.set(prepared.file.relative, prepared.declarationsByPosition);
 377       relativePathByAbsolute.set(prepared.file.absolute, prepared.file.relative);
 378     }
 379     return {
 380       byRelativePath,
 381       declarationsByIdentity,
 382       declarationsByPosition,
 383       relativePathByAbsolute,
 384     };
 385   }

 387   private publishProjectRemovals(
 388     preparedFiles: readonly PreparedFileState[],
 389     removedRelativePaths: readonly string[],
 390   ): void {
 391     const obsoleteAbsolutePaths = removedRelativePaths.flatMap((relativePath) => {
 392       const prepared = this.preparedIndex.byRelativePath.get(relativePath);
 393       return prepared ? [prepared.file.absolute] : [];
 394     });
 395     for (const prepared of preparedFiles) {
 396       const previous = this.preparedIndex.byRelativePath.get(prepared.file.relative);
 397       if (previous && previous.file.absolute !== prepared.file.absolute) {
 398         obsoleteAbsolutePaths.push(previous.file.absolute);
 399       }
 400     }
 401     for (const absolutePath of obsoleteAbsolutePaths) {
 402       const sourceFile = this.project.getSourceFile(absolutePath);
 403       if (sourceFile) {
 404         this.project.removeSourceFile(sourceFile);
 405       }
 406     }
 407   }

 412       absolutePath: file.absolute,
 413       size: file.metadata.size,
 414       modifiedAtMs: file.metadata.modifiedAtMs,
 415       changeToken: file.metadata.changeToken,
 416     };
 417   }

## Context

The daemon architecture spec assigns reusable revision comparison, prepared-file indexing, and publication to core, but TypeScript still owned them after #123 moved source caching. This layer moves those mechanics into core while preserving selection retention, TypeScript source/node identity, and `LanguageBackend` results and errors.

## Shape

Before — TypeScript state mixed portable indexes with toolchain mutation:

```mermaid

    Backend["TypeScriptBackend<br/>async LanguageBackend"] -->|uses| State["TypeScriptWorkspaceState<br/>revision diff + indexes + ts-morph"]
    State -->|uses| Project["ts-morph Project<br/>source/node owner"]
    style State fill:#ffdddd,stroke:#cc0000
```

After — core publishes portable candidate state; TypeScript supplies one mutation transaction:


## Decisions

- Chose an incremental overlay candidate over requiring complete preparation output, because another language backend may prepare only changed files.
- Chose candidate validation before subclass commit over post-commit validation, because malformed output must leave portable and toolchain state unpublished.
- Chose one-file selection transactions for `ensureFiles` over one aggregate transaction, because existing behavior retains earlier progress when a later file fails.
- Chose a TypeScript-owned mutation journal over core mutation callbacks, because source files and node restoration are `ts-morph` concerns.
- Chose reversible path staging before source removal over recreating removed sources, because published lookups retain exact `SourceFile` handles.
