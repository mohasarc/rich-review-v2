 157   private async routeFor(
 158     identity: DaemonWorkspaceIdentity,
 159     runtime: DaemonDispatchRuntime,
 160     symnavVersion: string,
 161   ): Promise<DaemonRouteSnapshot> {
 162     let record: DaemonRecord | undefined;
 163     try {
 164       record = runtime.registry.read(identity);
 165     } catch {
 166       return { kind: "cold", reason: "recovering" };
 167     }
 168     if (record === undefined) return { kind: "cold", reason: "absent" };
 169     if (record.state === "starting") return { kind: "cold", reason: "starting" };
 170     if (record.symnavVersion !== symnavVersion) {
 171       return { kind: "fallback", reason: "incompatible" };
 172     }
 173     let observation: DaemonObservation;
 174     try {
 175       observation = await runtime.observer.observe(record);
 176     } catch {
 177       return { kind: "cold", reason: "recovering" };
 178     }
 179     if (observation.kind === "responsive") {
 180       if (observation.pong.symnavVersion !== symnavVersion) {
 181         return { kind: "fallback", reason: "incompatible" };
 182       }
 183       if (observation.pong.state === "starting") {
 184         return { kind: "cold", reason: "recovering" };
 185       }
 186       return { kind: "warm", record };
 187     }
 188     if (observation.kind === "starting") return { kind: "cold", reason: "starting" };
 189     if (observation.kind === "unresponsive") return { kind: "cold", reason: "recovering" };
 190     if (observation.kind === "exited") {
 191       try {
 192         runtime.registry.removeIfProcess(identity, record.instanceId, record.processToken);
 193       } catch {}
 194       return { kind: "fallback", reason: "dead" };
 195     }
 196     return { kind: "fallback", reason: "incompatible" };
 197   }

 229         return { mode: "warm", result: ControlledCommandResult.acceptedRequestDidNotComplete() };
 230       }
 231       return {
 232         mode: "warm",

 234       };
 235     } catch {
 236       return { mode: "warm", result: ControlledCommandResult.acceptedRequestDidNotComplete() };
 237     }

## Context

Daemon process ownership still lived in CLI-local mechanisms, and hosts needed registry, transport, startup, and lifecycle knowledge to compose daemon execution. This group establishes the package-owned process and client boundary while leaving the active CLI consumer switch to the next PR.

## Decisions

- Chose a Node-free public façade with a dynamically loaded internal runtime over exposing Node-backed mechanisms, because host declarations must not acquire Node ambient dependencies.
- Chose ordered lazy routing guards over eager registry and transport observation, because the first routing decision must prevent every later side effect.
- Chose package staging with a frozen CLI compatibility graph over switching production consumers during relocation, because mechanism ownership and host invocation coordination need separate review boundaries.
- Chose package-owned result capture and controlled failure outputs over host-provided storage or output factories, because warm transfer cleanup and replay safety belong to the daemon client.
- Chose to preserve acceptance-based idle timing over correcting it during extraction, because readiness- and completion-based lifetime changes are explicitly deferred behavior.
