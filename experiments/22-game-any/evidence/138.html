 110   append(bytes: Buffer): readonly unknown[] {
 111     this.buffered = Buffer.concat([this.buffered, bytes]);
 112     const values: unknown[] = [];
 113     while (this.buffered.length >= 4) {
 114       const payloadLength = this.buffered.readUInt32BE(0);
 115       if (payloadLength > this.maximumFrameBytes) {
 116         throw new Error(`Daemon frame exceeds ${this.maximumFrameBytes} bytes`);

 123       } catch {

 125       }
 126     }

 580           completionSettled = true;
 581           socket.end();
 582           await output.dispose();
 583           resolveCompletion({ status: "failed", code: failedCode });
 584         }
 585       };
 586       socket.on("data", (bytes) => {
 587         socket.pause();
 588         consumption = consumption
 589           .then(() => consume(Buffer.isBuffer(bytes) ? bytes : Buffer.from(bytes)))
 590           .then(() => {
 591             if (!socket.destroyed && !terminal) socket.resume();
 592           });
 593         void consumption.catch(fail);
 594       });
 595       socket.once("end", () => {
 596         void consumption
 597           .then(() => {
 598             if (terminal && completionSettled) return;
 599             if (resume()) return;
 600             decoder.assertComplete();
 601             fail(
 602               new DaemonTransportError(
 603                 "closed",
 604                 delivery,
 605                 acceptance === undefined
 606                   ? "Daemon connection ended before acceptance"
 607                   : "Daemon connection ended after acceptance before completion",
 608                 acceptance?.instanceId,
 609               ),
 610             );
 611           })
 612           .catch(fail);
 613       });
 614     });
 615   }

 629   private fetchCompletion(
 630     endpoint: string,
 631     request: DaemonExecuteRequest,
 632     output: OrderedCommandOutput,
 633     transfer: DaemonResultTransferReceiver,
 634   ): DaemonExecutionReceipt["completion"] {
 635     return new Promise((resolve, reject) => {
 636       const decoder = new DaemonTransferFrameDecoder(DAEMON_MAXIMUM_CONTROL_FRAME_BYTES);
 637       const socket = createConnection(endpoint);
 638       let ended = false;
 639       let settled = false;
 640       let consumption = Promise.resolve();
 641       const fail = (error: unknown): void => {
 642         if (settled) return;
 643         settled = true;
 644         socket.destroy();
 645         reject(LocalDaemonTransport.transportError(error, "accepted"));
 646       };
 647       socket.once("error", (error) => {
 648         void consumption
 649           .then(() =>
 650             fail(new DaemonTransportError("closed", "accepted", error.message, request.instanceId)),
 651           )
 652           .catch(fail);
 653       });
 654       socket.once("connect", () => {
 655         this.writeFrame(socket, {
 656           kind: "result-fetch",
 657           protocolVersion: request.protocolVersion,
 658           instanceId: request.instanceId,
 659           processToken: request.processToken,
 660           requestId: request.requestId,
 661           offset: transfer.nextOffset,
 662         });
 663       });
 664       const consume = async (bytes: Buffer): Promise<void> => {
 665         let completedResult = false;
 666         let failedCode: DaemonExecutionFailureCode | undefined;
 667         for (const value of decoder.append(bytes)) {
 668           if (ended) throw new Error("Daemon resumed with a duplicate terminal frame");
 669           if (LocalDaemonTransport.isResultChunk(value)) {
 670             await transfer.acceptChunk(value);
 671             continue;
 672           }
 673           const frame = LocalDaemonTransport.executionFrameFor(request, value);
 674           if (frame.kind === "result-manifest") {
 675             transfer.acceptManifest(frame);
 676             continue;
 677           }
 678           if (frame.kind === "execution-failed") {
 679             ended = true;
 680             failedCode = frame.code;
 681             continue;
 682           }
 683           if (frame.kind !== "result-end") {
 684             throw new Error("Daemon resumed with an invalid terminal frame");
 685           }
 686           transfer.acceptEnd(frame);
 687           ended = true;
 688           completedResult = true;
 689         }
 690         if (completedResult) {
 691           const result = await transfer.finish();
 692           const manifest = transfer.manifest;
 693           if (manifest === undefined) throw new Error("Completion manifest is missing");
 694           await this.acknowledgeResult(endpoint, request, manifest);
 695           settled = true;
 696           socket.end();
 697           resolve({ status: "completed", result });
 698         }
 699         if (failedCode !== undefined) {
 700           settled = true;
 701           socket.end();
 702           await output.dispose();
 703           resolve({ status: "failed", code: failedCode });
 704         }
 705       };
 706       socket.on("data", (bytes) => {
 707         socket.pause();
 708         consumption = consumption
 709           .then(() => consume(Buffer.isBuffer(bytes) ? bytes : Buffer.from(bytes)))
 710           .then(() => {
 711             if (!socket.destroyed && !ended) socket.resume();
 712           });
 713         void consumption.catch(fail);
 714       });
 715       socket.once("end", () => {
 716         void consumption
 717           .then(() => {
 718             if (ended) return;
 719             decoder.assertComplete();
 720             fail(new Error("Daemon result resume ended before completion"));
 721           })
 722           .catch(fail);
 723       });
 724     });
 725   }

 727   private async acknowledgeResult(
 728     endpoint: string,
 729     request: DaemonExecuteRequest,
 730     manifest: Extract<DaemonExecutionServerFrame, { kind: "result-manifest" }>["manifest"],
 731   ): Promise<void> {
 732     const acknowledgement: DaemonResultAcknowledgement = {
 733       kind: "result-ack",
 734       protocolVersion: request.protocolVersion,
 735       instanceId: request.instanceId,
 736       processToken: request.processToken,
 737       requestId: request.requestId,
 738       transferId: manifest.transferId,
 739     };
 740     await new Promise<void>((resolve, reject) => {
 741       const decoder = new DaemonFrameDecoder(this.maximumFrameBytes);
 742       const socket = createConnection(endpoint);
 743       let responseReceived = false;
 744       let settled = false;
 745       const fail = (error: unknown): void => {
 746         if (settled) return;
 747         settled = true;
 748         socket.destroy();
 749         reject(error);
 750       };
 751       const succeed = (): void => {
 752         if (settled) return;
 753         settled = true;
 754         socket.destroy();
 755         resolve();
 756       };
 757       const complete = (): void => {
 758         try {
 759           decoder.assertComplete();
 760           if (!responseReceived) throw new Error("Daemon acknowledgement response is missing");
 761           succeed();
 762         } catch (error) {
 763           fail(error);
 764         }
 765       };
 766       socket.setTimeout(this.requestTimeoutMs, () =>
 767         fail(new Error("Daemon result acknowledgement timed out")),
 768       );
 769       socket.once("error", fail);
 770       socket.once("connect", () => {
 771         try {
 772           this.writeFrame(socket, acknowledgement);
 773         } catch (error) {
 774           fail(error);
 775         }
 776       });
 777       socket.on("data", (bytes) => {
 778         try {
 779           const values = decoder.append(Buffer.isBuffer(bytes) ? bytes : Buffer.from(bytes));
 780           for (const response of values) {
 781             if (responseReceived) throw new Error("Duplicate daemon result acknowledgement");
 782             LocalDaemonTransport.assertResponse(response);
 783             if (
 784               response.kind !== "result-acknowledged" ||
 785               response.instanceId !== request.instanceId ||
 786               response.processToken !== request.processToken ||
 787               response.requestId !== request.requestId ||
 788               response.transferId !== manifest.transferId
 789             ) {
 790               throw new Error("Invalid daemon result acknowledgement");
 791             }
 792             responseReceived = true;
 793           }
 794           if (responseReceived) socket.end();
 795         } catch (error) {
 796           fail(error);
 797         }
 798       });
 799       socket.once("end", complete);
 800       socket.once("close", () => {
 801         if (!settled) complete();
 802       });
 803     });
 804   }

 865   private serve(
 866     socket: Socket,
 867     handler: (request: DaemonRequest, send: DaemonServerSend) => Promise<DaemonResponse | void>,
 868   ): void {
 869     const decoder = new DaemonFrameDecoder(this.maximumFrameBytes);
 870     let responses = Promise.resolve();
 871     let writes = Promise.resolve();
 872     const closeListeners = new Set<() => void>();
 873     const send: DaemonServerSend = Object.assign(
 874       (message: DaemonServerMessage) => {
 875         const write = writes.then(() => this.writeServerMessage(socket, message));
 876         writes = write;
 877         return write;
 878       },
 879       {
 880         onClose: (listener: () => void): (() => void) => {
 881           closeListeners.add(listener);
 882           return () => closeListeners.delete(listener);
 883         },
 884       },
 885     );
 886     socket.on("data", (bytes) => {
 887       try {
 888         for (const value of decoder.append(Buffer.isBuffer(bytes) ? bytes : Buffer.from(bytes))) {
 889           LocalDaemonTransport.assertRequest(value);
 890           responses = responses
 891             .then(async () => {
 892               const response = await handler(value, send);
 893               if (response !== undefined) await send(response);
 894             })
 895             .catch(() => {
 896               socket.destroy();
 897             });
 898         }
 899       } catch {
 900         socket.destroy();
 901       }
 902     });
 903     socket.once("end", () => {
 904       try {
 905         decoder.assertComplete();
 906       } catch {
 907         socket.destroy();
 908       }
 909     });
 910     socket.once("close", () => {
 911       for (const listener of closeListeners) {
 912         try {
 913           listener();
 914         } catch {}
 915       }
 916       closeListeners.clear();
 917     });
 918     socket.once("error", () => socket.destroy());
 919   }

 933   private async writeEncodedServerFrame(socket: Socket, frame: Buffer): Promise<void> {
 934     const chunkSize = this.writeChunkSize ?? frame.length;
 935     for (let offset = 0; offset < frame.length; offset += chunkSize) {
 936       if (socket.destroyed) throw new Error("Daemon socket closed during response delivery");
 937       const accepted = socket.write(frame.subarray(offset, offset + chunkSize));
 938       if (!accepted) await LocalDaemonTransport.waitForDrain(socket);
 939     }
 940   }

 942   private static waitForDrain(socket: Socket): Promise<void> {
 943     return new Promise((resolve, reject) => {
 944       const cleanup = (): void => {
 945         socket.off("drain", drained);
 946         socket.off("error", failed);
 947         socket.off("close", closed);
 948       };
 949       const drained = (): void => {
 950         cleanup();
 951         resolve();
 952       };
 953       const failed = (error: Error): void => {
 954         cleanup();
 955         reject(error);
 956       };
 957       const closed = (): void => {
 958         cleanup();
 959         reject(new Error("Daemon socket closed during response delivery"));
 960       };
 961       socket.once("drain", drained);
 962       socket.once("error", failed);
 963       socket.once("close", closed);
 964     });
 965   }

 967   private encodeFrame(value: unknown): Buffer {
 968     const payload = Buffer.from(JSON.stringify(value), "utf8");
 969     if (payload.length > this.maximumFrameBytes) {
 970       throw new Error(`Daemon frame exceeds ${this.maximumFrameBytes} bytes`);
 971     }
 972     const prefix = Buffer.alloc(4);
 973     prefix.writeUInt32BE(payload.length);
 974     return Buffer.concat([prefix, payload]);
 975   }

## Decisions

- Chose daemon-scoped capture over extending CLI `OrderedCommandOutput`, because warm result storage must not depend on cold-command output internals.
- Chose to advance resume offsets after awaited append over advancing on frame receipt, because fetch must restart at the first record not durably stored.
- Chose one receiver across initial and fetch connections over rebuilding receipt state, because manifest identity, digest progress, and offsets must survive reconnection.
- Chose fresh wire decoders per connection over retaining decoder state, because framing state ends at the socket boundary while transfer state continues.
- Chose caller ownership after successful finish over receiver-owned cleanup, because replay and acknowledgement failures need one explicit disposal path.
