26   static async *decodeFile(
27     filePath: string,
28     maximumRecordBytes: number,
29   ): AsyncIterable<CommandOutputRecord> {
30     const noFollow = "O_NOFOLLOW" in constants ? constants.O_NOFOLLOW : 0;
31     const handle = await open(filePath, constants.O_RDONLY | noFollow);
32     try {
33       const metadata = await handle.stat();
34       if (!metadata.isFile()) throw new Error("Command output spool is not a regular file");
35       let position = 0;
36       let expectedSequence = 0;
37       while (position < metadata.size) {
38         const header = Buffer.alloc(RECORD_HEADER_BYTES);
39         await CompletionSpoolRecordCodec.readExact(handle, header, position);
40         position += RECORD_HEADER_BYTES;
41         const sequence = header.readUInt32BE(0);
42         const streamByte = header.readUInt8(4);
43         const length = header.readUInt32BE(5);
44         if (sequence !== expectedSequence || streamByte > 1 || length > maximumRecordBytes) {
45           throw new Error("Corrupt command output");
46         }
47         const bytes = Buffer.alloc(length);
48         await CompletionSpoolRecordCodec.readExact(handle, bytes, position);
49         position += length;
50         expectedSequence += 1;
51         yield { sequence, stream: streamByte === 0 ? "stdout" : "stderr", bytes };
52       }
53     } finally {
54       await handle.close();
55     }
56   }

401       remove: () => {
402         if (this.spools.get(requestId) !== spool) return;
403         this.spools.delete(requestId);
404       },
405       releaseCompletion: () => {
406         if (completionCounted) this.completionCount -= 1;
407         completionCounted = false;
408       },

410         this.pendingCleanup.delete(spool);
411       },
412       storage: this.storage,

437   }
