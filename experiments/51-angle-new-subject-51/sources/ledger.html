152   status(requestId: string): DaemonExecutionStatus {
153     const entry = this.entries.get(requestId);
154     if (entry === undefined) return { state: "unknown" };
155     const state = entry.state;
156     if (state.state === "queued") {
157       return { state: "queued", queuePosition: entry.queuePosition };
158     }
159     if (state.state === "running") return { state: "running", startedAt: state.startedAt };
160     if (state.state === "completed") return { state: "completed" };

187   subscribe(requestId: string, subscriber: AcceptedRequestSubscriber): () => void {
188     const entry = this.entry(requestId);
189     let requestSubscribers = this.subscribers.get(requestId);
190     if (requestSubscribers === undefined) {

214   private transitionable(requestId: string): AcceptedRequestEntry {
215     const entry = this.entry(requestId);

222   private entry(requestId: string): AcceptedRequestEntry {

228   private publish(entry: AcceptedRequestEntry): AcceptedRequestEntry {
229     this.entries.set(entry.requestId, entry);
