160   private send(
161     key: string,
162     request: DaemonNavigationWorkerRequest,
163     appendOutput: ((record: DaemonSequencedOutputRecord) => Promise<void>) | undefined = undefined,
164   ): Promise<DaemonNavigationWorkerResponse> {
165     if (this.exit !== undefined)
166       return Promise.reject(new Error("Daemon navigation worker exited"));
167     if (this.pending.has(key) || this.completed.has(key)) {
168       return Promise.reject(new Error(`Duplicate daemon navigation worker request: ${key}`));
169     }
170     const response = new Promise<DaemonNavigationWorkerResponse>((resolve, reject) => {
171       this.pending.set(key, {
172         resolve,
173         reject,
174         appendOutput:
175           appendOutput ?? (() => Promise.reject(new Error("Worker output sink is unavailable"))),
176         nextSequence: 0,
177         chunkInFlight: false,
178       });
179     });
180     this.worker.postMessage(DaemonNavigationWorkerProtocol.request(request));
181     return response;
182   }
183 
184   private async receive(value: unknown): Promise<void> {
185     let response: DaemonNavigationWorkerResponse;
186     try {
187       response = DaemonNavigationWorkerProtocol.response(value, this.maximumChunkRawBytes);
188     } catch (error) {
189       this.failCommunication(error);
190       return;
191     }
192     if (response.generation !== this.generation) return;
193     if (response.kind === "output-chunk") {
194       await this.receiveOutputChunk(response);
195       return;
196     }
197     const key = NodeDaemonNavigationWorker.responseKey(response);
198     if (this.completed.has(key)) {
199       this.failCommunication(new Error(`Duplicate daemon navigation worker response: ${key}`));
200       return;
201     }
202     const pending = this.pending.get(key);
203     if (pending === undefined) {
204       this.failCommunication(new Error(`Uncorrelated daemon navigation worker response: ${key}`));
205       return;
206     }
207     this.pending.delete(key);
208     this.completed.add(key);
209     if (response.kind === "failed") {
210       pending.reject(new Error(`Daemon navigation worker ${response.failureCode} failure`));
211       return;
212     }
213     if (response.kind === "closed") this.closeAcknowledged = true;
214     pending.resolve(response);
215   }

217   private async receiveOutputChunk(
218     response: Extract<DaemonNavigationWorkerResponse, { kind: "output-chunk" }>,
219   ): Promise<void> {
220     const key = `execute:${response.requestId}`;
221     const pending = this.pending.get(key);
222     if (
223       pending === undefined ||
224       pending.chunkInFlight ||
225       response.sequence !== pending.nextSequence
226     ) {
227       this.failCommunication(new Error(`Unexpected daemon navigation worker output: ${key}`));
228       return;
229     }
230     pending.chunkInFlight = true;
231     try {
232       const record: DaemonSequencedOutputRecord = {
233         sequence: response.sequence,
234         stream: response.stream,
235         bytes: response.bytes,
236       };
237       await pending.appendOutput(record);
238       pending.nextSequence += 1;
239       pending.chunkInFlight = false;
240       const acknowledgement: DaemonNavigationWorkerRequest = {
241         kind: "output-ack",
242         generation: this.generation,
243         requestId: response.requestId,
244         sequence: response.sequence,
245       };
246       this.worker.postMessage(DaemonNavigationWorkerProtocol.request(acknowledgement));
247     } catch (error) {
248       this.failCommunication(error);
249     }
250   }

252   private failCommunication(error: unknown): void {
253     if (this.exit !== undefined) return;
254     const failure = error instanceof Error ? error : new Error(String(error));
255     this.communicationFailure = failure;
256     this.communicationFailureCode =
257       typeof error === "object" &&
258       error !== null &&
259       "code" in error &&
260       typeof error.code === "string"
261         ? error.code
262         : undefined;
263     void this.worker.terminate();
264   }
265 
266   private finishExit(): void {
267     if (this.exit !== undefined) return;
268     const communicationErrorName = this.communicationFailure?.name;
269     const communicationFailureIdentity = this.communicationFailureCode ?? communicationErrorName;
270     const cause = communicationFailureIdentity
271       ? communicationFailureIdentity === "ERR_WORKER_OUT_OF_MEMORY"
272         ? "out-of-memory"
273         : "error"
274       : this.terminating
275         ? "terminated"
276         : this.closeAcknowledged
277           ? "closed"
278           : "error";
279     this.exit = {
280       generation: this.generation,
281       cause,
282       ...(communicationFailureIdentity === undefined
283         ? {}
284         : { errorName: communicationFailureIdentity }),
285     };
286     const failure = new DaemonNavigationWorkerExitedError(
287       this.exit,
288       this.communicationFailure?.message,
289     );
290     for (const pending of this.pending.values()) pending.reject(failure);
291     this.pending.clear();
292     this.resolveExited(this.exit);
293   }

295   private static responseKey(response: DaemonNavigationWorkerResponse): string {
296     if (response.kind === "ready") return "initialize";
297     if (response.kind === "result") return `execute:${response.requestId}`;
298     if (response.kind === "heap") return `release-transient:${response.operationId}`;
299     if (response.kind === "closed") return "close";
300     if (response.kind === "failed" && response.operationId !== undefined) {
301       return `release-transient:${response.operationId}`;
302     }
303     return response.requestId === undefined ? "initialize" : `execute:${response.requestId}`;
304   }
