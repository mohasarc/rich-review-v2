238     generated: GeneratedDaemonWorkspace,
239     stateDirectory: string,

243       const cold = await this.runCommandAsync(

246         benchmark.argv,

249       );
250       for (let repetition = 0; repetition < DAEMON_BENCHMARK_WARM_REPETITIONS; repetition += 1) {
251         const telemetryBefore = this.telemetryEvents(stateDirectory);
252         const startedAt = performance.now();
253         const warm = await this.runCommandAsync(
254           generated.workspaceRoot,
255           stateDirectory,
256           benchmark.argv,

259         );
260         const wallMs = performance.now() - startedAt;
261         const telemetryAfter = this.telemetryEvents(stateDirectory);
262         samples.push(
263           BenchmarkSampleEvidence.from(
264             command as keyof typeof generated.commands,

266             cold,
267             warm,

269             benchmark,
270             BenchmarkSampleEvidence.telemetryMatched(
271               telemetryBefore,
272               telemetryAfter,
273               command as keyof typeof generated.commands,
274             ),
275           ),
276         );
277       }
278     }
279     return samples;

282   private async waitForFixedDiagnostics(
283     inspector: DaemonTestingInspector,
284     canonicalWorkspaceRoot: string,
285     cursor: number,
286     expectedOperations: number,
287   ): Promise<DaemonBenchmarkDiagnostics> {
288     const deadline = performance.now() + 10_000;
289     let diagnostics = DaemonBenchmarkDiagnostics.from(
290       inspector.readDiagnostics(canonicalWorkspaceRoot),
291       cursor,
292     );
293     while (!diagnostics.ready(expectedOperations) && performance.now() < deadline) {
294       await new Promise((resolve) => setTimeout(resolve, 10));
295       diagnostics = DaemonBenchmarkDiagnostics.from(
296         inspector.readDiagnostics(canonicalWorkspaceRoot),
297         cursor,
298       );
299     }
300     return diagnostics;

361       ["resolve", generated.mutations.removeSymbol],

365     const removedEvidence = this.changedParity(
366       root,
367       stateDirectory,
368       ["resolve", generated.mutations.removeSymbol],
369       beforeRemove,

394       ["overview", generated.mutations.add],

481   private async runLargeResponseAndBusyStatus(
482     generated: GeneratedDaemonWorkspace,
483     stateDirectory: string,
484   ): Promise<LargeResponseEvidence> {
485     const relativePath = "large-response.ts";
486     const sourcePath = join(generated.workspaceRoot, relativePath);
487     const longType = `"${"x".repeat(90_000)}"`;
488     const declarations = Array.from(
489       { length: 100 },
490       (_, index) =>
491         `export function largeSymbol${String(index).padStart(5, "0")}(value: ${longType}): string { return value; }\n`,
492     );
493     writeFileSync(sourcePath, declarations.join(""), "utf8");
494     const argv = ["overview", relativePath] as const;
495     const cold = await this.runCommandAsync(
496       generated.workspaceRoot,
497       stateDirectory,
498       argv,
499       false,
500       false,
501     );
502     const telemetryBefore = this.telemetryEvents(stateDirectory);
503     let warmSettled = false;
504     const warmPromise = this.runCommandAsync(
505       generated.workspaceRoot,
506       stateDirectory,
507       argv,
508       true,
509       true,
510     ).finally(() => {
511       warmSettled = true;
512     });
513     let busyStatusObserved = false;
514     let statusMaximumMs = 0;
515     while (!busyStatusObserved && !warmSettled) {
516       const statusStartedAt = performance.now();
517       const status = this.runCommand(
518         generated.workspaceRoot,
519         stateDirectory,
520         ["daemon", "status", "--json"],
521         false,
522       );
523       statusMaximumMs = Math.max(statusMaximumMs, performance.now() - statusStartedAt);
524       if (status.status !== 0) throw new Error("Daemon benchmark busy status failed");
525       busyStatusObserved = DaemonScaleBenchmarkHarness.hasBusyDaemon(status.stdout);
526       if (!busyStatusObserved) await DaemonScaleBenchmarkHarness.yieldToChild();
527     }
528     const warm = await warmPromise;
529     const telemetryAfter = this.telemetryEvents(stateDirectory);
530     unlinkSync(sourcePath);
531     return {
532       responseBytes: Buffer.byteLength(warm.stdout) + Buffer.byteLength(warm.stderr),
533       stdoutParity: cold.stdout === warm.stdout,
534       stderrParity: cold.stderr === warm.stderr,
535       exitParity: cold.status === warm.status,
536       busyStatusObserved,
537       statusMaximumMs,
538       telemetryMatched: BenchmarkInvocationTelemetry.matches({
539         before: telemetryBefore,
540         after: telemetryAfter,
541         command: "overview",
542       }),
543     };
544   }

546   private runCommandAsync(
547     workspaceRoot: string,
548     stateDirectory: string,
549     argv: readonly string[],
550     daemon: boolean,
551     telemetry: boolean,
552   ): Promise<RunSymnavBinaryResult> {
553     return new Promise((resolveResult, reject) => {
554       const child = spawn(process.execPath, [DaemonScaleBenchmarkHarness.cliPath, ...argv], {
555         cwd: workspaceRoot,
556         env: {
557           ...process.env,
558           SYMNAV_STATE_DIR: stateDirectory,
559           SYMNAV_DAEMON: daemon ? "1" : "0",
560           SYMNAV_TELEMETRY: telemetry ? "1" : "0",
561         },
562       });
563       let stdout = "";
564       let stderr = "";
565       child.stdout.setEncoding("utf8").on("data", (chunk: string) => (stdout += chunk));
566       child.stderr.setEncoding("utf8").on("data", (chunk: string) => (stderr += chunk));
567       child.once("error", reject);
568       child.once("close", (status) => resolveResult({ status, stdout, stderr }));
569     });
570   }

744 export class DaemonBenchmarkSemanticResult {
745   static matches(
746     expectation: DaemonBenchmarkResultExpectation | undefined,
747     stdout: string,
748   ): boolean {
749     if (expectation === undefined) return stdout.trim().length > 0;
750     let result: Record<string, unknown>;
751     try {
752       const parsed = JSON.parse(stdout) as unknown;
753       if (!this.isRecord(parsed)) return false;
754       result = parsed;
755     } catch {
756       return false;
757     }
758     if (expectation.kind === "overview") {
759       return this.arrayLength(result.entries) === expectation.symbols;
760     }
761     if (expectation.kind === "resolve" || expectation.kind === "definition") {
762       return this.arrayLength(result.symbols) === expectation.symbols;
763     }
764     if (expectation.kind === "references") return result.total === expectation.total;
765     if (expectation.kind === "context") {
766       return (
767         this.edgeCount(result.callers) === expectation.callers &&
768         this.edgeCount(result.callees) === expectation.callees &&
769         this.arrayLength(result.history) === expectation.history
770       );
771     }
772     if (expectation.kind === "graph") {
773       return (
774         this.pathCount(result.incoming) === expectation.incomingPaths &&
775         this.pathCount(result.outgoing) === expectation.outgoingPaths
776       );
777     }
778     return (
779       typeof result.totalEvents === "number" &&
780       Array.isArray(result.perCommand) &&
781       Array.isArray(result.outcomes) &&
782       this.isRecord(result.duration)
783     );
784   }

787     if (!this.isRecord(value)) return -1;
788     const visible = this.arrayLength(value.sortedEdges);
789     const omitted = value.omittedCertainEdgeCount;
790     return visible < 0 || typeof omitted !== "number" ? -1 : visible + omitted;
791   }

794     if (!this.isRecord(value) || typeof value.totalPathCount !== "number") return -1;
795     return value.totalPathCount;
796   }

799     return Array.isArray(value) ? value.length : -1;
