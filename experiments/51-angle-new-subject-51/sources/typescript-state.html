135   private readonly mutations: ProjectMutation[] = [];
136   private rolledBack = false;

138   constructor(
139     private readonly project: Project,

141     private readonly request: RevisionedBackendPreparationRequest<TypeScriptPreparedFileDetails>,

146   async prepare(): Promise<
147     readonly RevisionedBackendPreparedFile<TypeScriptPreparedFileDetails>[]
148   > {
149     return this.request.changes.map((change) => {
150       const previous = change.kind === "changed" ? change.previous : undefined;
151       const existingSourceFile =
152         previous === undefined || previous.file.absolute === change.file.absolute
153           ? this.project.getSourceFile(change.file.absolute)
154           : undefined;
155       return this.prepareFile(change.file, existingSourceFile, previous !== undefined);
156     });
157   }
158 
159   async commit(): Promise<void> {
160     const obsoleteSourceFiles = this.obsoleteAbsolutePaths().flatMap((absolutePath) => {
161       const sourceFile = this.project.getSourceFile(absolutePath);
162       return sourceFile ? [sourceFile] : [];
163     });
164     for (const [index, sourceFile] of obsoleteSourceFiles.entries()) {
165       const absolutePath = sourceFile.getFilePath();
166       const stagingPath = this.obsoleteStagingPath(absolutePath, index);
167       sourceFile.move(stagingPath);
168       this.mutations.push({
169         rollback: () => {
170           sourceFile.move(absolutePath, { overwrite: true });
171         },
172       });
173     }
174     for (const sourceFile of obsoleteSourceFiles) {
175       this.tryForgetObsoleteSource(sourceFile);
176     }
177   }
178 
179   async rollback(): Promise<void> {
180     if (this.rolledBack) return;
181     this.rolledBack = true;
182     for (const mutation of [...this.mutations].reverse()) {
183       mutation.rollback();
184     }
185   }
186 
187   private prepareFile(
188     file: RevisionedBackendPreparedFile<TypeScriptPreparedFileDetails>["file"],
189     existingSourceFile: SourceFile | undefined,
190     readChangedSource: boolean,
191   ): RevisionedBackendPreparedFile<TypeScriptPreparedFileDetails> {
192     let sourceFile: SourceFile;
193     if (existingSourceFile && readChangedSource) {
194       const previousText = existingSourceFile.getFullText();
195       const content = this.project.getFileSystem().readFileSync(file.absolute);
196       existingSourceFile.replaceWithText(content);
197       this.mutations.push({ rollback: () => existingSourceFile.replaceWithText(previousText) });
198       sourceFile = existingSourceFile;
199     } else if (existingSourceFile) {
200       sourceFile = existingSourceFile;
201     } else {
202       sourceFile = this.project.addSourceFileAtPath(file.absolute);
203       this.mutations.push({ rollback: () => this.project.removeSourceFile(sourceFile) });
204     }
205     return this.buildPreparedFile(sourceFile, file);
206   }

208   private buildPreparedFile(
209     sourceFile: SourceFile,
210     file: RevisionedBackendPreparedFile<TypeScriptPreparedFileDetails>["file"],
211   ): RevisionedBackendPreparedFile<TypeScriptPreparedFileDetails> {
212     const declarationsByPosition = new Map<number, SymbolOverviewNode>();
213     const diagnosticSink = new CollectingDiagnosticSink();
214     const extracted = this.extractor.extract({

216       filePath: file.relative,
217       diagnostics: diagnosticSink,
218     });
219     const diagnostics = diagnosticSink.diagnostics();
220     const entries = diagnostics.length === 0 ? extracted : { ...extracted, diagnostics };
221     for (const { declaration, node } of new DeclarationLocator(sourceFile).locateAll(
222       entries.entries,
223     )) {
224       declarationsByPosition.set(node.getStart(), declaration);
225     }
226     return {
227       file,
228       entries,
229       details: { sourceFile, declarationsByPosition },
230     };
231   }

233   private obsoleteAbsolutePaths(): readonly string[] {
234     const paths = this.request.removedFiles.map((prepared) => prepared.file.absolute);
235     for (const change of this.request.changes) {
236       if (change.kind === "changed" && change.previous.file.absolute !== change.file.absolute) {
237         paths.push(change.previous.file.absolute);
238       }
239     }
240     return paths;
241   }

243   private obsoleteStagingPath(absolutePath: string, initialSuffix: number): string {
244     let suffix = initialSuffix;
245     while (true) {
246       const candidate = `${absolutePath}.symnav-obsolete-${suffix}`;
247       const sourceExists = this.project.getSourceFile(candidate) !== undefined;
248       const fileExists = this.project.getFileSystem().fileExistsSync(candidate);
249       if (!sourceExists && !fileExists) return candidate;
250       suffix += 1;
251     }
252   }
