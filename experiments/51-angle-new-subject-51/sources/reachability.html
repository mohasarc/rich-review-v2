45 class TypeScriptProductionGraph {
46   constructor(private readonly repositoryRoot: string) {}
47 
48   reachableFrom(entry: string): readonly string[] {
49     const pending = [join(this.repositoryRoot, entry)];
50     const visited = new Set<string>();
51     while (pending.length > 0) {
52       const file = pending.pop()!;
53       if (visited.has(file)) continue;
54       visited.add(file);
55       for (const dependency of this.localDependencies(file)) {
56         if (!visited.has(dependency)) pending.push(dependency);
57       }
58     }
59     return [...visited].map((file) => relative(this.repositoryRoot, file)).sort();
60   }
61 
62   private localDependencies(file: string): readonly string[] {
63     const source = readFileSync(file, "utf8");
64     return TypeScriptImportSpecifierExtractor.extract(source)
65       .filter((specifier) => specifier.startsWith("."))
66       .map((specifier) => this.resolveTypeScriptImport(file, specifier))
67       .filter((dependency): dependency is string => dependency !== undefined);
68   }

72     if (existsSync(sourcePath)) return sourcePath;
73     const indexPath = join(sourcePath, "index.ts");
