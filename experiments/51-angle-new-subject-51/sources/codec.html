157 class DaemonResultChunkCodec {
158   static encode(chunk: DaemonResultChunk, limits: DaemonWireLimits): Uint8Array {
159     DaemonResultChunkCodec.assertChunk(chunk, limits.maximumChunkRawBytes);
160     const header: DaemonResultChunkHeader = {
161       transferId: chunk.transferId,
162       requestId: chunk.requestId,
163       offset: chunk.offset,
164       sequence: chunk.sequence,
165       stream: chunk.stream,
166       payloadLength: chunk.bytes.byteLength,
167       sha256: createHash("sha256").update(chunk.bytes).digest("hex"),
168     };
169     const headerBytes = Buffer.from(JSON.stringify(header));
170     if (
171       BINARY_HEADER_LENGTH_BYTES + headerBytes.byteLength >
172       limits.maximumExecutionControlPayloadBytes
173     ) {
174       throw new Error("Daemon result frame exceeds capacity");
175     }
176     const payloadLength =
177       BINARY_HEADER_LENGTH_BYTES + headerBytes.byteLength + chunk.bytes.byteLength;
178     const encoded = Buffer.alloc(FRAME_PREFIX_BYTES + payloadLength);
179     encoded.writeUInt32BE((BINARY_FRAME_FLAG | payloadLength) >>> 0, 0);
180     encoded.writeUInt32BE(headerBytes.byteLength, FRAME_PREFIX_BYTES);
181     headerBytes.copy(encoded, FRAME_PREFIX_BYTES + BINARY_HEADER_LENGTH_BYTES);
182     Buffer.from(chunk.bytes).copy(
183       encoded,
184       FRAME_PREFIX_BYTES + BINARY_HEADER_LENGTH_BYTES + headerBytes.byteLength,
185     );
186     return encoded;
187   }

189   static decode(payload: Buffer, limits: DaemonWireLimits): DaemonResultChunk {
190     if (payload.byteLength < BINARY_HEADER_LENGTH_BYTES) {
191       throw new Error("Truncated daemon result chunk header");
192     }
193     const headerLength = payload.readUInt32BE(0);
194     if (headerLength === 0 || headerLength > payload.byteLength - BINARY_HEADER_LENGTH_BYTES) {
195       throw new Error("Invalid daemon result chunk header length");
196     }
197     if (BINARY_HEADER_LENGTH_BYTES + headerLength > limits.maximumExecutionControlPayloadBytes) {
198       throw new Error("Daemon result frame exceeds capacity");
199     }
200     let parsed: unknown;
201     try {
202       parsed = JSON.parse(
203         payload
204           .subarray(BINARY_HEADER_LENGTH_BYTES, BINARY_HEADER_LENGTH_BYTES + headerLength)
205           .toString("utf8"),
206       );
207     } catch {
208       throw new Error("Malformed daemon result chunk header");
209     }
210     const bytes = payload.subarray(BINARY_HEADER_LENGTH_BYTES + headerLength);
211     if (
212       !DaemonResultChunkCodec.isHeader(parsed, limits.maximumChunkRawBytes) ||
213       parsed.payloadLength !== bytes.byteLength
214     ) {
215       throw new Error("Invalid daemon result chunk header");
216     }
217     if (createHash("sha256").update(bytes).digest("hex") !== parsed.sha256) {
218       throw new Error("Corrupt daemon result chunk payload");
219     }
220     const chunk: DaemonResultChunk = {
221       transferId: parsed.transferId,
222       requestId: parsed.requestId,
223       offset: parsed.offset,
224       sequence: parsed.sequence,
225       stream: parsed.stream,
226       bytes: Uint8Array.from(bytes),
227     };
228     DaemonResultChunkCodec.assertChunk(chunk, limits.maximumChunkRawBytes);
229     return chunk;
230   }
