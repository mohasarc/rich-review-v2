54 findDefinitions(identity: SymbolIdentity): Promise<readonly SymbolOverviewNode[]> {

68 async findReferences(identity: SymbolIdentity): Promise<readonly SymbolReference[]> {
69 const locations = await this.referenceLocations(identity);
70 return locations.flatMap((location) => {
71 if (location.isDefinition) return [];
72 const node = this.workspaceState.nodeAt(location.relativePath, location.start);
73 if (!node) return [];
74 const sourceFile = node.getSourceFile();
75 const { line, character } = sourceFile.compilerNode.getLineAndCharacterOfPosition(
76 location.start,
77 );
78 return [
79 {
80 file: location.relativePath,
81 line: line + 1,
82 previewSource: TypeScriptSemanticQueryService.lineText(sourceFile, line),
83 matchStart: character,
84 matchEnd: character + location.length,
85 kind: classifyReferenceKind(node),
86 },
87 ];
88 });

161 private referenceLocations(
162 identity: SymbolIdentity,
163 ): Promise<readonly SemanticReferenceLocation[]> {
164 const key = formatSymbolIdentity(identity);
165 const existing = this.referencesByIdentity.get(key);
166 if (existing) return existing;

173 private findReferenceLocations(identity: SymbolIdentity): readonly SemanticReferenceLocation[] {
174 const declarationNodes = this.workspaceState
175 .locateSemanticCopies(identity)
176 .map((located) => located.node);
177 if (declarationNodes.length === 0) throw new SymbolNotFoundError(identity);
178 const locations = declarationNodes.flatMap((declarationNode) =>
179 TypeScriptSemanticQueryService.referenceEntriesOf(declarationNode).flatMap((entry) => {
180 const node = entry.getNode();
181 const relativePath = this.workspaceState.relativePathOf(node.getSourceFile());
182 return relativePath
183 ? [
184 {
185 relativePath,
186 start: node.getStart(),
187 length: node.getWidth(),
188 isDefinition: entry.isDefinition() ?? false,
189 },
190 ]
191 : [];
192 }),
193 );
194 const byLocation = new Map<string, SemanticReferenceLocation>();
195 for (const location of locations) {
196 byLocation.set(
197 `${location.relativePath}:${location.start}:${location.length}:${location.isDefinition}`,
198 location,
199 );
200 }
201 return [...byLocation.values()];
202 }

204 private async resolveCallTarget(identity: SymbolIdentity): Promise<CallTargetResolution> {
205 const definitions = await this.findDefinitions(identity);
206 if (definitions.length === 0) return { outcome: "not-found" };
207 const implementations = definitions.filter((definition) =>
208 definition.kind.nativeLabel.endsWith("-implementation"),
209 );
210 if (implementations.length > 1) {
211 return { outcome: "ambiguous", candidates: implementations };
212 }
213 if (implementations.length === 1) {
214 return { outcome: "resolved", target: implementations[0]! };
215 }
216 return { outcome: "resolved", target: definitions[0]! };
217 }

228 private nodeAtSemanticLocation(
229 location: SemanticNodeLocation,
230 sourceFile: SourceFile | undefined,
231 ): Node | undefined {
232 let node =
233 sourceFile?.getDescendantAtPos(location.start) ??
234 this.workspaceState.nodeAt(location.relativePath, location.start);
235 while (node) {
236 if (node.getStart() === location.start && node.getKind() === location.kind) return node;
237 node = node.getParent();
238 }
239 return undefined;
240 }

242 private static referenceEntriesOf(declarationNode: Node): readonly ReferencedSymbolEntry[] {
243 if (!Node.isReferenceFindable(declarationNode)) return [];
244 return declarationNode
245 .findReferences()
246 .flatMap((referencedSymbol) => referencedSymbol.getReferences());
247 }

249 private static lineText(sourceFile: SourceFile, zeroBasedLine: number): string {
250 const fullText = sourceFile.getFullText();
251 const lineStarts = sourceFile.compilerNode.getLineStarts();
252 const start = lineStarts[zeroBasedLine] ?? 0;
253 const end =
254 zeroBasedLine + 1 < lineStarts.length ? lineStarts[zeroBasedLine + 1]! : fullText.length;
255 return fullText.slice(start, end).replace(/\r?\n$/, "");
256 }
