126   async attach(attachment: DaemonDeliveryAttachment, send: DaemonServerSend): Promise<void> {
127     const entry = this.options.journal.entryFor(attachment.requestId);
128     if (entry === undefined) throw new Error("Accepted request result is missing");
129     const traceWasDisconnected = this.traceWasDisconnected(attachment.requestId);
130     const disconnectTraceConnection = this.attachOperationTraceConnection(
131       attachment.requestId,
132       send,
133     );
134     if (traceWasDisconnected) this.reattachOperationTrace(attachment.requestId);
135     try {
136       await this.deliver(send, {
137         kind: "accepted",
138         ...this.options.coordinates,
139         ...attachment,
140       });
141       if (entry.state.state === "completed") {
142         await this.deliverStoredCompletion(attachment.requestId, send);
143         return;
144       }
145       if (entry.state.state === "failed") {
146         await this.deliver(send, this.failedFrame(attachment.requestId, entry.state.code));
147         this.completeOperationTrace(attachment.requestId, "delivered");
148         return;
149       }
150     } catch (error) {
151       disconnectTraceConnection();
152       throw error;
153     }
154     let unsubscribe: (() => void) | undefined;
155     unsubscribe = this.options.journal.subscribe(attachment.requestId, (updated) => {
156       if (updated.state.state === "completed") {
157         this.trackCompletionDelivery(attachment.requestId, send, disconnectTraceConnection);
158         unsubscribe?.();
159       } else if (updated.state.state === "failed") {
160         void this.deliver(send, this.failedFrame(attachment.requestId, updated.state.code))
161           .then(() => this.completeOperationTrace(attachment.requestId, "delivered"))
162           .catch((error) =>
163             this.recordDeliveryFailure(attachment.requestId, error, disconnectTraceConnection),
164           );
165         unsubscribe?.();
166       }
167     });
168   }

174   async fetch(
175     request: AuthenticatedDaemonResultFetchRequest,
176     send: DaemonServerSend,
177   ): Promise<void> {
178     let disconnectTraceConnection: (() => void) | undefined;
179     if (this.options.journal.entryFor(request.requestId)?.state.state === "completed") {
180       const traceWasDisconnected = this.traceWasDisconnected(request.requestId);
181       disconnectTraceConnection = this.attachOperationTraceConnection(request.requestId, send);
182       if (traceWasDisconnected) this.reattachOperationTrace(request.requestId);
183     }
184     try {
185       await this.deliverStoredCompletion(request.requestId, send, request.offset);
186     } catch (error) {
187       disconnectTraceConnection?.();
188       throw error;
189     }
190   }

192   async acknowledge(
193     request: AuthenticatedDaemonResultAcknowledgement,
194   ): Promise<DaemonResultAcknowledged> {
195     const spool = await this.options.spoolStore.open(request.requestId);
196     if (spool === undefined) throw new Error("Accepted request completion is unavailable");
197     if (spool.completedManifest?.transferId !== request.transferId) {
198       throw new Error("Result acknowledgement does not match completion transfer");
199     }
200     await spool.acknowledge().catch((error) => {
201       this.recordFailure("completion-cleanup", error);
202     });
203     this.options.journal.acknowledge(request.requestId);
204     this.completeOperationTrace(request.requestId, "delivered");
205     return {
206       kind: "result-acknowledged",
207       ...this.options.coordinates,
208       requestId: request.requestId,
209       transferId: request.transferId,
210     };
211   }

213   completeRetainedTraces(): void {
214     for (const requestId of this.operationTraces.keys()) {
215       this.completeOperationTrace(requestId, "disconnected");
216     }
217   }

259   private async deliverStoredCompletion(
260     requestId: string,
261     send: DaemonServerSend,
262     offset = 0,
263   ): Promise<void> {
264     const spool = await this.options.spoolStore.open(requestId);
265     if (spool === undefined) throw new Error("Accepted request result is missing");
266     try {
267       await this.deliverCompletion(requestId, spool, offset, send);
268     } catch (error) {
269       if (!(error instanceof CompletionSpoolReadError)) throw error;
270       this.recordFailure("completion-delivery", error);
271       await spool.dispose().catch((cleanupError) => {
272         this.recordFailure("completion-cleanup", cleanupError);
273       });
274       this.options.journal.invalidateCompletion(
275         requestId,
276         "internal",
277         this.options.clock.wallNowMs(),
278       );
279       await this.deliver(send, this.failedFrame(requestId, "internal"));
280       this.completeOperationTrace(requestId, "failed");
281     }
282   }

284   private trackCompletionDelivery(
285     requestId: string,
286     send: DaemonServerSend,
287     disconnectTraceConnection: () => void,
288   ): void {
289     const delivery = this.deliverStoredCompletion(requestId, send).catch((error) =>
290       this.recordDeliveryFailure(requestId, error, disconnectTraceConnection),
291     );
292     this.completionDeliveries.set(requestId, delivery);
293     void delivery.finally(() => {
294       if (this.completionDeliveries.get(requestId) === delivery) {
295         this.completionDeliveries.delete(requestId);
296       }
297     });
298   }

300   private async deliverCompletion(
301     requestId: string,
302     spool: CompletionSpool,
303     offset: number,
304     send: DaemonServerSend,
305   ): Promise<void> {
306     const completedManifest = spool.completedManifest;
307     if (completedManifest === undefined) throw new Error("Completion manifest is missing");
308     await this.deliver(send, {
309       kind: "result-manifest",
310       ...this.options.coordinates,
311       requestId,
312       manifest: completedManifest,
313     });
314     for await (const record of spool.read(offset)) {
315       await this.deliver(send, {
316         transferId: completedManifest.transferId,
317         requestId,
318         offset: record.sequence,
319         sequence: record.sequence,
320         stream: record.stream,
321         bytes: record.bytes,
322       });
323     }
324     await this.deliver(send, {
325       kind: "result-end",
326       ...this.options.coordinates,
327       requestId,
328       transferId: completedManifest.transferId,
329       rawBytes: completedManifest.rawBytes,
330       recordCount: completedManifest.recordCount,
331       sha256: completedManifest.sha256,
332     });
333     this.terminateOperationDelivery(requestId, "delivered");
334   }

369   private completeOperationTrace(requestId: string, outcome: DaemonDeliveryOutcome): void {
370     const expiration = this.operationTraceExpirations.get(requestId);
371     if (expiration !== undefined) clearTimeout(expiration);
372     this.operationTraceExpirations.delete(requestId);
373     this.operationTraceConnections.delete(requestId);
374     this.terminateOperationDelivery(requestId, outcome);
375     this.operationTraces.delete(requestId);
376   }
377 
378   private terminateOperationDelivery(requestId: string, outcome: DaemonDeliveryOutcome): void {
379     if (!this.options.journal.terminateDelivery(requestId)) return;
380     const trace = this.operationTraces.get(requestId);
381     if (trace === undefined) this.options.observer.deliveryTerminated(requestId, outcome, 0);
382     else trace.deliveryTerminated(outcome);
383   }
384 
385   private disconnectOperationTrace(requestId: string): void {
386     if (this.operationTraceExpirations.has(requestId)) return;
387     const trace = this.operationTraces.get(requestId);
388     if (trace === undefined) return;
389     trace.clientDisconnected();
390     const expiration = setTimeout(
391       () => this.expireOperationTrace(requestId),
392       this.options.policy.diagnostics.disconnectedTraceRetentionMs,
393     );
394     expiration.unref();
395     this.operationTraceExpirations.set(requestId, expiration);
396     this.enforceOperationTraceCapacity();
397   }
398 
399   private attachOperationTraceConnection(requestId: string, send: DaemonServerSend): () => void {
400     const connectionCount = this.operationTraceConnections.get(requestId) ?? 0;
401     if (connectionCount === 0) {
402       const expiration = this.operationTraceExpirations.get(requestId);
403       if (expiration !== undefined) clearTimeout(expiration);
404       this.operationTraceExpirations.delete(requestId);
405     }
406     this.operationTraceConnections.set(requestId, connectionCount + 1);
407     let connectionClosed = false;
408     const disconnect = (): void => {
409       if (connectionClosed) return;
410       connectionClosed = true;
411       const remainingConnections = (this.operationTraceConnections.get(requestId) ?? 1) - 1;
412       if (remainingConnections > 0) {
413         this.operationTraceConnections.set(requestId, remainingConnections);
414         return;
415       }
416       this.operationTraceConnections.delete(requestId);
417       this.disconnectOperationTrace(requestId);
418     };
419     send.onClose(disconnect);
420     return disconnect;
421   }
422 
423   private traceWasDisconnected(requestId: string): boolean {
424     if (this.operationTraceExpirations.has(requestId)) return true;
425     return (
426       !this.operationTraces.has(requestId) && !this.options.journal.isDeliveryTerminated(requestId)
427     );
428   }
429 
430   private reattachOperationTrace(requestId: string): void {
431     if (this.options.journal.isDeliveryTerminated(requestId)) return;
432     const trace = this.operationTraces.get(requestId);
433     if (trace === undefined) this.options.observer.reattached(requestId);
434     else trace.reattached();
435   }
436 
437   private expireOperationTrace(requestId: string): void {
438     this.operationTraceExpirations.delete(requestId);
439     this.operationTraceConnections.delete(requestId);
440     if (!this.operationTraces.delete(requestId)) return;
441     this.options.observer.traceExpired(requestId);
442   }
443 
444   private enforceOperationTraceCapacity(): void {
445     const capacity = Math.max(1, this.options.policy.diagnostics.maximumDisconnectedTraces);
446     while (this.operationTraceExpirations.size > capacity) {
447       const oldestRequestId = this.operationTraceExpirations.keys().next().value as
448         | string
449         | undefined;
450       if (oldestRequestId === undefined) return;
451       const expiration = this.operationTraceExpirations.get(oldestRequestId);
452       if (expiration !== undefined) clearTimeout(expiration);
453       this.expireOperationTrace(oldestRequestId);
454     }
455   }

497 class ObservedDaemonCompletionWriter implements DaemonCompletionWriter {
498   constructor(
499     private readonly spool: CompletionSpool,
500     private readonly trace: DaemonOperationTrace | undefined,
501     private readonly clock: Pick<DaemonClock, "monotonicNowMs">,
502   ) {}
503 
504   append(record: DaemonSequencedOutputRecord): Promise<void> {
505     return this.spool.append(record);
506   }
507 
508   async finish(exitCode: number): Promise<CompletionSpoolManifest> {
509     const startedAt = this.clock.monotonicNowMs();
510     const manifest = await this.spool.finish(exitCode);
511     const durationMs = Math.max(0, this.clock.monotonicNowMs() - startedAt);
512     this.trace?.spooled(manifest, durationMs);
513     return manifest;
514   }
515 
516   dispose(): Promise<void> {
517     return this.spool.dispose();
518   }
