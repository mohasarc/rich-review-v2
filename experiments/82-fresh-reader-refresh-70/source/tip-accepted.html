94   private async executeAccepted(
95     request: AuthenticatedDaemonExecuteRequest,
96     trace: DaemonOperationTrace,
97   ): Promise<void> {
98     let completion: DaemonCompletionWriter | undefined;
99     try {
100       await this.options.queue.enqueue(
101         {
102           requestId: request.requestId,
103           command: request.commandName,
104           acceptedAt: this.options.clock.monotonicNowMs(),
105         },
106         async () => {
107           trace.turnStarted(this.options.resourceSupervisor.snapshot.generation);
108           try {
109             completion = await this.options.delivery.createCompletion(request.requestId);
110             this.options.ledger.markRunning(request.requestId, this.options.clock.wallNowMs());
111             const response = await this.options.worker.execute(
112               request.requestId,
113               { commandName: request.commandName, request: request.request },
114               completion,
115             );
116             this.options.resourceSupervisor.workerHeapReported(
117               response.generation,
118               response.resources.workerHeapUsedBytes,
119               response.resources.workerHeapLimitBytes,
120               response.resources.peakWorkerHeapUsedBytes,
121             );
122             trace.workerCompleted(
123               {
124                 freshnessMs: response.durations.freshnessMs,
125                 navigationMs: response.durations.navigationMs,
126                 renderMs: response.durations.renderMs,
127                 workerOutputMs: response.durations.outputMs,
128               },
129               response.refresh,
130             );
131             await completion.finish(response.result.exitCode);
132             this.lastCompletedMonotonicAt = this.options.clock.monotonicNowMs();
133             const workspaceExists = await this.options.processLifecycle.workspaceExists();
134             trace.executionTerminated("completed");
135             this.options.ledger.complete(
136               request.requestId,
137               request.requestId,
138               this.options.clock.wallNowMs(),
139             );
140             await this.options.delivery.trackedCompletion(request.requestId);
141             if (!workspaceExists) {
142               await this.options.processLifecycle.workspaceDeletedAfterDelivery();
143             }
144           } finally {
145             this.scheduleTurnCompleteResourceSample();
146           }
147         },
148       );
149     } catch (error) {
150       trace.executionTerminated("failed");
151       this.recordFailure("request", "internal", error);
152       const shutdown = this.options.processLifecycle.shutdownSnapshot();
153       const code = DaemonExecutionFailures.classify({
154         resourceInterrupted: this.resourceInterruptedRequests.delete(request.requestId),
155         responseCapacityExceeded: error instanceof CompletionSpoolCapacityError,
156         workerExited: error instanceof DaemonNavigationWorkerExitedError,
157         ...(shutdown.failureCode === undefined
158           ? {}
159           : { shutdownFailureCode: shutdown.failureCode }),
160         shutdownStarted: shutdown.started,
161       });
162       await completion?.dispose().catch((cleanupError) => {
163         this.recordFailure("completion-cleanup", "internal", cleanupError);
164       });
165       this.options.ledger.fail(request.requestId, code, this.options.clock.wallNowMs());
166     } finally {
167       if (this.options.queue.isIdle) this.options.lifetime.queueBecameIdle();
168     }
169   }
