230   private writeClaimedStartingRecord(
231     identity: DaemonWorkspaceIdentity,
232     record: DaemonRecord,
233   ): boolean {
234     const mutation = this.beginStartupMutation(identity);
235     if (mutation === undefined) return false;
236     try {
237       const owner = this.startupOwnershipMatches(identity, {
238         identityKey: identity.identityKey,
239         instanceId: record.instanceId,
240       });
241       if (owner === undefined) return false;
242       const adoptedOwner: StartupOwner = {
243         ...owner,
244         identityKey: identity.identityKey,
245         ownerPid: record.pid > 0 ? record.pid : owner.ownerPid,
246         processToken: record.processToken,
247         ownerKind: record.pid > 0 ? "daemon" : "launcher",
248         heartbeatAt: this.clock.wallNowMs(),
249         revision: randomUUID(),
250       };
251       const ownerPath = identity.startupOwnerPath(identity.lockPath);
252       const temporaryPath = `${identity.lockPath}.${process.pid}.${randomUUID()}.owner.tmp`;
253       writeFileSync(temporaryPath, JSON.stringify(adoptedOwner), {
254         encoding: "utf8",
255         mode: 0o600,
256       });
257       if (!mutation.isOwned() || this.startupOwnershipMatches(identity, owner) === undefined) {
258         rmSync(temporaryPath, { force: true });
259         return false;
260       }
261       this.write(record);
262       try {
263         this.replaceStartupOwner(temporaryPath, ownerPath);
264       } catch (error) {
265         rmSync(temporaryPath, { force: true });
266         if (DaemonRegistry.errorCode(error) === "ENOENT") return false;
267         throw error;
268       }
269       return this.startupOwnershipMatches(identity, adoptedOwner) !== undefined;
270     } finally {
271       mutation.release();
272     }
273   }

661     identity: DaemonWorkspaceIdentity,
662   ): RegistryStartupMutationLease | undefined {
663     const claimed = this.claimStartupMutation(identity);
664     if (claimed !== undefined) return claimed;

680   private claimStartupMutation(
681     identity: DaemonWorkspaceIdentity,
682   ): RegistryStartupMutationLease | undefined {
683     mkdirSync(identity.identityDirectory, { recursive: true, mode: 0o700 });
684     const token = randomUUID();
685     const owner: StartupMutationOwner = {
686       ownerPid: process.pid,
687       acquiredAt: this.clock.wallNowMs(),
688       token,
689     };
690     const claimPath = identity.startupMutationClaimPath(token);
691     mkdirSync(claimPath, { mode: 0o700 });
692     writeFileSync(identity.startupOwnerPath(claimPath), JSON.stringify(owner), {
693       encoding: "utf8",
694       flag: "wx",

697     try {
698       renameSync(claimPath, identity.startupMutationPath);
699       return new RegistryStartupMutationLease(this, identity, owner);
700     } catch (error) {
701       rmSync(claimPath, { recursive: true, force: true });
702       if (DaemonRegistry.startupMutationClaimWasContended(error)) return undefined;
703       if (existsSync(identity.startupMutationPath)) return undefined;

713   private recoverStartupMutation(
714     identity: DaemonWorkspaceIdentity,
715     observedOwner: StartupMutationOwner | undefined,
716   ): boolean {
717     const currentOwner = DaemonRegistry.readStartupMutationOwner(
718       identity,
719       identity.startupMutationPath,
720     );
721     if (!DaemonRegistry.sameOptionalStartupMutationOwner(currentOwner, observedOwner)) return false;
722     const recoveryToken = observedOwner?.token ?? "ownerless";
723     const releasedPath = identity.releasedStartupMutationPath(recoveryToken);
724     try {
725       renameSync(identity.startupMutationPath, releasedPath);
726       return true;
727     } catch (error) {
728       if (existsSync(releasedPath)) return true;
729       if (DaemonRegistry.errorCode(error) === "ENOENT") return false;
730       throw error;
731     }
732   }

790   private static isCurrentRecord(record: DaemonRecord): boolean {
791     if (
792       record.schemaVersion !== DAEMON_RECORD_SCHEMA_VERSION ||
793       record.protocolVersion !== DAEMON_PROTOCOL_VERSION
794     )
795       return false;
796     if (record.state === "starting") {
797       return record.readyAt === undefined && record.fileCount === undefined;
798     }
799     return typeof record.readyAt === "number" && typeof record.fileCount === "number";
800   }

802   private static isStoredRecord(value: unknown): value is DaemonRecord {
803     if (typeof value !== "object" || value === null) return false;
804     const record = value as Record<string, unknown>;
805     return (
806       Number.isInteger(record.schemaVersion) &&
807       Number.isInteger(record.protocolVersion) &&
808       typeof record.symnavVersion === "string" &&
809       typeof record.workspaceRoot === "string" &&
810       typeof record.workspaceKey === "string" &&
811       typeof record.stateKey === "string" &&
812       typeof record.identityKey === "string" &&
813       typeof record.instanceId === "string" &&
814       typeof record.processToken === "string" &&
815       typeof record.endpoint === "string" &&
816       Number.isInteger(record.pid) &&
817       (record.state === "starting" || record.state === "ready") &&
818       typeof record.startedAt === "number" &&
819       typeof record.memoryCapBytes === "number" &&
820       (record.readyAt === undefined || typeof record.readyAt === "number") &&
821       (record.fileCount === undefined || typeof record.fileCount === "number") &&
822       (record.memoryBytes === undefined || typeof record.memoryBytes === "number") &&
823       (record.lastNavigationAt === undefined || typeof record.lastNavigationAt === "number")
824     );
825   }

843     identity: DaemonWorkspaceIdentity,
844     expectation: StartupOwnershipExpectation,
845   ): StartupOwner | undefined {
846     const owner = this.startupOwner(identity);
847     if (
848       owner?.identityKey !== expectation.identityKey ||
849       owner.instanceId !== expectation.instanceId ||
850       (expectation.processToken !== undefined && owner.processToken !== expectation.processToken) ||
851       (expectation.ownerKind !== undefined && owner.ownerKind !== expectation.ownerKind) ||
852       (expectation.ownerPid !== undefined && owner.ownerPid !== expectation.ownerPid) ||
853       (expectation.acquiredAt !== undefined && owner.acquiredAt !== expectation.acquiredAt) ||
854       (expectation.heartbeatAt !== undefined && owner.heartbeatAt !== expectation.heartbeatAt) ||
855       (expectation.revision !== undefined && owner.revision !== expectation.revision)
856     ) {
857       return undefined;
858     }
859     return owner;
860   }

915   private static processIsAlive(pid: number): boolean {
916     if (!Number.isInteger(pid) || pid <= 0) return false;
917     try {
918       process.kill(pid, 0);
919       return true;
920     } catch (error) {
921       return DaemonRegistry.errorCode(error) === "EPERM";
922     }
923   }

925   private replaceStartupOwner(temporaryPath: string, ownerPath: string): void {
926     if (this.platform !== "win32") {
927       this.renamePath(temporaryPath, ownerPath);
928       return;
929     }
930     const previousPath = `${ownerPath}.${process.pid}.${randomUUID()}.previous`;
931     this.renamePath(ownerPath, previousPath);
932     try {
933       this.renamePath(temporaryPath, ownerPath);
934     } catch (error) {
935       try {
936         this.renamePath(previousPath, ownerPath);
937       } catch {}
938       throw error;
939     }
940     try {
941       rmSync(previousPath, { force: true });
942     } catch {}
943   }

945   private replaceStartupOwnerIfOwner(
946     identity: DaemonWorkspaceIdentity,
947     observedOwner: StartupOwner,
948     replacementOwner: StartupOwner,
949   ): StartupOwner | undefined {
950     const mutation = this.beginStartupMutation(identity);
951     if (mutation === undefined) return undefined;
952     const ownerPath = identity.startupOwnerPath(identity.lockPath);
953     const temporaryPath = `${identity.lockPath}.${process.pid}.${randomUUID()}.owner.tmp`;
954     try {
955       writeFileSync(temporaryPath, JSON.stringify(replacementOwner), {
956         encoding: "utf8",
957         mode: 0o600,
958       });
959       if (
960         !mutation.isOwned() ||
961         this.startupOwnershipMatches(identity, observedOwner) === undefined
962       ) {
963         rmSync(temporaryPath, { force: true });
964         return undefined;
965       }
966       try {
967         this.replaceStartupOwner(temporaryPath, ownerPath);
968       } catch (error) {
969         rmSync(temporaryPath, { force: true });
970         if (DaemonRegistry.errorCode(error) === "ENOENT") return undefined;
971         throw error;
972       }
973       return this.startupOwnershipMatches(identity, replacementOwner) !== undefined
974         ? replacementOwner
975         : undefined;
976     } finally {
977       mutation.release();
978     }
979   }
