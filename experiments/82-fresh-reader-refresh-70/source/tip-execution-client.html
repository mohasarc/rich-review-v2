47   private async completeWithReattachments(
48     endpoint: string,
49     request: DaemonExecuteRequest,
50     completion: DaemonExecutionReceipt["completion"],
51   ): DaemonExecutionReceipt["completion"] {
52     let currentCompletion = completion;
53     let originalAcceptedClose: DaemonTransportError | undefined;
54     let reattachmentCount = 0;
55     while (true) {
56       try {
57         return await currentCompletion;
58       } catch (error) {
59         if (!DaemonExecutionClient.isAcceptedConnectionClose(error, request)) throw error;
60         originalAcceptedClose ??= error;
61         if (
62           reattachmentCount >= this.options.deliveryPolicy.postAcceptanceExecutionReattachmentLimit
63         )
64           throw originalAcceptedClose;
65         try {
66           const reattached = await this.executeOnce(endpoint, request);
67           currentCompletion = reattached.completion;
68           reattachmentCount += 1;
69         } catch {
70           throw originalAcceptedClose;
71         }
72       }
73     }
74   }

76   private executeOnce(
77     endpoint: string,
78     request: DaemonExecuteRequest,
79   ): Promise<DaemonExecutionReceipt> {
80     this.options.validator.request(request);
81     return new Promise((resolve, reject) => {
82       const decoder = this.options.codec.transferDecoder();
83       const transfer = new DaemonResultTransferReceiver(
84         request.requestId,
85         this.options.createOutput(),
86       );
87       let connection: Awaited<ReturnType<DaemonSocketClient["connect"]>> | undefined;
88       let delivery: DaemonDeliveryState = "not-submitted";
89       let acceptance: DaemonExecutionAcceptance | undefined;
90       let terminal = false;
91       let outerSettled = false;
92       let completionSettled = false;
93       let resumeCount = 0;
94       let resolveCompletion!: (value: Awaited<DaemonExecutionReceipt["completion"]>) => void;
95       let rejectCompletion!: (error: DaemonTransportError) => void;
96       const completion = new Promise<Awaited<DaemonExecutionReceipt["completion"]>>(
97         (completionResolve, completionReject) => {
98           resolveCompletion = completionResolve;
99           rejectCompletion = completionReject;
100         },
101       );
102       const fail = (error: unknown): void => {
103         const transportError = DaemonExecutionClient.transportError(error, delivery);
104         connection?.destroy();
105         if (!outerSettled) {
106           outerSettled = true;
107           void transfer.dispose().finally(() => reject(transportError));
108           return;
109         }
110         if (!completionSettled) {
111           completionSettled = true;
112           void transfer.dispose().finally(() => rejectCompletion(transportError));
113         }
114       };
115       const resume = (): boolean => {
116         if (
117           resumeCount >= this.options.deliveryPolicy.resultTransferResumeLimitPerExecutionAttempt ||
118           completionSettled ||
119           acceptance === undefined ||
120           transfer.manifest === undefined ||
121           terminal
122         ) {
123           return false;
124         }
125         resumeCount += 1;
126         connection?.destroy();
127         transfer.beginConnection();
128         void this.fetchCompletion(endpoint, request, transfer)
129           .then((completionValue) => {
130             if (completionSettled) return;
131             completionSettled = true;
132             resolveCompletion(completionValue);
133           })
134           .catch((error: unknown) => {
135             if (DaemonExecutionClient.isAcceptedConnectionClose(error, request) && resume()) {
136               return;
137             }
138             fail(
139               error instanceof DaemonResultFetchEndedError
140                 ? new DaemonTransportError("corrupt", "accepted", error.message)
141                 : error,
142             );
143           });
144         return true;
145       };
146       const publishAcceptance = (): void => {
147         if (acceptance === undefined || outerSettled) return;
148         outerSettled = true;
149         connection?.disableTimeout();
150         resolve({ acceptance, completion });
151       };
152       const consume = async (bytes: Uint8Array): Promise<void> => {
153         const values = decoder.append(bytes);
154         let completedResult = false;
155         let failedCode: DaemonExecutionFailureCode | undefined;
156         for (const value of values) {
157           if (DaemonExecutionClient.isResultChunk(value)) {
158             if (acceptance === undefined)
159               throw new Error("Daemon returned output before acceptance");
160             await transfer.acceptChunk(value);
161             continue;
162           }
163           const frame = this.options.validator.executionFrame(request, value);
164           if (frame.kind === "rejected") {
165             if (acceptance !== undefined || terminal) {
166               throw new Error("Daemon rejected an already accepted request");
167             }
168             throw new DaemonTransportError(
169               "rejected",
170               "submitted-unconfirmed",
171               `Daemon rejected execution: ${frame.code}`,
172               frame.instanceId,
173               frame.code,
174             );
175           }
176           if (frame.kind === "accepted") {
177             if (acceptance !== undefined || terminal) {
178               throw new Error("Daemon returned duplicate acceptance");
179             }
180             delivery = "accepted";
181             acceptance = {
182               requestId: frame.requestId,
183               instanceId: frame.instanceId,
184               acceptedAt: frame.acceptedAt,
185               queuePosition: frame.queuePosition,
186             };
187             publishAcceptance();
188             continue;
189           }
190           if (acceptance === undefined) {
191             throw new Error("Daemon returned completion before acceptance");
192           }
193           if (frame.kind === "result-manifest") {
194             if (terminal) throw new Error("Daemon returned a manifest after completion");
195             transfer.acceptManifest(frame);
196             continue;
197           }
198           if (frame.kind === "result-end") {
199             transfer.acceptEnd(frame);
200             terminal = true;
201             completedResult = true;
202             continue;
203           }
204           if (terminal) throw new Error("Daemon returned duplicate terminal frame");
205           terminal = true;
206           failedCode = frame.code;
207         }
208         if (completedResult) {
209           const result = await transfer.finish();
210           const completedManifest = transfer.manifest;
211           if (completedManifest === undefined) throw new Error("Completion manifest is missing");
212           try {
213             await this.options.lifecycle.acknowledgeResult(
214               endpoint,
215               request,
216               completedManifest.transferId,
217             );
218           } catch (error) {
219             await result.output.dispose();
220             throw error;
221           }
222           if (!completionSettled) {
223             completionSettled = true;
224             connection?.end();
225             resolveCompletion({ status: "completed", result });
226           }
227         }
228         if (failedCode !== undefined && !completionSettled) {
229           completionSettled = true;
230           connection?.end();
231           await transfer.dispose();
232           resolveCompletion({ status: "failed", code: failedCode });
233         }
234       };
235       const receive = async (): Promise<void> => {
236         try {
237           connection = await this.options.sockets.connect(
238             endpoint,
239             this.options.transportPolicy.executionAdmissionTimeoutMs,
240           );
241         } catch (error) {
242           if (DaemonExecutionClient.isSocketTimeout(error)) {
243             fail(new DaemonTransportError("timeout", delivery, "Daemon request timed out"));
244             return;
245           }
246           const message = error instanceof Error ? error.message : String(error);
247           fail(new DaemonTransportError("unreachable", delivery, message));
248           return;
249         }
250         try {
251           connection.write(this.options.codec.encodeControl(request));
252           delivery = "submitted-unconfirmed";
253         } catch (error) {
254           fail(error);
255           return;
256         }
257         try {
258           for await (const bytes of connection.incoming) {
259             try {
260               await consume(bytes);
261             } catch (error) {
262               throw DaemonExecutionClient.transportError(error, delivery);
263             }
264           }
265           if (terminal && completionSettled) return;
266           if (resume()) return;
267           try {
268             decoder.assertComplete();
269           } catch (error) {
270             fail(error);
271             return;
272           }
273           fail(
274             new DaemonTransportError(
275               "closed",
276               delivery,
277               acceptance === undefined
278                 ? "Daemon connection ended before acceptance"
279                 : "Daemon connection ended after acceptance before completion",
280               acceptance?.instanceId,
281             ),
282           );
283         } catch (error) {
284           if (error instanceof DaemonTransportError || error instanceof DaemonProtocolError) {
285             fail(error);
286             return;
287           }
288           if (DaemonExecutionClient.isSocketTimeout(error)) {
289             fail(new DaemonTransportError("timeout", delivery, "Daemon request timed out"));
290             return;
291           }
292           if (resume()) return;
293           const message = error instanceof Error ? error.message : String(error);
294           fail(new DaemonTransportError("closed", delivery, message, acceptance?.instanceId));
295         }
296       };
297       void receive();
298     });
299   }

301   private async fetchCompletion(
302     endpoint: string,
303     request: DaemonExecuteRequest,
304     transfer: DaemonResultTransferReceiver,
305   ): DaemonExecutionReceipt["completion"] {
306     const decoder = this.options.codec.transferDecoder();
307     let connection: Awaited<ReturnType<DaemonSocketClient["connect"]>>;
308     try {
309       connection = await this.options.sockets.connect(endpoint);
310     } catch (error) {
311       const message = error instanceof Error ? error.message : String(error);
312       throw new DaemonTransportError("closed", "accepted", message, request.instanceId);
313     }
314     try {
315       connection.write(
316         this.options.codec.encodeControl({
317           kind: "result-fetch",
318           protocolVersion: request.protocolVersion,
319           instanceId: request.instanceId,
320           processToken: request.processToken,
321           requestId: request.requestId,
322           offset: transfer.nextOffset,
323         }),
324       );
325     } catch (error) {
326       connection.destroy();
327       throw DaemonExecutionClient.transportError(error, "accepted");
328     }
329     let ended = false;
330     try {
331       for await (const bytes of connection.incoming) {
332         try {
333           let completedResult = false;
334           let failedCode: DaemonExecutionFailureCode | undefined;
335           for (const value of decoder.append(bytes)) {
336             if (ended) throw new Error("Daemon resumed with a duplicate terminal frame");
337             if (DaemonExecutionClient.isResultChunk(value)) {
338               await transfer.acceptChunk(value);
339               continue;
340             }
341             const frame = this.options.validator.executionFrame(request, value);
342             if (frame.kind === "result-manifest") {
343               transfer.acceptManifest(frame);
344               continue;
345             }
346             if (frame.kind === "execution-failed") {
347               ended = true;
348               failedCode = frame.code;
349               continue;
350             }
351             if (frame.kind !== "result-end") {
352               throw new Error("Daemon resumed with an invalid terminal frame");
353             }
354             transfer.acceptEnd(frame);
355             ended = true;
356             completedResult = true;
357           }
358           if (completedResult) {
359             const result = await transfer.finish();
360             const manifest = transfer.manifest;
361             if (manifest === undefined) throw new Error("Completion manifest is missing");
362             try {
363               await this.options.lifecycle.acknowledgeResult(
364                 endpoint,
365                 request,
366                 manifest.transferId,
367               );
368             } catch (error) {
369               await result.output.dispose();
370               throw error;
371             }
372             connection.end();
373             return { status: "completed", result };
374           }
375           if (failedCode !== undefined) {
376             connection.end();
377             await transfer.dispose();
378             return { status: "failed", code: failedCode };
379           }
380         } catch (error) {
381           throw DaemonExecutionClient.transportError(error, "accepted");
382         }
383       }
384       try {
385         decoder.assertComplete();
386       } catch (error) {
387         throw DaemonExecutionClient.transportError(error, "accepted");
388       }
389       throw new DaemonResultFetchEndedError(request.instanceId);
390     } catch (error) {
391       connection.destroy();
392       if (error instanceof DaemonTransportError || error instanceof DaemonProtocolError) {
393         throw DaemonExecutionClient.transportError(error, "accepted");
394       }
395       const message = error instanceof Error ? error.message : String(error);
396       throw new DaemonTransportError("closed", "accepted", message, request.instanceId);
397     }
398   }

400   private static isAcceptedConnectionClose(
401     error: unknown,
402     request: DaemonExecuteRequest,
403   ): error is DaemonTransportError {
404     return (
405       error instanceof DaemonTransportError &&
406       error.code === "closed" &&
407       error.delivery === "accepted" &&
408       error.authenticatedInstanceId === request.instanceId
409     );
410   }

412   private static transportError(
413     error: unknown,
414     delivery: DaemonDeliveryState,
415   ): DaemonTransportError {
416     if (error instanceof DaemonTransportError) return error;
417     if (error instanceof DaemonProtocolError) {
418       return new DaemonTransportError(
419         error.code,
420         "accepted",
421         error.message,
422         error.authenticatedInstanceId,
423       );
424     }
425     const message = error instanceof Error ? error.message : String(error);
426     return new DaemonTransportError("corrupt", delivery, message);
427   }
