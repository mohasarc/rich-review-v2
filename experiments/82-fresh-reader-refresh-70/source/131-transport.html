101 class DaemonFrameDecoder {
102   private buffered = Buffer.alloc(0);
103 
104   constructor(private readonly maximumFrameBytes: number) {}
105 
106   append(bytes: Buffer): readonly unknown[] {
107     this.buffered = Buffer.concat([this.buffered, bytes]);
108     const values: unknown[] = [];
109     while (this.buffered.length >= 4) {
110       const payloadLength = this.buffered.readUInt32BE(0);
111       if (payloadLength > this.maximumFrameBytes) {
112         throw new Error(`Daemon frame exceeds ${this.maximumFrameBytes} bytes`);
113       }
114       if (this.buffered.length < payloadLength + 4) break;
115       const payload = this.buffered.subarray(4, payloadLength + 4);
116       this.buffered = this.buffered.subarray(payloadLength + 4);
117       try {
118         values.push(JSON.parse(payload.toString("utf8")));
119       } catch {
120         throw new Error("Daemon frame contains malformed JSON");
121       }
122     }
123     return values;
124   }
125 
126   assertComplete(): void {
127     if (this.buffered.length !== 0) {
128       throw new Error("Daemon connection ended with a truncated frame");
129     }
130   }

209       frame.transferId !== manifest.transferId ||
210       frame.rawBytes !== manifest.rawBytes ||
211       frame.recordCount !== manifest.recordCount ||
212       frame.sha256 !== manifest.sha256 ||
213       this.nextRecordOffset !== manifest.recordCount

754       protocolVersion: request.protocolVersion,
755       instanceId: request.instanceId,
756       processToken: request.processToken,
757       requestId: request.requestId,
758       transferId: manifest.transferId,
759     };
760     await new Promise<void>((resolve, reject) => {
761       const decoder = new DaemonFrameDecoder(this.maximumFrameBytes);
762       const socket = createConnection(endpoint);
763       let responseReceived = false;
764       let settled = false;
765       const fail = (error: unknown): void => {
766         if (settled) return;
767         settled = true;
768         socket.destroy();
769         reject(error);
770       };
771       const succeed = (): void => {
772         if (settled) return;
773         settled = true;
774         socket.destroy();
775         resolve();
776       };
777       const complete = (): void => {
778         try {
779           decoder.assertComplete();
780           if (!responseReceived) throw new Error("Daemon acknowledgement response is missing");
781           succeed();
782         } catch (error) {
783           fail(error);
784         }
785       };
786       socket.setTimeout(this.requestTimeoutMs, () =>
787         fail(new Error("Daemon result acknowledgement timed out")),
788       );
789       socket.once("error", fail);
790       socket.once("connect", () => {
791         try {
792           this.writeFrame(socket, acknowledgement);
793         } catch (error) {
794           fail(error);
795         }
796       });
797       socket.on("data", (bytes) => {
798         try {
799           const values = decoder.append(Buffer.isBuffer(bytes) ? bytes : Buffer.from(bytes));
800           for (const response of values) {
801             if (responseReceived) throw new Error("Duplicate daemon result acknowledgement");
802             LocalDaemonTransport.assertResponse(response);
803             if (
804               response.kind !== "result-acknowledged" ||
805               response.instanceId !== request.instanceId ||
806               response.processToken !== request.processToken ||
807               response.requestId !== request.requestId ||
808               response.transferId !== manifest.transferId
809             ) {
810               throw new Error("Invalid daemon result acknowledgement");
811             }
812             responseReceived = true;
813           }
814           if (responseReceived) socket.end();
815         } catch (error) {
816           fail(error);
817         }
818       });
819       socket.once("end", complete);
820       socket.once("close", () => {
821         if (!settled) complete();
822       });
823     });
824   }

885   private serve(
886     socket: Socket,
887     handler: (request: DaemonRequest, send: DaemonServerSend) => Promise<DaemonResponse | void>,
888   ): void {
889     const decoder = new DaemonFrameDecoder(this.maximumFrameBytes);
890     let responses = Promise.resolve();
891     let writes = Promise.resolve();
892     const closeListeners = new Set<() => void>();
893     const send: DaemonServerSend = Object.assign(
894       (message: DaemonServerMessage) => {
895         const write = writes.then(() => this.writeServerMessage(socket, message));
896         writes = write;
897         return write;
898       },
899       {
900         onClose: (listener: () => void): (() => void) => {
901           closeListeners.add(listener);
902           return () => closeListeners.delete(listener);
903         },
904       },
905     );
906     socket.on("data", (bytes) => {
907       try {
908         for (const value of decoder.append(Buffer.isBuffer(bytes) ? bytes : Buffer.from(bytes))) {
909           LocalDaemonTransport.assertRequest(value);
910           responses = responses
911             .then(async () => {
912               const response = await handler(value, send);
913               if (response !== undefined) await send(response);
914             })
915             .catch(() => {
916               socket.destroy();
917             });
918         }
919       } catch {
920         socket.destroy();
921       }
922     });
923     socket.once("end", () => {
924       try {
925         decoder.assertComplete();
926       } catch {
927         socket.destroy();
928       }
929     });
930     socket.once("close", () => {
931       for (const listener of closeListeners) {
932         try {
933           listener();
934         } catch {}
935       }
936       closeListeners.clear();
937     });
938     socket.once("error", () => socket.destroy());
939   }

945   private async writeServerMessage(socket: Socket, message: DaemonServerMessage): Promise<void> {
946     if ("kind" in message) {

948       return;

956   private async writeEncodedServerFrame(socket: Socket, frame: Buffer): Promise<void> {
957     const chunkSize = this.writeChunkSize ?? frame.length;
958     for (let offset = 0; offset < frame.length; offset += chunkSize) {
959       if (socket.destroyed) throw new Error("Daemon socket closed during response delivery");
960       const accepted = socket.write(frame.subarray(offset, offset + chunkSize));
961       if (!accepted) await LocalDaemonTransport.waitForDrain(socket);
962     }
963   }

965   private static waitForDrain(socket: Socket): Promise<void> {
966     return new Promise((resolve, reject) => {
967       const cleanup = (): void => {
968         socket.off("drain", drained);
969         socket.off("error", failed);
970         socket.off("close", closed);
971       };
972       const drained = (): void => {
973         cleanup();
974         resolve();
975       };
976       const failed = (error: Error): void => {
977         cleanup();
978         reject(error);
979       };
980       const closed = (): void => {
981         cleanup();
982         reject(new Error("Daemon socket closed during response delivery"));
983       };
984       socket.once("drain", drained);
985       socket.once("error", failed);
986       socket.once("close", closed);
987     });
988   }

990   private encodeFrame(value: unknown): Buffer {
991     const payload = Buffer.from(JSON.stringify(value), "utf8");
992     if (payload.length > this.maximumFrameBytes) {
993       throw new Error(`Daemon frame exceeds ${this.maximumFrameBytes} bytes`);
994     }
995     const prefix = Buffer.alloc(4);
996     prefix.writeUInt32BE(payload.length);
997     return Buffer.concat([prefix, payload]);
998   }

1000   private writeEncodedFrame(socket: Pick<Socket, "write">, frame: Buffer): void {
1001     if (this.writeChunkSize === undefined) {
1002       socket.write(frame);
1003       return;
1004     }
1005     for (let offset = 0; offset < frame.length; offset += this.writeChunkSize) {
1006       socket.write(frame.subarray(offset, offset + this.writeChunkSize));
1007     }
1008   }

1012     value: unknown,

1027     }

1029       const expectedKind = request.kind === "terminate" ? "terminating" : "killing";
1030       if (value.kind !== expectedKind) {

1039       }

1075       throw new DaemonResponseError(
1076         "authentication",
1077         "Daemon stop response does not match instance",
1078       );
1079     }
1080     return value;
1081   }

1427     }
1428     if (lifecycle === "busy") expectedKeys.push("current");
1429     if (lifecycle === "recovering") expectedKeys.push("recoveryDetail");
1430     return (
1431       (lifecycle === "starting" ||
1432         lifecycle === "ready" ||
1433         lifecycle === "busy" ||
1434         lifecycle === "recovering" ||
1435         lifecycle === "draining") &&
1436       LocalDaemonTransport.hasExactKeys(value, expectedKeys) &&
1437       (lifecycle !== "recovering" ||
1438         value.recoveryDetail === "resource-pressure" ||
1439         value.recoveryDetail === "worker-replacement") &&
1440       Number.isInteger(value.pid) &&
1441       (value.pid as number) > 0 &&
1442       LocalDaemonTransport.isMetric(value.startedAt) &&
1443       LocalDaemonTransport.isMetric(value.startupElapsedMs) &&
1444       (lifecycle === "ready" || lifecycle === "busy"
1445         ? LocalDaemonTransport.isCount(value.fileCount)
1446         : lifecycle === "starting"
1447           ? value.fileCount === undefined
1448           : value.fileCount === undefined || LocalDaemonTransport.isCount(value.fileCount)) &&
1449       LocalDaemonTransport.isCount(value.processRssBytes) &&
1450       LocalDaemonTransport.isCount(value.hardProcessRssBytes) &&
1451       (value.workerHeapUsedBytes === undefined ||
1452         LocalDaemonTransport.isCount(value.workerHeapUsedBytes)) &&
1453       LocalDaemonTransport.isCount(value.workerGeneration) &&
1454       (lifecycle !== "busy" ||
1455         (LocalDaemonTransport.isRecord(current) &&
1456           LocalDaemonTransport.hasExactKeys(current, ["requestId", "command", "elapsedMs"]) &&
1457           DaemonRuntimeValues.isRequestId(current.requestId) &&
1458           DaemonRuntimeValues.isCommandName(current.command) &&
1459           LocalDaemonTransport.isMetric(current.elapsedMs))) &&
1460       LocalDaemonTransport.isCount(value.queued) &&
1461       (value.lastCompletedAgoMs === undefined ||
1462         LocalDaemonTransport.isMetric(value.lastCompletedAgoMs)) &&
1463       LocalDaemonTransport.isCount(value.spoolBytes)
1464     );
1465   }

1467   private static isCount(value: unknown): boolean {
1468     return Number.isInteger(value) && (value as number) >= 0;
1469   }

1471   private static isMetric(value: unknown): boolean {
1472     return typeof value === "number" && Number.isFinite(value) && value >= 0;
1473   }

1475   private static isExecutionRequest(value: unknown): boolean {
1476     if (!LocalDaemonTransport.isRecord(value)) return false;
1477     const expectedKeys = ["argv", "cwd", "telemetryEnabled"];
1478     if (value.executionMode !== undefined) expectedKeys.push("executionMode");
1479     return (
1480       LocalDaemonTransport.hasExactKeys(value, expectedKeys) &&
1481       Array.isArray(value.argv) &&
1482       value.argv.every((arg) => typeof arg === "string") &&
1483       typeof value.cwd === "string" &&
1484       typeof value.telemetryEnabled === "boolean" &&
1485       (value.executionMode === undefined ||
1486         value.executionMode === "cold" ||
1487         value.executionMode === "warm" ||
1488         value.executionMode === "fallback")
1489     );
1490   }

1492   private static isRecord(value: unknown): value is Record<string, unknown> {
1493     return typeof value === "object" && value !== null && !Array.isArray(value);
1494   }

1496   private static isResultChunk(value: unknown): value is DaemonResultChunk {
1497     return (
1498       LocalDaemonTransport.isRecord(value) &&
1499       !("kind" in value) &&
1500       value.bytes instanceof Uint8Array
1501     );
1502   }

1504   private static hasExactKeys(value: Record<string, unknown>, keys: readonly string[]): boolean {
1505     const actual = Object.keys(value).sort();
1506     const expected = [...keys].sort();
1507     return (
1508       actual.length === expected.length && actual.every((key, index) => key === expected[index])
1509     );
1510   }
