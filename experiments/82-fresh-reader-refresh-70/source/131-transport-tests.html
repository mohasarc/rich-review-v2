997 async function sendRecords(
998   socket: Socket,
999   spool: import("./completion-spool.js").CompletionSpool,
1000   transferId: string,
1001   offset: number,
1002   stopBefore = Number.POSITIVE_INFINITY,
1003 ): Promise<void> {
1004   for await (const record of spool.read(offset)) {
1005     if (record.sequence >= stopBefore || socket.destroyed || !socket.writable) return;
1006     socket.write(
1007       DaemonResultChunkCodec.encode(
1008         {
1009           transferId,
1010           requestId: request.requestId,
1011           offset: record.sequence,
1012           sequence: record.sequence,
1013           stream: record.stream,
1014           bytes: record.bytes,
1015         },
1016         TEST_CHUNK_BYTES,
1017       ),
1018     );
1019   }
1020 }

1066 async function rawExecutionServer(
1067   servers: Server[],
1068   sockets: Socket[],
1069   directories: string[],
1070   connected: (socket: Socket) => void,
1071 ): Promise<string> {
1072   const endpoint = executionEndpoint(directories);
1073   const server = createServer((socket) => {
1074     sockets.push(socket);
1075     socket.on("error", (error: NodeJS.ErrnoException) => {
1076       if (error.code === "EPIPE" || error.code === "ECONNRESET") return;
1077       throw error;
1078     });
1079     connected(socket);
1080   });
1081   servers.push(server);
1082   await new Promise<void>((resolve, reject) => {
1083     server.once("error", reject);
1084     server.listen(endpoint, resolve);
1085   });
1086   return endpoint;
1087 }

1094   directories.push(directory);
1095   return join(directory, "daemon.sock");

1098 function frame(value: unknown): Buffer {
1099   const payload = Buffer.from(JSON.stringify(value), "utf8");
1100   const prefix = Buffer.alloc(4);
1101   prefix.writeUInt32BE(payload.length);
1102   return Buffer.concat([prefix, payload]);
1103 }

1105 async function settleWithin<T>(operation: Promise<T>, milliseconds: number): Promise<T> {
1106   let timeout: NodeJS.Timeout | undefined;
1107   const expired = new Promise<never>((_resolve, reject) => {
1108     timeout = setTimeout(() => reject(new Error("Operation did not settle")), milliseconds);
1109   });
1110   try {
1111     return await Promise.race([operation, expired]);
1112   } finally {
1113     if (timeout !== undefined) clearTimeout(timeout);
1114   }
