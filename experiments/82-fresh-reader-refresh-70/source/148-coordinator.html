287   private async waitForStartupAuthorization(): Promise<{
288     readonly lease: DaemonStartupLease;
289     readonly record: DaemonRecord;
290   }> {
291     const deadline = this.clock.wallNowMs() + this.policy.values.startup.coordinationGraceMs;
292     while (this.clock.wallNowMs() <= deadline) {
293       const record = this.options.registry.readInstance(
294         this.options.identity,
295         this.options.coordinates.instanceId,
296       );
297       if (
298         record?.state === "starting" &&
299         (record.pid === 0 || record.pid === process.pid) &&
300         record.processToken === this.options.coordinates.processToken
301       ) {
302         const lease = this.options.registry.claimStartupForDaemon(
303           this.options.identity,
304           this.options.coordinates.instanceId,
305           this.options.coordinates.processToken,
306           process.pid,
307         );
308         if (lease === undefined) {
309           await this.pause();
310           continue;
311         }
312         this.options.registry.writeStartingIfStartupOwner(this.options.identity, {
313           ...record,
314           pid: process.pid,
315         });
316         const adoptedRecord = this.options.registry.readInstance(
317           this.options.identity,
318           this.options.coordinates.instanceId,
319         );
320         if (adoptedRecord?.pid !== process.pid) {
321           lease.release();
322           await this.pause();
323           continue;
324         }
325         if (
326           this.options.registry.startupOwnerMatchesProcess(this.options.identity, adoptedRecord)
327         ) {
328           return { lease, record: adoptedRecord };
329         }
330         lease.release();
331       }
332       await this.pause();
333     }
334     throw new Error("Daemon process did not receive startup authorization");
335   }

353   private async handle(
354     request: DaemonRequest,
355     send: DaemonServerSend,
356   ): Promise<DaemonResponse | void> {
357     if (request.kind === "identify") return this.identify(request);
358     if (request.kind === "terminate" || request.kind === "kill") {
359       return this.terminate(request);
360     }
361     if (
362       request.protocolVersion !== DAEMON_PROTOCOL_VERSION ||
363       request.instanceId !== this.options.coordinates.instanceId
364     ) {
365       throw new Error("Daemon request does not match protocol or instance");
366     }
367     if (request.kind === "ping") return this.pong();
368     if (request.kind === "execute") return this.acceptExecution(request, send);
369     if (
370       request.kind === "execution-status" ||
371       request.kind === "result-fetch" ||
372       request.kind === "result-ack"
373     ) {
374       if (request.processToken !== this.options.coordinates.processToken) {
375         throw new Error("Daemon execution request does not match process instance");
376       }
377     }
378     if (request.kind === "result-fetch") {
379       await this.deliverySession.fetch(request, send);
380       return;
381     }
382     if (request.kind === "result-ack") {
383       return this.deliverySession.acknowledge(request);
384     }
385     if (request.kind === "execution-status") {
386       return {
387         kind: "execution-status",
388         instanceId: this.options.coordinates.instanceId,
389         processToken: this.options.coordinates.processToken,
390         requestId: request.requestId,
391         status: this.acceptedExecutionSession.status(request.requestId),
392       };
393     }
394     this.beginGracefulShutdown();
395     await this.acceptedExecutionSession.drain();
396     await this.deliverySession.waitForCompletionAcknowledgements();
397     setTimeout(() => void this.shutdown("graceful"), 0);
398     return { kind: "stopped", instanceId: this.options.coordinates.instanceId };
399   }

471     request: Extract<DaemonRequest, { kind: "execute" }>,
472     send: DaemonServerSend,
473   ): Promise<DaemonResponse | void> {
474     const decision = this.decideAdmission(request);
475     if (decision.kind === "disconnect") {

477     }
478     if (decision.kind === "reject") return this.rejection(request, decision.code);
479     const admission = this.acceptedExecutionSession.accept(request);
480     await this.deliverySession.attach(
481       {
482         ...admission.acceptance,
483       },
484       send,
485     );

488   private decideAdmission(
489     request: Extract<DaemonRequest, { kind: "execute" }>,
490   ): DaemonAdmissionDecision {
491     const authenticated = request.processToken === this.options.coordinates.processToken;
492     if (!authenticated) {
493       return this.admissionPolicy.decide({
494         request,
495         authenticated,
496         workerReady: true,
497         resourceAdmissionPaused: false,
498         queueState: "accepting",
499         compatibility: "unseen",
500       });
501     }
502     void this.resourceSupervisor.sample("admission").catch((error) => {
503       this.logger.record({
504         kind: "failure",
505         operation: "resource-sample",
506         failureCode: "operation-failed",
507         errorName: DaemonLogger.errorName(error),
508       });
509     });
510     return this.admissionPolicy.decide({
511       request,
512       authenticated,
513       workerReady: this.workerManager.snapshot.ready,
514       resourceAdmissionPaused: this.resourceSupervisor.snapshot.admissionPaused,
515       queueState: this.acceptedExecutionSession.snapshot.queue.state,
516       compatibility: this.acceptedExecutionSession.compatibilityFor(request),
517     });
518   }

589   private async shutdown(
590     reason: "graceful" | "idle" | "resource" | "workspace-deleted",
591     force = false,
592   ): Promise<void> {
593     this.shutdownFailureCode ??= reason === "resource" ? "controlled-resource" : "stopping";
594     if (force) this.forceWorkerShutdown();
595     if (this.shutdownOperation !== undefined) return this.shutdownOperation;
596     this.shutdownStarted = true;
597     this.shutdownOperation = this.completeShutdown(reason, force);

601   private async completeShutdown(
602     reason: "graceful" | "idle" | "resource" | "workspace-deleted",
603     force: boolean,
604   ): Promise<void> {
605     this.lifetime.stop();
606     this.resourceSupervisor.stop();
607     if (force) await this.forceWorkerShutdown();
608     else await this.gracefullyShutdownWorker();
609     this.operationObserver.shutdown({ kind: "shutdown", reason, force });
610     this.logger.record({ kind: "stop", reason });
611     try {
612       await this.server?.close();
613     } catch (error) {
614       this.logger.record({
615         kind: "failure",
616         operation: "transport-close",
617         failureCode: "operation-failed",
618         errorName: DaemonLogger.errorName(error),
619       });
620     }
621     await this.deliverySession.cleanupInstance();
622     this.deliverySession.completeRetainedTraces();
623     await this.logger.close();
624     this.exit(0);
625   }

651   ): Promise<void> {
652     if (this.shutdownStarted) return;
653     this.logger.record({
654       kind: "failure",
655       operation: "worker-exit",
656       failureCode: "worker-exit",
657       errorName: DaemonLogger.errorName(
658         workerExit.errorName === undefined ? undefined : { name: workerExit.errorName },
659       ),
660     });
661     try {
662       await this.resourceSupervisor.recover(workerExit);
663     } catch (error) {
664       this.logger.record({
665         kind: "failure",
666         operation: "worker-replacement",
667         failureCode: "controlled-resource",
668         errorName: DaemonLogger.errorName(error),
669       });
670       throw error;
671     }
672   }
673 }
