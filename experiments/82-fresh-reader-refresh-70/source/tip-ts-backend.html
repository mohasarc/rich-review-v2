32   static accepts(filePath: string): boolean {
33     const name = basename(filePath);
34     for (const ext of TypeScriptBackend.extensions) {

36         return true;

38     }

79   async refresh(request: BackendRefreshRequest): Promise<BackendRefreshSummary> {
80     this.sourceCache?.refresh(request.snapshot);
