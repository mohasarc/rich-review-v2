91   private static diagnosticPaths(identity: DaemonWorkspaceIdentity): readonly string[] {
92     let names: readonly string[];
93     try {
94       names = readdirSync(identity.identityDirectory);
95     } catch (error) {
96       if (["ENOENT", "ENOTDIR"].includes(DaemonTestingInspector.errorCode(error) ?? "")) return [];
97       throw error;
98     }
99     const backupPattern = /^daemon\.log\.(\d+)$/;
100     const backups = names
101       .flatMap((name) => {
102         const match = backupPattern.exec(name);
103         return match === null ? [] : [{ name, index: Number(match[1]) }];
104       })
105       .sort((left, right) => right.index - left.index)
106       .map(({ name }) => join(identity.identityDirectory, name));
107     return names.includes("daemon.log") ? [...backups, identity.logPath] : backups;
108   }

110   private static diagnosticEventsFrom(path: string): readonly DaemonTestingDiagnosticEvent[] {
111     let contents: string;
112     try {
113       contents = readFileSync(path, "utf8");
114     } catch (error) {
115       if (DaemonTestingInspector.errorCode(error) === "ENOENT") return [];
116       throw error;
117     }
118     return contents
119       .split("\n")
120       .filter((line) => line.length > 0)
121       .flatMap((line) => {
122         try {
123           const value: unknown = JSON.parse(line);
124           return DaemonDiagnosticValues.isDiagnostics(value) ? [value] : [];
125         } catch (error) {
126           if (error instanceof SyntaxError) return [];
127           throw error;
128         }
129       });
130   }

132   private static directoryUsage(directory: string): DaemonTestingSpoolUsage {
133     let entries;
134     try {
135       entries = readdirSync(directory, { withFileTypes: true });
136     } catch (error) {
137       if (["ENOENT", "ENOTDIR"].includes(DaemonTestingInspector.errorCode(error) ?? "")) {
138         return { fileCount: 0, bytes: 0 };
139       }
140       throw error;
141     }
142     return entries.reduce<DaemonTestingSpoolUsage>(
143       (usage, entry) => {
144         const path = join(directory, entry.name);
145         if (entry.isDirectory()) {
146           const nested = DaemonTestingInspector.directoryUsage(path);
147           return {
148             fileCount: usage.fileCount + nested.fileCount,
149             bytes: usage.bytes + nested.bytes,
150           };
151         }
152         if (!entry.isFile()) return usage;
153         return {
154           fileCount: usage.fileCount + 1,
155           bytes: usage.bytes + lstatSync(path).size,
156         };
157       },
158       { fileCount: 0, bytes: 0 },
159     );
160   }
