17   constructor(private readonly fileSystem: FileSystem) {}

19   read(path: string): string | undefined {
20     let observation = this.observationByPath.get(path);
21     if (!observation) {
22       observation = { path, content: this.readCurrentInput(path) };
23       this.observationByPath.set(path, observation);
24     }
25     return observation.content ?? undefined;
26   }
27 
28   observations(): readonly ProjectInputObservation[] {
29     return [...this.observationByPath.values()];
30   }

32   private readCurrentInput(path: string): string | null {
33     if (!this.fileSystem.existsSync(path) || this.fileSystem.isDirectorySync(path)) return null;
34     return this.fileSystem.readFileSync(path);
35   }

99   protected async refreshProjectGraph(
100     snapshot: WorkspaceSnapshot,
101   ): Promise<ProjectGraphRefreshSummary> {
102     if (this.projectStateUnchanged(snapshot)) return this.currentSummary(0);
103     const inputCollector = new ProjectInputCollector(this.fileSystem);
104     const discovered = await this.discoverConfigurations(snapshot, inputCollector);
105     const candidateConfigurations = discovered.map(({ path, parsed }) => ({
106       path,
107       configuration: parsed.configuration,
108       files: this.filesForConfiguration(parsed.configuration, snapshot),
109     }));
110     const configurations = ProjectGraph.canonicalizeMembership(snapshot, candidateConfigurations);
111     const ownedRelativePaths = new Set(
112       configurations.flatMap((configuration) => configuration.files.map((file) => file.relative)),
113     );
114     const inferredFiles = snapshot.files.filter((file) => !ownedRelativePaths.has(file.relative));
115     const prepared = await this.prepareProjects({
116       snapshot,
117       configurations,
118       inferredFiles,
119       inputCollector,
120     });
121     this.validatePreparedProjectCount(configurations, prepared);
122     const activeInputs = [...discovered.flatMap(({ parsed }) => parsed.inputs), ...prepared.inputs];
123     const observations = inputCollector.observations();
124     ProjectGraph.validateActiveInputs(activeInputs, observations);
125     const inputsByPath = ProjectGraph.collectInputs(activeInputs);
126     const changedInputCount = ProjectGraph.changedInputCount(
127       this.state?.inputsByPath ?? new Map(),
128       inputsByPath,
129     );
130     const ownership = ProjectGraph.buildOwnership(configurations, prepared.configuredProjects);
131     this.state = {
132       root: snapshot.root,
133       filesByRelativePath: new Map(snapshot.files.map((file) => [file.relative, file])),
134       configuredProjects: prepared.configuredProjects,
135       ...ownership,
136       inferredProject: prepared.inferredProject,
137       inferredFileCount: inferredFiles.length,
138       inputsByPath,
139       observations,
140     };
141     return this.currentSummary(changedInputCount);
142   }

144   async releaseTransientResources(): Promise<void> {
145     if (!this.state) return;
146     for (const project of this.state.configuredProjects) {
147       const release = project.releaseTransientResources();

149     }
150     const inferredRelease = this.state.inferredProject.releaseTransientResources();
151     if (inferredRelease) await inferredRelease;

190   private async discoverConfigurations(
191     snapshot: WorkspaceSnapshot,
192     inputCollector: ProjectInputCollector,
193   ): Promise<readonly DiscoveredProjectConfiguration<ConfigurationUnit>[]> {
194     const pending = [...this.initialConfigurationPaths(snapshot.root)];
195     const seen = new Set<string>();
196     const configurations: DiscoveredProjectConfiguration<ConfigurationUnit>[] = [];
197     while (pending.length > 0) {
198       const path = pending.shift() as string;
199       if (seen.has(path)) continue;
200       seen.add(path);
201       const content = inputCollector.read(path);
202       if (content === undefined) continue;
203       const parsed = await this.parseConfiguration({
204         path,
205         content,
206         snapshot,
207         inputCollector,
208       });
209       if (!parsed) continue;
210       configurations.push({ path, parsed });
211       pending.push(...parsed.referencedConfigurationPaths);
212     }
213     return configurations;
214   }

226   private projectStateUnchanged(snapshot: WorkspaceSnapshot): boolean {
227     if (!this.workspaceFilesUnchanged(snapshot)) return false;
228     return this.state!.observations.every(
229       (observation) => this.readCurrentInput(observation.path) === observation.content,
230     );
231   }

233   private workspaceFilesUnchanged(snapshot: WorkspaceSnapshot): boolean {
234     if (!this.state || this.state.root !== snapshot.root) return false;
235     if (this.state.filesByRelativePath.size !== snapshot.files.length) return false;
236     return snapshot.files.every((file) => {
237       const current = this.state?.filesByRelativePath.get(file.relative);
238       return current?.metadata.changeToken === file.metadata.changeToken;
239     });
240   }

247   private static collectInputs(inputs: readonly ProjectInput[]): ReadonlyMap<string, string> {
248     const byPath = new Map<string, string>();
249     for (const input of inputs) byPath.set(input.path, input.content);
250     return byPath;

253   private static validateActiveInputs(
254     inputs: readonly ProjectInput[],
255     observations: readonly ProjectInputObservation[],
256   ): void {
257     const observationByPath = new Map(
258       observations.map((observation) => [observation.path, observation]),
259     );
260     for (const input of inputs) {
261       const observation = observationByPath.get(input.path);
262       if (!observation || observation.content === null) {
263         throw new Error(`Project input ${input.path} was not observed successfully`);
264       }
265       if (observation.content !== input.content) {
266         throw new Error(`Project input ${input.path} does not match observed content`);
267       }
268     }
269   }

271   private static canonicalizeMembership<ConfigurationUnit>(
272     snapshot: WorkspaceSnapshot,
273     configurations: readonly ProjectConfigurationMembership<ConfigurationUnit>[],
274   ): readonly ProjectConfigurationMembership<ConfigurationUnit>[] {
275     const snapshotFileByRelativePath = new Map(snapshot.files.map((file) => [file.relative, file]));
276     return configurations.map((configuration) => ({
277       ...configuration,
278       files: configuration.files.map((file) => {
279         const snapshotFile = snapshotFileByRelativePath.get(file.relative);
280         if (snapshotFile && ProjectGraph.sameWorkspaceFile(file, snapshotFile)) {
281           return snapshotFile;
282         }
283         throw new Error(
284           `Project configuration ${configuration.path} returned non-snapshot file ${file.relative}`,
285         );
286       }),
287     }));
288   }

290   private static sameWorkspaceFile(left: WorkspaceFile, right: WorkspaceFile): boolean {
291     return (
292       left.relative === right.relative &&
293       left.absolute === right.absolute &&
294       left.metadata.size === right.metadata.size &&
295       left.metadata.modifiedAtMs === right.metadata.modifiedAtMs &&
296       left.metadata.changeToken === right.metadata.changeToken &&
297       left.metadata.fileIdentity === right.metadata.fileIdentity
298     );
299   }

301   private validatePreparedProjectCount(
302     configurations: readonly ProjectConfigurationMembership<ConfigurationUnit>[],

305     if (prepared.configuredProjects.length === configurations.length) return;

311   private static changedInputCount(
312     current: ReadonlyMap<string, string>,
313     next: ReadonlyMap<string, string>,
314   ): number {
315     const paths = new Set([...current.keys(), ...next.keys()]);
316     return [...paths].filter((path) => current.get(path) !== next.get(path)).length;
317   }

319   private static buildOwnership<ConfigurationUnit, Project>(
320     configurations: readonly ProjectConfigurationMembership<ConfigurationUnit>[],
321     configuredProjects: readonly Project[],
322   ): {
323     readonly projectsByRelativePath: ReadonlyMap<string, readonly Project[]>;
324     readonly primaryProjectByRelativePath: ReadonlyMap<string, Project>;
325   } {
326     const projectsByRelativePath = new Map<string, Project[]>();
327     const primaryProjectByRelativePath = new Map<string, Project>();
328     for (const [index, configuration] of configurations.entries()) {
329       const project = configuredProjects[index];
330       if (!project) continue;
331       for (const file of configuration.files) {
332         const projects = projectsByRelativePath.get(file.relative) ?? [];
333         projects.push(project);
334         projectsByRelativePath.set(file.relative, projects);
335         primaryProjectByRelativePath.set(file.relative, project);
336       }
337     }
338     return { projectsByRelativePath, primaryProjectByRelativePath };
339   }
