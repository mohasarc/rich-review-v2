58   openWorkspace(startDirectory: string, coverage: BackendRefreshCoverage): Promise<Workspace> {
59     if (coverage === "selection" || this.#catalog === undefined) {
60       return createWorkspace({ startDir: startDirectory, fs: this.#fileSystem });
61     }
62     return this.#catalog.refresh(startDirectory);
63   }
64 
65   async prepareWorkspace(
66     workspace: Workspace,
67     preparation: WorkspacePreparation,
68   ): Promise<PreparedWorkspaceScope> {
69     const router = new BackendRouter(this.#backends);
70     const snapshot =

75     return { workspace, snapshot, router, refresh };
