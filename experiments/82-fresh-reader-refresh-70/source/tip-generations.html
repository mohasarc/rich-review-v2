58 export class DaemonWorkerGenerationManager {
59   private currentGeneration: DaemonWorkerGeneration;
60   private startOperation: Promise<DaemonWorkerReadyReport> | undefined;
61   private replacementOperation: Promise<DaemonWorkerReadyReport> | undefined;
62   private recoveryOperation: Promise<void> | undefined;
63   private closeOperation: Promise<void> | undefined;
64   private terminationOperation: Promise<void> | undefined;
65   private protocolReadyGeneration: DaemonWorkerGeneration | undefined;
66   private readinessActivated = false;
67   private workerReady = false;
68   private fileCount: number | undefined;
69   private stopping = false;
70 
71   constructor(private readonly options: DaemonWorkerGenerationManagerOptions) {
72     const worker = options.initialWorker ?? options.createWorker(1);
73     this.currentGeneration = this.createGeneration(worker);
74   }
75 
76   get snapshot(): DaemonWorkerGenerationSnapshot {
77     return Object.freeze({
78       generation: this.currentGeneration.worker.generation,
79       ready: this.workerReady,
80       ...(this.fileCount === undefined ? {} : { fileCount: this.fileCount }),
81     });
82   }
83 
84   start(): Promise<DaemonWorkerReadyReport> {
85     if (this.startOperation !== undefined) return this.startOperation;
86     const generation = this.currentGeneration;
87     generation.ready = this.startGeneration(generation);
88     this.startOperation = this.waitForReadyGeneration(generation).then((report) => {
89       this.publishProtocolReady(generation, report);
90       return report;
91     });
92     return this.startOperation;
93   }
94 
95   activateReadiness(): void {
96     if (this.protocolReadyGeneration !== this.currentGeneration) {
97       throw new Error("Navigation worker protocol readiness is unavailable");
98     }
99     this.readinessActivated = true;
100     this.workerReady = true;
101   }
102 
103   execute(
104     requestId: string,
105     request: DaemonWorkerExecuteRequest,
106     output: DaemonOutputSink,
107   ): Promise<DaemonWorkerExecutionReport> {
108     const generation = this.currentGeneration;
109     const ready = generation.ready ?? this.start();
110     return ready
111       .then(() =>
112         generation.worker.execute(requestId, request.commandName, request.request, output),
113       )
114       .then((response): DaemonWorkerExecutionReport => {
115         if (
116           response.kind !== "result" ||
117           response.requestId !== requestId ||
118           response.generation !== generation.worker.generation
119         ) {
120           throw new Error("Navigation worker returned an uncorrelated result");
121         }
122         return response;
123       });
124   }
125 
126   replace(cause: DaemonWorkerReplacementCause): Promise<DaemonWorkerReadyReport> {
127     if (this.replacementOperation !== undefined) return this.replacementOperation;
128     const operation = this.replaceGeneration(cause);
129     this.replacementOperation = operation;
130     void operation.then(
131       () => this.clearReplacement(operation),
132       () => this.clearReplacement(operation),
133     );
134     return operation;
135   }
136 
137   async releaseTransientResources(): Promise<DaemonWorkerResourceReport> {
138     const response = await this.currentGeneration.worker.releaseTransientResources();
139     if (response.kind !== "heap") {
140       throw new Error("Navigation worker did not report heap usage");
141     }
142     this.options.onDiagnostic({
143       kind: "resources-released",
144       workerGeneration: response.generation,
145       workerHeapUsedBytes: response.usedHeapBytes,
146       workerHeapLimitBytes: response.heapLimitBytes,
147     });
148     return response;
149   }
150 
151   close(): Promise<void> {
152     if (this.closeOperation !== undefined) return this.closeOperation;
153     this.stopping = true;
154     this.workerReady = false;
155     this.closeOperation = this.currentGeneration.worker.drainAndClose();
156     return this.closeOperation;
157   }
158 
159   terminate(): Promise<void> {
160     if (this.terminationOperation !== undefined) return this.terminationOperation;
161     this.stopping = true;
162     this.workerReady = false;
163     this.terminationOperation = this.currentGeneration.worker.terminate();
164     return this.terminationOperation;
165   }
166 
167   private async replaceGeneration(
168     cause: DaemonWorkerReplacementCause,
169   ): Promise<DaemonWorkerReadyReport> {
170     const previous = this.currentGeneration;
171     if (cause !== "worker-exit") this.options.onActiveResourceInterruption(cause);
172     this.workerReady = false;
173     this.protocolReadyGeneration = undefined;
174     const nextWorker = this.options.createWorker(previous.worker.generation + 1);
175     if (nextWorker.generation !== previous.worker.generation + 1) {
176       throw new Error("Navigation worker factory returned the wrong generation");
177     }
178     const next = this.createGeneration(nextWorker);
179     this.currentGeneration = next;
180     next.ready = this.startGeneration(next);
181     await previous.worker.terminate().catch(() => undefined);
182     const report = await next.ready;
183     this.publishProtocolReady(next, report);
184     this.options.onDiagnostic({
185       kind: "worker-replaced",
186       cause,
187       previousWorkerGeneration: previous.worker.generation,
188       workerGeneration: next.worker.generation,
189       fileCount: report.fileCount,
190       ...report.startupDurations,
191     });
192     return report;
193   }
194 
195   private startGeneration(generation: DaemonWorkerGeneration): Promise<DaemonWorkerReadyReport> {
196     return generation.worker
197       .start(this.options.workspaceRoot)
198       .then((response): DaemonWorkerReadyReport => {
199         if (response.kind !== "ready" || response.generation !== generation.worker.generation) {
200           throw new Error("Navigation worker did not become ready");
201         }
202         return response;
203       });
204   }
205 
206   private async waitForReadyGeneration(
207     generation: DaemonWorkerGeneration,
208   ): Promise<DaemonWorkerReadyReport> {
209     try {
210       if (generation.ready === undefined) throw new Error("Navigation worker is not starting");
211       return await generation.ready;
212     } catch (error) {
213       if (!(error instanceof DaemonNavigationWorkerExitedError)) throw error;
214       await generation.worker.exited;
215       const recovery = this.recoveryOperation;
216       if (recovery === undefined) throw error;
217       await recovery;
218       const replacement = this.currentGeneration;
219       if (replacement === generation || replacement.ready === undefined) throw error;
220       return replacement.ready;
221     }
222   }
223 
224   private createGeneration(worker: DaemonNavigationWorker): DaemonWorkerGeneration {
225     const generation: DaemonWorkerGeneration = { worker, ready: undefined };
226     void worker.exited.then((exit) => this.observeExit(generation, exit));
227     return generation;
228   }
229 
230   private observeExit(generation: DaemonWorkerGeneration, exit: DaemonNavigationWorkerExit): void {
231     if (this.stopping || generation !== this.currentGeneration) return;
232     const recovery = this.options.exitRecovery.recover(exit);
233     this.recoveryOperation = recovery;
234     void recovery.catch(() => undefined);
235   }
236 
237   private publishProtocolReady(
238     generation: DaemonWorkerGeneration,
239     report: DaemonWorkerReadyReport,
240   ): void {
241     if (generation !== this.currentGeneration) return;
242     this.protocolReadyGeneration = generation;
243     this.fileCount = report.fileCount;
244     if (this.readinessActivated) this.workerReady = true;
245   }
246 
247   private clearReplacement(operation: Promise<DaemonWorkerReadyReport>): void {
248     if (this.replacementOperation === operation) this.replacementOperation = undefined;
249   }
