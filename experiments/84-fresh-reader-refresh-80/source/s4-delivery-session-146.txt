123:   async attach(attachment: DaemonDeliveryAttachment, send: DaemonServerSend): Promise<void> {
124:     const entry = this.options.journal.entryFor(attachment.requestId);
125:     if (entry === undefined) throw new Error("Accepted request result is missing");

127:     const disconnectTraceConnection = this.attachOperationTraceConnection(
128:       attachment.requestId,
129:       send,
130:     );

134:         kind: "accepted",
135:         ...this.options.coordinates,
136:         ...attachment,
137:       });
138:       if (entry.state.state === "completed") {
139:         await this.deliverStoredCompletion(attachment.requestId, send);
140:         return;
141:       }
142:       if (entry.state.state === "failed") {
143:         await this.deliver(send, this.failedFrame(attachment.requestId, entry.state.code));

297:   private async deliverCompletion(
298:     requestId: string,
299:     spool: CompletionSpool,
300:     offset: number,
301:     send: DaemonServerSend,
302:   ): Promise<void> {
303:     const completedManifest = spool.completedManifest;
304:     if (completedManifest === undefined) throw new Error("Completion manifest is missing");
305:     await this.deliver(send, {
306:       kind: "result-manifest",
307:       ...this.options.coordinates,

309:       manifest: completedManifest,

312:       await this.deliver(send, {
313:         transferId: completedManifest.transferId,

316:         sequence: record.sequence,
317:         stream: record.stream,
318:         bytes: record.bytes,
319:       });
320:     }
321:     await this.deliver(send, {
322:       kind: "result-end",
323:       ...this.options.coordinates,
324:       requestId,
325:       transferId: completedManifest.transferId,
326:       rawBytes: completedManifest.rawBytes,

375:   private terminateOperationDelivery(requestId: string, outcome: DaemonDeliveryOutcome): void {
376:     if (!this.options.journal.terminateDelivery(requestId)) return;
377:     const trace = this.operationTraces.get(requestId);
378:     if (trace === undefined) this.options.observer.deliveryTerminated(requestId, outcome, 0);

382:   private disconnectOperationTrace(requestId: string): void {
383:     if (this.operationTraceExpirations.has(requestId)) return;
384:     const trace = this.operationTraces.get(requestId);
385:     if (trace === undefined) return;
386:     trace.clientDisconnected();
387:     const expiration = setTimeout(
388:       () => this.expireOperationTrace(requestId),
389:       this.options.policy.diagnostics.disconnectedTraceRetentionMs,
390:     );
391:     expiration.unref();
392:     this.operationTraceExpirations.set(requestId, expiration);
393:     this.enforceOperationTraceCapacity();
394:   }

396:   private attachOperationTraceConnection(requestId: string, send: DaemonServerSend): () => void {
397:     const connectionCount = this.operationTraceConnections.get(requestId) ?? 0;
398:     if (connectionCount === 0) {
399:       const expiration = this.operationTraceExpirations.get(requestId);
400:       if (expiration !== undefined) clearTimeout(expiration);
401:       this.operationTraceExpirations.delete(requestId);
402:     }
403:     this.operationTraceConnections.set(requestId, connectionCount + 1);
404:     let connectionClosed = false;
405:     const disconnect = (): void => {
406:       if (connectionClosed) return;
407:       connectionClosed = true;
408:       const remainingConnections = (this.operationTraceConnections.get(requestId) ?? 1) - 1;
409:       if (remainingConnections > 0) {
410:         this.operationTraceConnections.set(requestId, remainingConnections);
411:         return;
412:       }
413:       this.operationTraceConnections.delete(requestId);
414:       this.disconnectOperationTrace(requestId);
415:     };
416:     send.onClose(disconnect);
417:     return disconnect;
418:   }

420:   private traceWasDisconnected(requestId: string): boolean {
421:     if (this.operationTraceExpirations.has(requestId)) return true;
422:     return (
423:       !this.operationTraces.has(requestId) && !this.options.journal.isDeliveryTerminated(requestId)
424:     );
425:   }

427:   private reattachOperationTrace(requestId: string): void {
428:     if (this.options.journal.isDeliveryTerminated(requestId)) return;
429:     const trace = this.operationTraces.get(requestId);
430:     if (trace === undefined) this.options.observer.reattached(requestId);
431:     else trace.reattached();
432:   }

434:   private expireOperationTrace(requestId: string): void {
435:     this.operationTraceExpirations.delete(requestId);
436:     this.operationTraceConnections.delete(requestId);
437:     if (!this.operationTraces.delete(requestId)) return;
438:     this.options.observer.traceExpired(requestId);
439:   }

441:   private enforceOperationTraceCapacity(): void {
442:     const capacity = Math.max(1, this.options.policy.diagnostics.maximumDisconnectedTraces);
443:     while (this.operationTraceExpirations.size > capacity) {
444:       const oldestRequestId = this.operationTraceExpirations.keys().next().value as
445:         | string
446:         | undefined;
447:       if (oldestRequestId === undefined) return;
448:       const expiration = this.operationTraceExpirations.get(oldestRequestId);
449:       if (expiration !== undefined) clearTimeout(expiration);
450:       this.expireOperationTrace(oldestRequestId);
451:     }
