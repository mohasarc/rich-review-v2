93:   private async executeAccepted(
94:     request: AuthenticatedDaemonExecuteRequest,
95:     trace: DaemonOperationTrace,
96:   ): Promise<void> {
97:     let completion: DaemonCompletionWriter | undefined;
98:     try {
99:       await this.options.queue.enqueue(
100:         {
101:           requestId: request.requestId,
102:           command: request.commandName,
103:           acceptedAt: this.options.clock.monotonicNowMs(),
104:         },
105:         async () => {
106:           trace.turnStarted(this.options.resourceSupervisor.snapshot.generation);
107:           try {
108:             completion = await this.options.delivery.createCompletion(request.requestId);
109:             this.options.ledger.markRunning(request.requestId, this.options.clock.wallNowMs());
110:             const response = await this.options.worker.execute(
111:               request.requestId,
112:               { commandName: request.commandName, request: request.request },
113:               completion,
114:             );
115:             this.options.resourceSupervisor.workerHeapReported(
116:               response.generation,
117:               response.resources.workerHeapUsedBytes,
118:               response.resources.workerHeapLimitBytes,
119:               response.resources.peakWorkerHeapUsedBytes,
120:             );
121:             trace.workerCompleted(
122:               {
123:                 freshnessMs: response.durations.freshnessMs,
124:                 navigationMs: response.durations.navigationMs,
125:                 renderMs: response.durations.renderMs,
126:                 workerOutputMs: response.durations.outputMs,
127:               },
128:               response.refresh,
129:             );
130:             await completion.finish(response.result.exitCode);
131:             this.lastCompletedMonotonicAt = this.options.clock.monotonicNowMs();
132:             const workspaceExists = await this.options.processLifecycle.workspaceExists();
133:             trace.executionTerminated("completed");
134:             this.options.ledger.complete(
135:               request.requestId,
136:               request.requestId,
137:               this.options.clock.wallNowMs(),
138:             );
139:             await this.options.delivery.trackedCompletion(request.requestId);
140:             if (!workspaceExists) {
141:               await this.options.processLifecycle.workspaceDeletedAfterDelivery();
142:             }
143:           } finally {
144:             this.scheduleTurnCompleteResourceSample();
145:           }
146:         },
147:       );
148:     } catch (error) {
149:       trace.executionTerminated("failed");
150:       this.recordFailure("request", "internal", error);
151:       const shutdown = this.options.processLifecycle.shutdownSnapshot();
152:       const code = DaemonExecutionFailures.classify({
153:         resourceInterrupted: this.resourceInterruptedRequests.delete(request.requestId),
154:         responseCapacityExceeded: error instanceof CompletionSpoolCapacityError,
155:         workerExited: error instanceof DaemonNavigationWorkerExitedError,
156:         ...(shutdown.failureCode === undefined
157:           ? {}
158:           : { shutdownFailureCode: shutdown.failureCode }),
159:         shutdownStarted: shutdown.started,
160:       });
161:       await completion?.dispose().catch((cleanupError) => {
162:         this.recordFailure("completion-cleanup", "internal", cleanupError);
163:       });
164:       this.options.ledger.fail(request.requestId, code, this.options.clock.wallNowMs());
165:     } finally {
166:       if (this.options.queue.isIdle) this.options.lifetime.queueBecameIdle();
167:     }
168:   }
