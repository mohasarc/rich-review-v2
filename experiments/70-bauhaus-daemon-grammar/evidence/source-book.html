284        if (adoptedRecord?.pid !== process.pid) {
285          lease.release();
286          await this.pause();
287          continue;
288        }
289        if (
290          this.options.registry.startupOwnerMatchesProcess(this.options.identity, adoptedRecord)
291        ) {
292          return { lease, record: adoptedRecord };
293        }
294        lease.release();
295      }
296      await this.pause();
297    }
298    throw new Error("Daemon process did not receive startup authorization");
299  }

317  private async handle(
318    request: DaemonRequest,
319    send: DaemonServerSend,
320  ): Promise<DaemonResponse | void> {
321    if (request.kind === "identify") return this.identify(request);
322    if (request.kind === "terminate" || request.kind === "kill") {
323      return this.terminate(request);
324    }
325    if (
326      request.protocolVersion !== DAEMON_PROTOCOL_VERSION ||
327      request.instanceId !== this.options.instanceId
328    ) {
329      throw new Error("Daemon request does not match protocol or instance");
330    }
331    if (request.kind === "ping") return this.pong();
332    if (
333      request.kind === "execute" ||
334      request.kind === "execution-status" ||
335      request.kind === "result-fetch" ||
336      request.kind === "result-ack"
337    ) {
338      if (request.processToken !== this.options.processToken) {
339        throw new Error("Daemon execution request does not match process instance");
340      }
341    }
342    if (request.kind === "execute") return this.acceptExecution(request, send);
343    if (request.kind === "result-fetch") {
344      let disconnectTraceConnection: (() => void) | undefined;
345      if (this.acceptedRequests.entryFor(request.requestId)?.state.state === "completed") {
346        const traceWasDisconnected = !this.operationTraceConnections.has(request.requestId);
347        disconnectTraceConnection = this.attachOperationTraceConnection(request.requestId, send);
348        if (traceWasDisconnected) this.reattachOperationTrace(request.requestId);
349      }
350      try {
351        await this.deliverStoredCompletion(request.requestId, send, request.offset);
352      } catch (error) {
353        disconnectTraceConnection?.();
354        throw error;
355      }
356      return;
357    }
358    if (request.kind === "result-ack") {
359      const spool = await this.completionSpools.open(request.requestId);
360      if (spool === undefined) throw new Error("Accepted request completion is unavailable");
361      if (spool.completedManifest?.transferId !== request.transferId) {
362        throw new Error("Result acknowledgement does not match completion transfer");
363      }
364      await spool.acknowledge().catch((error) => {
365        this.logger.record({
366          kind: "failure",
367          operation: "completion-cleanup",
368          failureCode: "internal",
369          errorName: DaemonLogger.errorName(error),
370        });
371      });
372      this.acceptedRequests.acknowledge(request.requestId);
373      this.completeOperationTrace(request.requestId, "delivered");
374      return {
375        kind: "result-acknowledged",
376        instanceId: this.options.instanceId,
377        processToken: this.options.processToken,
378        requestId: request.requestId,
379        transferId: request.transferId,
380      };
381    }
382    if (request.kind === "execution-status") {
383      return {
384        kind: "execution-status",
385        instanceId: this.options.instanceId,
386        processToken: this.options.processToken,
387        requestId: request.requestId,
388        status: this.acceptedRequests.status(request.requestId),
389      };
390    }
391    this.beginGracefulShutdown();
392    await this.requestQueue.drain();
393    await this.waitForCompletionAcknowledgements();
394    setTimeout(() => void this.shutdown("graceful"), 0);
395    return { kind: "stopped", instanceId: this.options.instanceId };
396  }

467  private async acceptExecution(
468    request: Extract<DaemonRequest, { kind: "execute" }>,
469    send: DaemonServerSend,
470  ): Promise<DaemonResponse | void> {
471    void this.resourceSupervisor.sample("admission").catch((error) => {
472      this.logger.record({
473        kind: "failure",
474        operation: "resource-sample",
475        failureCode: "operation-failed",
476        errorName: DaemonLogger.errorName(error),
477      });
478    });
479    if (!this.workerReady) return this.rejection(request, "not-ready", true);
480    if (this.resourceSupervisor.snapshot.admissionPaused) {
481      return this.rejection(request, "resource-pressure", true);
482    }
483    if (this.requestQueue.state !== "accepting") {
484      return this.rejection(request, "draining", true);
485    }
486    const existing = this.acceptedRequests.entryFor(request.requestId);
487    let entry;
488    try {
489      entry = this.acceptedRequests.accept(request.requestId, request.request);
490    } catch (error) {
491      if (error instanceof AcceptedRequestCorruptionError) {
492        return this.rejection(request, "incompatible", false);
493      }
494      throw error;
495    }
496    if (entry.state.state === "queued") {
497      this.acceptances.set(request.requestId, {
498        acceptedAt: entry.state.acceptedAt,
499        queuePosition: entry.state.queuePosition,
500      });
501      if (existing === undefined) {
502        const trace = this.operationObserver.start(
503          request.requestId,
504          WorkspaceDaemon.commandName(request.request.argv),
505        );
506        this.operationTraces.set(request.requestId, trace);
507        trace.accepted(entry.state.queuePosition, this.resourceSupervisor.snapshot.generation);
508      }
509    }
510    const acceptance = this.acceptances.get(request.requestId);
511    if (acceptance === undefined) throw new Error("Accepted request is missing admission metadata");
512    const traceWasDisconnected = !this.operationTraceConnections.has(request.requestId);
513    const disconnectTraceConnection = this.attachOperationTraceConnection(request.requestId, send);
514    if (existing === undefined) {
515      this.lastNavigationAt = this.now();
516      this.lifetime.navigationAccepted();
517      void this.executeAccepted(request);
518    } else if (traceWasDisconnected) {
519      this.reattachOperationTrace(request.requestId);
520    }
521    try {
522      await this.deliver(send, {
523        kind: "accepted",
524        instanceId: this.options.instanceId,
525        processToken: this.options.processToken,
526        requestId: request.requestId,
527        ...acceptance,
528      });
529    } catch (error) {
530      disconnectTraceConnection();
531      throw error;
532    }
533    if (entry.state.state === "completed") {
534      try {
535        await this.deliverStoredCompletion(request.requestId, send);
536      } catch (error) {
537        disconnectTraceConnection();
538        throw error;
539      }
540      return;
541    }
542    if (entry.state.state === "failed") {
543      try {
544        await this.deliver(send, this.failedFrame(request.requestId, entry.state.code));
545      } catch (error) {
546        disconnectTraceConnection();
547        throw error;
548      }
549      this.completeOperationTrace(request.requestId, "delivered");
550      return;
551    }
552    let unsubscribe: (() => void) | undefined;
553    unsubscribe = this.acceptedRequests.subscribe(request.requestId, (updated) => {
554      if (updated.state.state === "completed") {
555        this.trackCompletionDelivery(request.requestId, send, disconnectTraceConnection);
556        unsubscribe?.();
557      } else if (updated.state.state === "failed") {
558        void this.deliver(send, this.failedFrame(request.requestId, updated.state.code))
559          .then(() => this.completeOperationTrace(request.requestId, "delivered"))
560          .catch((error) =>
561            this.recordDeliveryFailure(request.requestId, error, disconnectTraceConnection),
562          );
563        unsubscribe?.();
564      }
565    });
566  }

568  private async executeAccepted(
569    request: Extract<DaemonRequest, { kind: "execute" }>,
570  ): Promise<void> {
571    let spool: CompletionSpool | undefined;
572    try {
573      await this.requestQueue.enqueue(
574        {
575          requestId: request.requestId,
576          command: WorkspaceDaemon.commandName(request.request.argv),
577          acceptedAt: this.clock.monotonicNowMs(),
578        },
579        async () => {
580          this.operationTraces
581            .get(request.requestId)
582            ?.turnStarted(this.resourceSupervisor.snapshot.generation);
583          try {
584            spool = await this.completionSpools.create(request.requestId);
585            this.acceptedRequests.markRunning(request.requestId, this.now());
586            const generation = this.workerGeneration;
587            if (generation === undefined) throw new Error("Navigation worker is unavailable");
588            const ready = await generation.ready;
589            if (ready.kind !== "ready") throw new Error("Navigation worker did not become ready");
590            const response = await generation.worker.execute(
591              request.requestId,
592              request.request,
593              spool,
594            );
595            if (response.kind !== "result" || response.requestId !== request.requestId) {
596              throw new Error("Navigation worker returned an uncorrelated result");
597            }
598            this.resourceSupervisor.workerHeapReported(
599              response.generation,
600              response.resources.workerHeapUsedBytes,
601              response.resources.workerHeapLimitBytes,
602              response.resources.peakWorkerHeapUsedBytes,
603            );
604            this.operationTraces.get(request.requestId)?.workerCompleted(
605              {
606                freshnessMs: response.durations.freshnessMs,
607                navigationMs: response.durations.navigationMs,
608                renderMs: response.durations.renderMs,
609                workerOutputMs: response.durations.outputMs,
610              },
611              response.refresh,
612            );
613            const spoolStartedAt = this.clock.monotonicNowMs();
614            const manifest = await spool.finish(response.result.exitCode);
615            this.operationTraces
616              .get(request.requestId)
617              ?.spooled(manifest, Math.max(0, this.clock.monotonicNowMs() - spoolStartedAt));
618            const workspaceDeleted = await this.recordCompletion();
619            this.operationTraces.get(request.requestId)?.executionTerminated("completed");
620            this.acceptedRequests.complete(request.requestId, request.requestId, this.now());
621            await this.completionDeliveries.get(request.requestId);
622            if (workspaceDeleted) {
623              await this.waitForCompletionAcknowledgements();
624              setTimeout(() => void this.shutdown("workspace-deleted", true), 0);
625            }
626          } finally {
627            this.scheduleTurnCompleteResourceSample();
628          }
629        },
630      );
631    } catch (error) {
632      this.operationTraces.get(request.requestId)?.executionTerminated("failed");
633      this.logger.record({
634        kind: "failure",
635        operation: "request",
636        failureCode: "internal",
637        errorName: DaemonLogger.errorName(error),
638      });
639      const code =
640        (this.resourceInterruptedRequests.delete(request.requestId)
641          ? "controlled-resource"
642          : undefined) ??
643        (error instanceof CompletionSpoolCapacityError
644          ? "response-capacity"
645          : error instanceof DaemonNavigationWorkerExitedError
646            ? this.shutdownFailureCode === "stopping"
647              ? "stopping"
648              : "worker-exit"
649            : (this.shutdownFailureCode ?? (this.shutdownStarted ? "stopping" : "internal")));
650      await spool?.dispose().catch((cleanupError) => {
651        this.logger.record({
652          kind: "failure",
653          operation: "completion-cleanup",
654          failureCode: "internal",
655          errorName: DaemonLogger.errorName(cleanupError),
656        });
657      });
658      this.acceptedRequests.fail(request.requestId, code, this.now());
659    } finally {
660      if (this.requestQueue.isIdle) this.lifetime.queueBecameIdle();
661    }
662  }

838  private disconnectOperationTrace(requestId: string): void {
839    if (this.operationTraceExpirations.has(requestId)) return;
840    const trace = this.operationTraces.get(requestId);
841    if (trace === undefined) return;
842    trace.clientDisconnected();

844      () => this.expireOperationTrace(requestId),
845      this.policy.values.diagnostics.disconnectedTraceRetentionMs,
846    );
847    expiration.unref();
848    this.operationTraceExpirations.set(requestId, expiration);
849    this.enforceOperationTraceCapacity();

852  private attachOperationTraceConnection(requestId: string, send: DaemonServerSend): () => void {
853    const connectionCount = this.operationTraceConnections.get(requestId) ?? 0;
854    if (connectionCount === 0) {
855      const expiration = this.operationTraceExpirations.get(requestId);
856      if (expiration !== undefined) clearTimeout(expiration);
857      this.operationTraceExpirations.delete(requestId);
858    }
859    this.operationTraceConnections.set(requestId, connectionCount + 1);
860    let connectionClosed = false;
861    const disconnect = (): void => {
862      if (connectionClosed) return;
863      connectionClosed = true;
864      const remainingConnections = (this.operationTraceConnections.get(requestId) ?? 1) - 1;
865      if (remainingConnections > 0) {
866        this.operationTraceConnections.set(requestId, remainingConnections);
867        return;
868      }
869      this.operationTraceConnections.delete(requestId);
870      this.disconnectOperationTrace(requestId);
871    };
872    send.onClose(disconnect);
873    return disconnect;
874  }

900    }
