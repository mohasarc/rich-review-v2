37    this.stopped = true;
38    if (this.timer !== undefined) clearTimeout(this.timer);
39    this.timer = undefined;
40  }

43    if (this.timer !== undefined) clearTimeout(this.timer);
44    const remainingMs = Math.max(0, this.deadline - this.clock.wallNowMs());
45    this.timer = setTimeout(() => this.deadlineReached(), remainingMs);
46    this.timer.unref?.();
47  }

50    this.timer = undefined;
51    if (this.stopped || this.navigationActive) return;
52    this.triggerIdle();
53  }
54

56    if (this.idleTriggered) return;
57    this.idleTriggered = true;
58    void this.onIdle();
59  }

## Daemon Package Extraction
 
**Purpose.** Give daemon policy and mechanism one home with an enforced boundary.
