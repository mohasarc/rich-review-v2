326class GatedWritable extends Writable {
327 private readonly callbacks: Array<(error?: Error | null) => void> = [];
328
329 constructor(
330 private readonly writes: string[],
331 private readonly name: string,
332 ) {
333 super({ highWaterMark: 1 });
334 }
335
336 get pendingCount(): number {
337 return this.callbacks.length;
338 }
339
340 release(): void {
341 this.callbacks.shift()?.();
342 }
343
344 fail(error: Error): void {
345 this.destroy(error);
346 }
347
348 override _write(
349 chunk: Buffer,
350 _encoding: BufferEncoding,
351 callback: (error?: Error | null) => void,
352 ): void {
353 this.writes.push(`${this.name}:${chunk.toString()}`);
354 this.callbacks.push(callback);
355 }
356}

358class ListingCountingFileSystem extends InMemoryFileSystem {
359 readonly directoryReads: string[] = [];
360
361 override async listDir(absPath: string): Promise<readonly string[]> {
362 this.directoryReads.push(absPath);
363 return super.listDir(absPath);
364 }
365
366 override listDirSync(absPath: string): readonly string[] {
367 this.directoryReads.push(absPath);
368 return super.listDirSync(absPath);
369 }
370
371 override metadataSync(absPath: string) {
372 if (!this.isDirectorySync(absPath)) return super.metadataSync(absPath);
373 const entries = super.listDirSync(absPath);
374 return {
375 size: entries.length,
376 modifiedAtMs: 0,
377 changeToken: entries.join("\0"),
378 fileIdentity: absPath,
379 };
380 }
