113function onlyRecord(stateDirectory: string): DaemonRecord {
114 const records = registry(stateDirectory).list();
115 if (records.length !== 1)
116 throw new Error(`Expected one daemon record, received ${records.length}`);
117 return records[0]!;
118}

120function statusProcessIds(stateDirectory: string): readonly number[] {
121 const result = runSymnavBinary(["daemon", "status", "--json"], {
122 cwd: tmpdir(),
123 env: { SYMNAV_STATE_DIR: stateDirectory },
124 });
125 expect(result.status).toBe(0);
126 return (
127 JSON.parse(result.stdout) as { readonly daemons: readonly { readonly pid: number }[] }
128 ).daemons.map((entry) => entry.pid);
129}
