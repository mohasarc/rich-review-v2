736function waitForProcess(child: ChildProcess): Promise<void> {
737 if (child.exitCode !== null) {
738 return child.exitCode === 0
739 ? Promise.resolve()
740 : Promise.reject(new Error(`Startup publisher exited with code ${String(child.exitCode)}`));
741 }
742 if (child.signalCode !== null) {
743 return Promise.reject(
744 new Error(`Startup publisher exited with signal ${String(child.signalCode)}`),
745 );
746 }
747 return new Promise((resolve, reject) => {
748 child.once("error", reject);
749 child.once("exit", (code) => {
750 if (code === 0) resolve();
751 else reject(new Error(`Startup publisher exited with code ${String(code)}`));
752 });
753 });
754}
755
756function waitForKilledProcess(child: ChildProcess): Promise<void> {
757 if (child.signalCode !== null || (child.exitCode !== null && child.exitCode !== 0)) {
758 return Promise.resolve();
759 }
760 if (child.exitCode === 0) {
761 return Promise.reject(new Error("Startup caller did not exit abruptly"));
762 }
763 return new Promise((resolve, reject) => {
764 child.once("error", reject);
765 child.once("exit", (code, signal) => {
766 if (signal !== null || (code !== null && code !== 0)) resolve();
767 else reject(new Error("Startup caller did not exit abruptly"));
768 });
769 });
770}

793 stream: "stdout" | "stderr",

795 const chunks = [];

802async function waitUntil(
803 predicate: () => boolean,
804 timeoutMs = 5_000,
805 failureMessage = "Timed out waiting for daemon shutdown",
806): Promise<void> {
807 const deadline = Date.now() + timeoutMs;
808 while (Date.now() <= deadline) {
809 if (predicate()) return;
810 await new Promise((resolve) => setTimeout(resolve, 10));
811 }
812 throw new Error(failureMessage);
