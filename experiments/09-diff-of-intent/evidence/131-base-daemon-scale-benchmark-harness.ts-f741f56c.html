709 responseBytes: Buffer.byteLength(warm.stdout) + Buffer.byteLength(warm.stderr),
710 processRssPeakBytes: 0,
711 spoolPeakBytes: 0,
712 freshnessMs: 0,
713 navigationMs: 0,
714 renderMs: 0,
715 workerOutputMs: 0,
716 spoolMs: 0,
717 stdoutParity: cold.stdout === warm.stdout,
718 stderrParity: cold.stderr === warm.stderr,
719 exitParity: cold.status === warm.status,
720 nonEmpty:
721 !benchmark.expectNonEmpty ||
722 (warm.status === 0 &&
723 warm.stdout.trim().length > 0 &&
724 DaemonBenchmarkSemanticResult.matches(benchmark.expectation, warm.stdout)),
725 diagnosticMatched: false,
726 telemetryMatched,
727 };
728 }
729
730 private static digest(value: string): string {
731 return createHash("sha256").update(value).digest("hex");
732 }
733}

735export class DaemonBenchmarkSemanticResult {
736 static matches(
737 expectation: DaemonBenchmarkResultExpectation | undefined,
738 stdout: string,
739 ): boolean {
740 if (expectation === undefined) return stdout.trim().length > 0;
741 let result: Record<string, unknown>;
742 try {
743 const parsed = JSON.parse(stdout) as unknown;
744 if (!this.isRecord(parsed)) return false;
745 result = parsed;
746 } catch {
747 return false;
748 }
749 if (expectation.kind === "overview") {
750 return this.arrayLength(result.entries) === expectation.symbols;
751 }
752 if (expectation.kind === "resolve" || expectation.kind === "definition") {
753 return this.arrayLength(result.symbols) === expectation.symbols;
754 }
755 if (expectation.kind === "references") return result.total === expectation.total;
756 if (expectation.kind === "context") {
757 return (
758 this.edgeCount(result.callers) === expectation.callers &&
759 this.edgeCount(result.callees) === expectation.callees &&
760 this.arrayLength(result.history) === expectation.history
761 );
762 }
763 if (expectation.kind === "graph") {
764 return (
765 this.pathCount(result.incoming) === expectation.incomingPaths &&
766 this.pathCount(result.outgoing) === expectation.outgoingPaths
767 );
768 }
769 return (
770 typeof result.totalEvents === "number" &&
771 Array.isArray(result.perCommand) &&
772 Array.isArray(result.outcomes) &&
773 this.isRecord(result.duration)
774 );
775 }
776
777 private static edgeCount(value: unknown): number {
778 if (!this.isRecord(value)) return -1;
779 const visible = this.arrayLength(value.sortedEdges);
780 const omitted = value.omittedCertainEdgeCount;
781 return visible < 0 || typeof omitted !== "number" ? -1 : visible + omitted;
782 }
783
784 private static pathCount(value: unknown): number {
785 if (!this.isRecord(value) || typeof value.totalPathCount !== "number") return -1;
786 return value.totalPathCount;
787 }
788
789 private static arrayLength(value: unknown): number {
790 return Array.isArray(value) ? value.length : -1;
791 }
792
793 private static isRecord(value: unknown): value is Record<string, unknown> {
794 return typeof value === "object" && value !== null && !Array.isArray(value);
795 }
796}

814export class DaemonBenchmarkDiagnostics {
815 private constructor(
816 private readonly operationMetrics: readonly OperationMetrics[],
817 readonly restartCount: number,
818 readonly capacityResultCount: number,
819 readonly rawRuntimeFailureCount: number,
820 readonly processRssPeakBytes: number,
821 readonly workerHeapPeakBytes: number | undefined,
822 readonly spoolPeakBytes: number,
823 readonly phasesComplete: boolean,
824 readonly startupDurations: {
825 readonly discoveryMs: number;
826 readonly indexingMs: number;
827 readonly totalMs: number;
828 },
829 private readonly uncorrelatedOperationEventCount: number,
830 ) {}
831
832 static read(logPath: string, cursor = 0): DaemonBenchmarkDiagnostics {
833 const allEvents = existsSync(logPath)
834 ? readFileSync(logPath, "utf8")
835 .split("\n")
836 .filter((line) => line.length > 0)
837 .map((line) => JSON.parse(line) as Record<string, unknown>)
838 : [];
839 const events = allEvents.slice(cursor);
840 const accepted = events.filter((event) => event.kind === "request-accepted");
841 const acceptedRequestIds = new Set(accepted.map((event) => String(event.requestId)));
842 const operationKinds = new Set([
843 "turn-started",
844 "worker-completed",
845 "response-spooled",
846 "execution-terminal",
847 "delivery-terminal",
848 ]);
849 const uncorrelatedOperationEventCount = events.filter(
850 (event) =>
851 operationKinds.has(String(event.kind)) && !acceptedRequestIds.has(String(event.requestId)),
852 ).length;
853 const operationMetrics = accepted.map((acceptedEvent) => {
854 const requestId = String(acceptedEvent.requestId);
855 const requestEvents = events.filter((event) => String(event.requestId) === requestId);
856 const acceptedEvents = requestEvents.filter((event) => event.kind === "request-accepted");
857 const turns = requestEvents.filter((event) => event.kind === "turn-started");
858 const workers = requestEvents.filter((event) => event.kind === "worker-completed");
859 const spools = requestEvents.filter((event) => event.kind === "response-spooled");
860 const terminals = requestEvents.filter((event) => event.kind === "execution-terminal");
861 const deliveries = requestEvents.filter((event) => event.kind === "delivery-terminal");
862 const turn = turns[0];
863 const spool = spools[0];
864 const terminal = terminals[0];
865 const worker = workers[0];
866 return {
867 requestId,
868 command: String(acceptedEvent.command),
869 queueWaitMs: Number(turn?.queueWaitMs ?? 0),
870 serviceMs: Number(terminal?.serviceMs ?? 0),
871 processRssPeakBytes: Number(
872 terminal?.peakProcessRssBytes ?? terminal?.processRssBytes ?? 0,
873 ),
874 ...(terminal?.peakWorkerHeapUsedBytes === undefined
875 ? {}
876 : { workerHeapPeakBytes: Number(terminal.peakWorkerHeapUsedBytes) }),
877 spoolBytes: Math.max(Number(terminal?.spoolBytes ?? 0), Number(spool?.rawBytes ?? 0)),
878 freshnessMs: Number(worker?.freshnessMs ?? 0),
879 navigationMs: Number(worker?.navigationMs ?? 0),
880 renderMs: Number(worker?.renderMs ?? 0),
881 workerOutputMs: Number(worker?.workerOutputMs ?? 0),
882 spoolMs: Number(spool?.spoolMs ?? 0),
883 complete:
884 acceptedEvents.length === 1 &&
885 turns.length === 1 &&
886 workers.length === 1 &&
887 spools.length === 1 &&
888 terminals.length === 1 &&
889 deliveries.length === 1 &&
890 terminals[0]?.outcome === "completed" &&
891 deliveries[0]?.outcome === "delivered",
892 };
893 });
894 const phases = new Set(events.map((event) => String(event.kind)));
895 const startup = allEvents.filter((event) => event.kind === "startup-completed");
896 const startupEvent = startup[0];
897 return new DaemonBenchmarkDiagnostics(
898 operationMetrics,
899 events.filter((event) => event.kind === "worker-replaced").length,
900 events.filter((event) => event.failureCode === "response-capacity").length,
901 events.filter((event) => event.kind === "process-termination").length,
902 Math.max(0, ...operationMetrics.map((metric) => metric.processRssPeakBytes)),
903 DaemonBenchmarkDiagnostics.maximumOptional(
904 operationMetrics.map((metric) => metric.workerHeapPeakBytes),
905 ),
906 Math.max(0, ...operationMetrics.map((metric) => metric.spoolBytes)),
907 [
908 "startup-completed",
909 "request-accepted",
910 "turn-started",
911 "worker-completed",
912 "response-spooled",
913 "execution-terminal",
914 "delivery-terminal",
915 ].every((phase) => phases.has(phase)) && startup.length === 1,
916 {
917 discoveryMs: Number(startupEvent?.discoveryMs ?? 0),
918 indexingMs: Number(startupEvent?.indexingMs ?? 0),
919 totalMs: Number(startupEvent?.totalMs ?? 0),
920 },
921 uncorrelatedOperationEventCount,
922 );
923 }
924
925 static startupProbeComplete(logPath: string): boolean {
926 if (!existsSync(logPath)) return false;
927 const events = readFileSync(logPath, "utf8")
928 .split("\n")
929 .filter((line) => line.length > 0)
930 .map((line) => JSON.parse(line) as Record<string, unknown>);
931 const versionRequests = new Set(
932 events
933 .filter((event) => event.kind === "request-accepted" && event.command === "version")
934 .map((event) => String(event.requestId)),
935 );
936 return (
937 events.some((event) => event.kind === "startup-completed") &&
938 events.some(
939 (event) =>
940 event.kind === "delivery-terminal" &&
941 event.outcome === "delivered" &&
942 versionRequests.has(String(event.requestId)),
943 )
944 );
945 }

947 static eventCount(logPath: string): number {
948 if (!existsSync(logPath)) return 0;
949 return readFileSync(logPath, "utf8")
950 .split("\n")
951 .filter((line) => line.length > 0).length;
952 }
953
954 enrich(samples: readonly BenchmarkSampleEvidence[]): BenchmarkSampleEvidence[] {
955 return samples.map((sample, index) => {
956 const metric = this.operationMetrics[index];
957 if (metric === undefined || metric.command !== sample.command || !metric.complete)
958 return sample;
959 return {
960 ...sample,
961 serviceMsExcludingQueue: metric!.serviceMs,
962 queueWaitMs: metric!.queueWaitMs,
963 processRssPeakBytes: metric!.processRssPeakBytes,
964 ...(metric!.workerHeapPeakBytes === undefined
965 ? {}
966 : { workerHeapPeakBytes: metric!.workerHeapPeakBytes }),
967 spoolPeakBytes: metric!.spoolBytes,
968 freshnessMs: metric!.freshnessMs,
969 navigationMs: metric!.navigationMs,
970 renderMs: metric!.renderMs,
971 workerOutputMs: metric!.workerOutputMs,
972 spoolMs: metric!.spoolMs,
973 diagnosticMatched: true,
974 };
975 });
976 }
977
978 complete(samples: readonly BenchmarkSampleEvidence[]): boolean {
979 return (
980 this.uncorrelatedOperationEventCount === 0 &&
981 this.operationMetrics.length === samples.length &&
982 this.operationMetrics.every((metric) => metric.complete) &&
983 samples.every((sample) => sample.diagnosticMatched)
984 );
985 }
986
987 ready(expectedOperations: number): boolean {
988 return (
989 this.uncorrelatedOperationEventCount === 0 &&
990 this.operationMetrics.length === expectedOperations &&
991 this.operationMetrics.every((metric) => metric.complete)
992 );
993 }

995 private static maximumOptional(values: readonly (number | undefined)[]): number | undefined {
996 const present = values.filter((value): value is number => value !== undefined);
997 return present.length === 0 ? undefined : Math.max(...present);
998 }

1016export class DaemonBenchmarkPhaseStatistics {

1018 samples: readonly BenchmarkSampleEvidence[],
1019 ): Readonly<Record<string, DaemonBenchmarkCommandPhaseStatistics>> {
1020 const statistics: Record<string, DaemonBenchmarkCommandPhaseStatistics> = {};
1021 for (const command of new Set(samples.map((sample) => sample.command))) {
1022 const commandSamples = samples.filter((sample) => sample.command === command);
1023 statistics[command] = {
1024 freshness: this.distribution(commandSamples.map((sample) => sample.freshnessMs)),
1025 navigation: this.distribution(commandSamples.map((sample) => sample.navigationMs)),
1026 render: this.distribution(commandSamples.map((sample) => sample.renderMs)),
1027 workerOutput: this.distribution(commandSamples.map((sample) => sample.workerOutputMs)),
1028 spool: this.distribution(commandSamples.map((sample) => sample.spoolMs)),
1029 };
1030 }
1031 return statistics;

1034 static startup(durations: {
1035 readonly discoveryMs: number;
1036 readonly indexingMs: number;
1037 readonly totalMs: number;
1038 }): Readonly<Record<string, DaemonBenchmarkDurationStatistics>> {
1039 return {
1040 discovery: this.distribution([durations.discoveryMs]),
1041 indexing: this.distribution([durations.indexingMs]),
1042 total: this.distribution([durations.totalMs]),
1043 };

1046 private static distribution(values: readonly number[]): DaemonBenchmarkDurationStatistics {
1047 const ordered = [...values].sort((left, right) => left - right);
1048 return {
1049 minimumMs: ordered[0]!,
1050 p50Ms: ordered[Math.ceil(ordered.length * 0.5) - 1]!,
1051 p95Ms: ordered[Math.ceil(ordered.length * 0.95) - 1]!,
1052 maximumMs: ordered.at(-1)!,
1053 };
