108 constructor(private readonly options: DaemonCommandDispatcherOptions) {
109 this.selector = options.selector ?? new InvocationWorkspaceSelector();
110 this.daemonEnabled = options.daemonEnabled ?? (() => true);
111 this.resolveWorkspaceRoot =
112 options.resolveWorkspaceRoot ??
113 (async (startDir, dependencies) =>
114 (await createWorkspace({ startDir, fs: dependencies.fs })).root);
115 this.runtimeFactory = options.runtimeFactory ?? DaemonCommandDispatcher.createRuntime;
116 this.executorFactory =
117 options.executorFactory ?? ((dependencies) => new CliProgramExecutor(dependencies));
118 this.requestId = options.requestId ?? randomUUID;
119 this.policy = options.policy;
120 }

122 async execute(request: CliExecutionRequest): Promise<DispatchedCommandResult> {
123 const selected = this.selector.select(request.argv, request.cwd);
124 const route = selected.route;
125 if (route.kind !== "workspace") {
126 return this.executeLocally(request, "cold");
127 }
128 const workspaceRequest: CliExecutionRequest = {
129 ...request,
130 argv: selected.argv,
131 };
132 if (!this.daemonEnabled()) {
133 return this.executeRoute({ kind: "disabled" }, workspaceRequest);
134 }
135
136 const workspaceDependencies = this.options.createDependencies(this.options.stateDirectory);
137 let workspaceRoot: string;
138 try {
139 workspaceRoot = await this.resolveWorkspaceRoot(route.startDir, workspaceDependencies);
140 } catch {
141 return this.executeLocally(workspaceRequest, "cold");
142 }
143
144 const identity = DaemonWorkspaceIdentity.from(workspaceRoot, this.options.stateDirectory);
145 const runtime = this.runtimeFactory(identity, workspaceDependencies, this.policy);
146 const routeSnapshot = await this.routeFor(
147 identity,
148 runtime,
149 workspaceDependencies.symnavVersion,
150 );
151 if (routeSnapshot.kind === "warm") {
152 return this.executeWarm(runtime, routeSnapshot.record, workspaceRequest);
153 }
154 if (
155 (routeSnapshot.kind === "cold" && routeSnapshot.reason === "absent") ||
156 routeSnapshot.kind === "fallback"
157 ) {
158 DaemonCommandDispatcher.triggerIndependently(runtime.coordinator, identity);
159 }
160 return this.executeRoute(routeSnapshot, workspaceRequest);
161 }
162
163 private async routeFor(
164 identity: DaemonWorkspaceIdentity,
165 runtime: DaemonDispatchRuntime,
166 symnavVersion: string,
167 ): Promise<DaemonRouteSnapshot> {
168 let record: DaemonRecord | undefined;
169 try {
170 record = runtime.registry.read(identity);
171 } catch {
172 return { kind: "cold", reason: "recovering" };
173 }
174 if (record === undefined) return { kind: "cold", reason: "absent" };
175 if (record.state === "starting") return { kind: "cold", reason: "starting" };
176 if (record.symnavVersion !== symnavVersion) {
177 return { kind: "fallback", reason: "incompatible" };
178 }
179 let observation: DaemonObservation;
180 try {
181 observation = await runtime.observer.observe(record);
182 } catch {
183 return { kind: "cold", reason: "recovering" };
184 }
185 if (observation.kind === "responsive") {
186 if (observation.pong.symnavVersion !== symnavVersion) {
187 return { kind: "fallback", reason: "incompatible" };
188 }
189 if (observation.pong.state === "starting") {
190 return { kind: "cold", reason: "recovering" };
191 }
192 return { kind: "warm", record };
193 }
194 if (observation.kind === "starting") return { kind: "cold", reason: "starting" };
195 if (observation.kind === "unresponsive") return { kind: "cold", reason: "recovering" };
196 if (observation.kind === "exited") {
197 try {
198 runtime.registry.removeIfProcess(identity, record.instanceId, record.processToken);
199 } catch {}
200 return { kind: "fallback", reason: "dead" };
201 }
202 return { kind: "fallback", reason: "incompatible" };
203 }

205 private async executeWarm(
206 runtime: DaemonDispatchRuntime,
207 record: DaemonRecord,
208 request: CliExecutionRequest,
209 ): Promise<DispatchedCommandResult> {
210 let receipt: DaemonExecutionReceipt;
211 try {
212 receipt = await runtime.transport.execute(record.endpoint, {
213 kind: "execute",
214 protocolVersion: record.protocolVersion,
215 instanceId: record.instanceId,
216 processToken: record.processToken,
217 requestId: this.requestId(),
218 request: { ...request, executionMode: "warm" },
219 });
220 } catch (error) {
221 if (DaemonCommandDispatcher.isRetrySafeFailure(error)) {
222 return this.executeLocally(request, "fallback");
223 }
224 return { mode: "warm", result: ControlledCommandResult.acceptedRequestDidNotComplete() };
225 }
226 try {
227 const completion = await receipt.completion;
228 if (completion.status === "failed") {
229 return {
230 mode: "warm",
231 result: DaemonCommandDispatcher.controlledFailure(completion.code),
232 };
233 }
234 if (!DaemonCommandDispatcher.isCompleteResult(completion.result)) {
235 return { mode: "warm", result: ControlledCommandResult.acceptedRequestDidNotComplete() };
236 }
237 return {
238 mode: "warm",
239 result: completion.result,
240 };
241 } catch {
242 return { mode: "warm", result: ControlledCommandResult.acceptedRequestDidNotComplete() };
243 }
244 }

246 private executeRoute(
247 route: Exclude<DaemonRouteSnapshot, { readonly kind: "warm" }>,
248 request: CliExecutionRequest,
249 ): Promise<DispatchedCommandResult> {
250 return this.executeLocally(request, route.kind === "fallback" ? "fallback" : "cold");

253 private executeLocally(
254 request: CliExecutionRequest,
255 mode: "cold" | "fallback",
256 ): Promise<DispatchedCommandResult> {
257 const executor = this.executorFactory(
258 this.options.createDependencies(this.options.stateDirectory),
259 );
260 return executor
261 .execute({ ...request, executionMode: mode })
262 .then((result) => ({ mode, result }));
263 }
