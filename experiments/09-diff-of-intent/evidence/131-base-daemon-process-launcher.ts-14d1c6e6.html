64 isAlive(pid: number): boolean {
65 if (!Number.isInteger(pid) || pid <= 0) return false;
66 try {
67 process.kill(pid, 0);
68 return true;
69 } catch (error) {
70 return NodeDaemonProcessTerminator.errorCode(error) === "EPERM";
71 }
72 }
73
74 async terminate(pid: number): Promise<void> {
75 if (!this.isAlive(pid)) return;
76 if (pid === process.pid) throw new Error("Refusing to terminate current process");
77 this.signal(pid, "SIGTERM");
78 if (await this.waitForExit(pid)) return;
79 this.signal(pid, "SIGKILL");
80 if (await this.waitForExit(pid)) return;
81 throw new DaemonProcessTerminationError(`Daemon process ${pid} did not terminate`);
82 }
83
84 private signal(pid: number, signal: NodeJS.Signals): void {
85 try {
86 process.kill(pid, signal);
87 } catch (error) {
88 if (NodeDaemonProcessTerminator.errorCode(error) !== "ESRCH") throw error;
89 }
90 }

92 private async waitForExit(pid: number): Promise<boolean> {
93 const deadline = Date.now() + this.gracefulTimeoutMs;
94 while (Date.now() <= deadline) {
95 if (!this.isAlive(pid)) return true;
96 await new Promise((resolve) => setTimeout(resolve, this.pollIntervalMs));
97 }
98 return !this.isAlive(pid);
99 }

102 if (typeof error !== "object" || error === null) return undefined;
