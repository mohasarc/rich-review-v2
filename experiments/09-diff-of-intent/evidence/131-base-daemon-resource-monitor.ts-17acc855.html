160 runAtBoundary: (operation: () => Promise<void>) => Promise<void>,
161 ): Promise<void> {
162 if (this.currentState === "draining" || this.currentState === "stopped") return;
163 this.captureUsage();

165 if (this.currentProcessRssBytes >= policy.hardProcessRssBytes) {
