219class FailingCompletionSpoolStorage extends completionSpoolModule.NodeCompletionSpoolStorage {
220 private failed = false;
221
222 constructor(private readonly operation: "sync" | "close" | "unlink") {
223 super();
224 }
225
226 override async createFile(path: string): Promise<completionSpoolModule.CompletionSpoolFile> {
227 const file = await super.createFile(path);
228 return {
229 write: (bytes) => file.write(bytes),
230 sync: async () => {
231 if (this.fail("sync")) throw new Error("sync failed");
232 await file.sync();
233 },
234 close: async () => {
235 if (this.fail("close")) throw new Error("close failed");
236 await file.close();
237 },
238 };
239 }
240
241 override async unlink(path: string): Promise<void> {
242 if (this.fail("unlink")) throw new Error("unlink failed");
243 await super.unlink(path);
244 }
245
246 private fail(operation: "sync" | "close" | "unlink"): boolean {
247 if (this.failed || this.operation !== operation) return false;
248 this.failed = true;
249 return true;
250 }
251}
