113 private rawBytes = 0;

123 get completedManifest(): CompletionSpoolManifest | undefined {

127 async append(record: CommandOutputRecord): Promise<void> {
128 if (this.terminal) throw new Error("Completion spool is already terminal");
129 if (record.sequence !== this.recordCount) throw new Error("Unexpected command output sequence");
130 if (record.bytes.byteLength > this.options.maximumChunkBytes) {
131 throw new Error("Command output record exceeds chunk capacity");
132 }
133 if (this.rawBytes + record.bytes.byteLength > this.options.maximumResultBytes) {
134 await this.failCapacity();
135 }
136 let currentRecordReserved = false;
137 try {
138 this.options.reserve(record.bytes.byteLength);
139 currentRecordReserved = true;
140 const stored: CommandOutputRecord = { ...record, bytes: Buffer.from(record.bytes) };
141 if (
142 this.file === undefined &&
143 this.rawBytes + stored.bytes.byteLength > this.options.inlineBytes
144 ) {
145 await this.spillInlineRecords();
146 }
147 const encoded = OrderedCommandOutput.encodeRecord(stored);
148 if (this.file === undefined) this.inlineRecords.push(stored);
149 else await this.file.write(encoded);
150 this.hash.update(encoded.subarray(4));
151 this.rawBytes += stored.bytes.byteLength;
152 this.recordCount += 1;
153 } catch (error) {
154 if (currentRecordReserved) this.options.release(record.bytes.byteLength);
155 await this.dispose();
156 throw error;
157 }
158 }
159
160 async finish(exitCode: number): Promise<CompletionSpoolManifest> {
161 if (this.terminal) throw new Error("Completion spool is already terminal");
162 if (!Number.isSafeInteger(exitCode) || exitCode < 0)
163 throw new Error("Invalid command exit code");
164 try {
165 await this.file?.sync();
166 await this.closeFile();
167 this.terminal = true;
168 this.manifest = {
169 transferId: this.options.identity.transferId,
170 requestId: this.options.requestId,
171 instanceId: this.options.identity.instanceId,
172 exitCode,
173 rawBytes: this.rawBytes,
174 recordCount: this.recordCount,
175 sha256: this.hash.digest("hex"),
176 };
177 this.options.complete();
178 return this.manifest;
179 } catch (error) {
180 this.terminal = true;
181 return this.disposeAfterFailure(error);
182 }
183 }
184
185 async *read(offset: number): AsyncIterable<CommandOutputRecord> {
186 if (!this.terminal || this.manifest === undefined) {
187 throw new Error("Completion spool is not complete");
188 }
189 if (!Number.isSafeInteger(offset) || offset < 0 || offset > this.recordCount) {
190 throw new Error("Invalid completion spool offset");
191 }
192 try {
193 const records =
194 this.filePath === undefined
195 ? this.inlineRecords
196 : this.options.storage.records(this.filePath, this.options.maximumChunkBytes);
197 for await (const record of records) {
198 if (record.sequence >= offset) yield record;
199 }
200 } catch (error) {
201 throw new CompletionSpoolReadError(error);
202 }
203 }
204
205 async acknowledge(): Promise<void> {
206 if (!this.terminal) throw new Error("Completion spool is not complete");
207 await this.dispose();
208 }
209
210 async dispose(): Promise<void> {
211 if (this.cleanupFinished) return;
212 if (this.disposal !== undefined) return this.disposal;
213 this.disposal = this.performDispose();
214 try {
215 await this.disposal;
216 } finally {
217 this.disposal = undefined;
218 }
219 }
220
221 private async failCapacity(): Promise<never> {
222 await this.dispose();
223 throw new CompletionSpoolCapacityError();
224 }
225
226 private async spillInlineRecords(): Promise<void> {
227 const instanceDirectory = join(this.options.directory, this.options.identity.instanceId);
228 await this.options.storage.ensureDirectory(this.options.directory);
229 await this.options.storage.ensureDirectory(instanceDirectory);
230 this.filePath = join(instanceDirectory, `${this.options.identity.transferId}.spool`);
231 this.file = await this.options.storage.createFile(this.filePath);
232 for (const record of this.inlineRecords) {
233 await this.file.write(OrderedCommandOutput.encodeRecord(record));
234 }
235 this.inlineRecords.length = 0;
236 }
237
238 private async closeFile(): Promise<void> {
239 if (this.file === undefined) return;
240 await this.file.close();
241 this.file = undefined;
242 }
243
244 private async performDispose(): Promise<void> {
245 this.terminal = true;
246 let cleanupError: unknown;
247 try {
248 await this.closeFile();
249 } catch (error) {
250 cleanupError = error;
251 }
252 if (this.filePath !== undefined) {
253 try {
254 await this.options.storage.unlink(this.filePath);
255 this.filePath = undefined;
256 } catch (error) {
257 if ((error as NodeJS.ErrnoException).code === "ENOENT") this.filePath = undefined;
258 else cleanupError ??= error;
259 }
260 }
261 this.releaseOwnership();
262 if (this.file === undefined && this.filePath === undefined) {
263 this.cleanupFinished = true;
264 this.options.cleanupComplete();
265 }
266 if (cleanupError !== undefined) throw cleanupError;
267 }
268
269 private releaseOwnership(): void {
270 if (this.ownershipReleased) return;
271 this.ownershipReleased = true;
272 this.inlineRecords.length = 0;
273 this.options.release(this.rawBytes);
274 this.options.releaseCompletion();
275 this.options.remove();
276 }
277
278 private async disposeAfterFailure(error: unknown): Promise<never> {
279 try {
280 await this.dispose();
281 } catch (cleanupError) {
282 throw new AggregateError(
283 [error, cleanupError],
284 "Completion spool operation and cleanup failed",
285 );
286 }
287 throw error;
288 }

312 async create(requestId: string): Promise<CompletionSpool> {
313 if (this.spools.has(requestId))
314 throw new Error(`Completion spool already exists: ${requestId}`);
315 let completionCounted = false;
316 const spool = new CompletionSpool({
317 directory: this.options.directory,
318 identity: {
319 workspaceKey: this.options.workspaceKey,
320 instanceId: this.options.instanceId,
321 transferId: randomUUID(),
322 },
323 requestId,
324 inlineBytes: this.inlineBytes,
325 maximumResultBytes: this.maximumResultBytes,
326 maximumChunkBytes: this.maximumChunkBytes,
327 reserve: (bytes) => this.reserve(bytes),
328 release: (bytes) => {
329 this.rawBytes -= bytes;
330 },
331 complete: () => {
332 completionCounted = true;
333 this.completionCount += 1;
334 },
335 remove: () => {
336 if (this.spools.get(requestId) !== spool) return;
337 this.spools.delete(requestId);
338 },
339 releaseCompletion: () => {
340 if (completionCounted) this.completionCount -= 1;
341 completionCounted = false;
342 },
343 cleanupComplete: () => {
344 this.pendingCleanup.delete(spool);
345 },
346 storage: this.storage,
347 });
348 this.spools.set(requestId, spool);
349 this.pendingCleanup.add(spool);
350 return spool;
351 }

353 async open(requestId: string): Promise<CompletionSpool | undefined> {
354 return this.spools.get(requestId);
355 }

359 }

361 async cleanupInstance(instanceId: string): Promise<void> {
362 if (instanceId !== this.options.instanceId) return;
363 const failures: unknown[] = [];
364 for (const spool of [...this.pendingCleanup]) {
365 await spool.dispose().catch((error) => failures.push(error));
366 }
367 await this.storage
368 .removeInstance(join(this.options.directory, instanceId))
369 .catch((error) => failures.push(error));
370 if (failures.length > 0) throw new AggregateError(failures, "Completion spool cleanup failed");
371 }

374 await this.cleanupInstance(instanceId);
