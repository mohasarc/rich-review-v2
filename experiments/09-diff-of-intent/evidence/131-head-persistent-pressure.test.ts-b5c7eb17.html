206function captureProcess(child: ChildProcess): Promise<CliResult> {
207 let stdout = "";
208 let stderr = "";
209 child.stdout?.setEncoding("utf8").on("data", (chunk: string) => (stdout += chunk));
210 child.stderr?.setEncoding("utf8").on("data", (chunk: string) => (stderr += chunk));
211 return new Promise((resolve, reject) => {
212 child.once("error", reject);
213 child.once("close", (status) => resolve({ status, stdout, stderr }));
214 });
215}

217function captureChildOutput(child: ChildProcess): Promise<{ stdout: string; stderr: string }> {
218 return captureProcess(child).then(({ stdout, stderr }) => ({ stdout, stderr }));
219}

221function waitForProcess(child: ChildProcess): Promise<void> {
222 if (child.exitCode !== null || child.signalCode !== null) return Promise.resolve();
223 return new Promise((resolve, reject) => {
224 child.once("error", reject);
225 child.once("close", () => resolve());
226 });

229async function waitUntil(predicate: () => boolean | Promise<boolean>): Promise<void> {
230 const deadline = Date.now() + 5_000;
231 while (Date.now() <= deadline) {
232 if (await predicate()) return;
233 await new Promise((resolve) => setTimeout(resolve, 10));
234 }
235 throw new Error("Timed out waiting for persistent pressure state");
