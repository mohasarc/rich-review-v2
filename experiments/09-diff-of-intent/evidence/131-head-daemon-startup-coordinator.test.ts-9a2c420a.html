1394function spawnIdentifiableDaemon(
1395 identity: DaemonWorkspaceIdentity,
1396 instanceId: string,
1397 processToken: string,
1398 startedAt: number,
1399 processIds: number[],
1400): Promise<number> {
1401 if (process.platform !== "win32") {
1402 mkdirSync(dirname(identity.endpoint(instanceId)), { recursive: true, mode: 0o700 });
1403 }
1404 return new Promise((resolve, reject) => {
1405 const child = spawn(
1406 process.execPath,
1407 [
1408 "-e",
1409 identifiableDaemonSource,
1410 identity.endpoint(instanceId),
1411 instanceId,
1412 processToken,
1413 String(startedAt),
1414 ],
1415 { stdio: ["ignore", "pipe", "ignore"] },
1416 );
1417 child.once("error", reject);
1418 child.stdout?.once("data", () => {
1419 processIds.push(child.pid!);
1420 resolve(child.pid!);
1421 });
1422 });
1423}
1424
1425const identifiableDaemonSource = `
1426const { createServer } = require("node:net");
1427const [endpoint, instanceId, processToken, startedAtText] = process.argv.slice(1);
1428const startedAt = Number(startedAtText);
1429const frame = (value) => {
1430 const payload = Buffer.from(JSON.stringify(value));
1431 const prefix = Buffer.alloc(4);
1432 prefix.writeUInt32BE(payload.length);
1433 return Buffer.concat([prefix, payload]);
1434};
1435const server = createServer((socket) => {
1436 let bytes = Buffer.alloc(0);
1437 socket.on("data", (chunk) => {
1438 bytes = Buffer.concat([bytes, chunk]);
1439 if (bytes.length < 4) return;
1440 const length = bytes.readUInt32BE(0);
1441 if (bytes.length < length + 4) return;
1442 const request = JSON.parse(bytes.subarray(4, length + 4).toString("utf8"));
1443 if (request.kind === "identify") {
1444 socket.end(frame({ kind: "identity", instanceId, processToken, pid: process.pid, startedAt }));
1445 return;
1446 }
1447 if (request.kind === "ping") {
1448 socket.end(frame({ kind: "pong", protocolVersion: ${String(
1449 DAEMON_PROTOCOL_VERSION,
1450 )}, instanceId, symnavVersion: "0.1.0", startedAt }));
1451 return;
1452 }
1453 if (request.kind === "terminate") {
1454 socket.end(frame({ kind: "terminating", instanceId, processToken }), () => server.close(() => process.exit(0)));
1455 }
1456 });
1457});
1458server.listen(endpoint, () => process.stdout.write("ready"));
1459`;
