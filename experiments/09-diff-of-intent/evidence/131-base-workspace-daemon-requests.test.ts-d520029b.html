1739 if (this.handler === undefined) throw new Error("Transport is not listening");
1740 const frames: DaemonServerMessage[] = [];
1741 let connected = true;
1742 const closeListeners = new Set<() => void>();
1743 let resolveTerminal!: (frame: DaemonExecutionServerFrame) => void;
1744 const terminal = new Promise<DaemonExecutionServerFrame>((resolve) => {
1745 resolveTerminal = resolve;

1748 if (!("kind" in response) && this.resultChunkGate !== undefined) {
1749 this.resultChunksInFlight += 1;

1751 this.maximumResultChunksInFlight,
1752 this.resultChunksInFlight,

1763 response.kind === "result-end" ||
1764 response.kind === "execution-failed"

1766 resolveTerminal(response);
1767 }

1770 onClose: (listener: () => void): (() => void) => {
1771 closeListeners.add(listener);
1772 return () => closeListeners.delete(listener);
1773 },
1774 });
1775 const response = await this.handler(request, send);
1776 if (response !== undefined) await receive(response);

1778 frames,
1779 terminal,
1780 disconnect: () => {
1781 connected = false;
1782 for (const listener of closeListeners) listener();
