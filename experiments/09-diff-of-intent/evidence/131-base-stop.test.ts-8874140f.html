226async function startControlledDaemon(
227 stateDirectory: string,
228 workspaceRoot: string,
229 releasePath?: string,
230): Promise<ControlledDaemon> {
231 const identity = DaemonWorkspaceIdentity.from(
232 workspaceRoot,
233 StateDirectoryResolver.canonicalize(stateDirectory),
234 );
235 const registry = new DaemonRegistry(identity.registryDirectory);
236 const instanceId = `controlled-${releasePath === undefined ? "forced" : "draining"}`;
237 const processToken = `${instanceId}-token`;
238 const readyPath = join(stateDirectory, `${instanceId}-ready`);
239 const requestStartedPath = join(stateDirectory, `${instanceId}-request`);
240 const lease = registry.acquireStartup(identity, instanceId);
241 if (lease === undefined) throw new Error("Expected controlled daemon startup ownership");
242 const child = spawn(
243 process.execPath,
244 [
245 fileURLToPath(new URL("../../../node_modules/tsx/dist/cli.mjs", import.meta.url)),
246 fileURLToPath(new URL("../../helpers/workspace-daemon-stuck.ts", import.meta.url)),
247 workspaceRoot,
248 stateDirectory,
249 instanceId,
250 processToken,
251 readyPath,
252 requestStartedPath,
253 ...(releasePath === undefined ? [] : [releasePath]),
254 ],
255 { stdio: "ignore" },
256 );
257 await waitUntil(() => existsSync(`${readyPath}.boot`));
258 const daemonPid = Number(readFileSync(`${readyPath}.boot`, "utf8"));
259 const record: DaemonRecord = {
260 schemaVersion: DAEMON_RECORD_SCHEMA_VERSION,
261 protocolVersion: DAEMON_PROTOCOL_VERSION,
262 symnavVersion: "test",
263 workspaceRoot,
264 workspaceKey: identity.workspaceKey,
265 stateKey: identity.stateKey,
266 identityKey: identity.identityKey,
267 instanceId,
268 processToken,
269 endpoint: identity.endpoint(instanceId),
270 pid: daemonPid,
271 state: "starting",
272 startedAt: Date.now(),
273 memoryCapBytes: Number.MAX_SAFE_INTEGER,
274 };
275 if (!registry.writeStartingIfStartupOwner(identity, record)) {
276 throw new Error("Controlled daemon lost startup ownership");
277 }
278 await waitUntil(() => existsSync(readyPath));
279 lease.release();
280 const readyRecord = registry.read(identity);
281 if (readyRecord?.state !== "ready") throw new Error("Controlled daemon did not become ready");
282 return { child, record: readyRecord, requestStartedPath };
283}

285function runBuiltStop(
286 cwd: string,
287 stateDirectory: string,
288): Promise<{ readonly status: number | null; readonly stdout: string; readonly stderr: string }> {
289 const child = spawn(
290 process.execPath,
291 [fileURLToPath(new URL("../../../dist/cli.js", import.meta.url)), "daemon", "stop", "--json"],
292 {
293 cwd,
294 env: { ...process.env, SYMNAV_STATE_DIR: stateDirectory, SYMNAV_TELEMETRY: "0" },
295 stdio: ["ignore", "pipe", "pipe"],
296 },
297 );
298 let stdout = "";
299 let stderr = "";
300 child.stdout?.setEncoding("utf8").on("data", (chunk: string) => {
301 stdout += chunk;
302 });
303 child.stderr?.setEncoding("utf8").on("data", (chunk: string) => {
304 stderr += chunk;
305 });
306 return new Promise((resolve, reject) => {
307 child.once("error", reject);
308 child.once("exit", (status) => resolve({ status, stdout, stderr }));
309 });
310}

312function waitForProcess(child: ChildProcess): Promise<void> {
313 if (child.exitCode !== null) {
314 return child.exitCode === 0
315 ? Promise.resolve()
316 : Promise.reject(new Error(`Controlled daemon exited with code ${String(child.exitCode)}`));
317 }
318 return new Promise((resolve, reject) => {
319 child.once("error", reject);
320 child.once("exit", (code) => {
321 if (code === 0) resolve();
322 else reject(new Error(`Controlled daemon exited with code ${String(code)}`));
323 });
324 });
325}

327async function waitUntil(predicate: () => boolean): Promise<void> {
328 const deadline = Date.now() + 5_000;
329 while (Date.now() <= deadline) {
330 if (predicate()) return;
331 await new Promise((resolve) => setTimeout(resolve, 10));
332 }
333 throw new Error("Timed out waiting for controlled daemon state");
334}

366function daemonLogEvents(stateDir: string): readonly Record<string, unknown>[] {
367 const logPath = DaemonStateFiles.matchingPaths(stateDir, ".log")[0];
368 if (logPath === undefined) return [];

370 .trim()
371 .split("\n")
372 .filter((line) => line.startsWith("{"))
373 .map((line) => JSON.parse(line) as Record<string, unknown>);
