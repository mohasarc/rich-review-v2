715class DeferredExecutor implements DaemonCommandExecutor {
716 readonly started: Promise<void>;
717 private resolveStarted!: () => void;
718 private resolveResult!: (result: CommandExecutionResult) => void;
719 private readonly result: Promise<CommandExecutionResult>;
720
721 constructor() {
722 this.started = new Promise((resolve) => {
723 this.resolveStarted = resolve;
724 });
725 this.result = new Promise((resolve) => {
726 this.resolveResult = resolve;
727 });
728 }
729
730 execute(_request: CliExecutionRequest): Promise<CommandExecutionResult> {
731 this.resolveStarted();
732 return this.result;
733 }
734
735 complete(result: CommandExecutionResult): void {
736 this.resolveResult(result);
737 }
738}

796function waitForProcess(child: ChildProcess): Promise<void> {
797 return new Promise((resolve, reject) => {
798 child.once("error", reject);
799 child.once("exit", (code) => {
800 if (code === 0) resolve();
801 else reject(new Error(`Stuck daemon exited with code ${String(code)}`));
802 });
803 });
804}

806async function waitUntil(predicate: () => boolean): Promise<void> {
807 const deadline = Date.now() + 5_000;
808 while (Date.now() <= deadline) {
809 if (predicate()) return;
810 await new Promise((resolve) => setTimeout(resolve, 10));
811 }
812 throw new Error("Timed out waiting for daemon runtime state");
