173 private findReferenceLocations(identity: SymbolIdentity): readonly SemanticReferenceLocation[] {
174 const declarationNodes = this.workspaceState
175 .locateSemanticCopies(identity)
176 .map((located) => located.node);
177 if (declarationNodes.length === 0) throw new SymbolNotFoundError(identity);
178 const locations = declarationNodes.flatMap((declarationNode) =>
179 TypeScriptSemanticQueryService.referenceEntriesOf(declarationNode).flatMap((entry) => {
180 const node = entry.getNode();
181 const relativePath = this.workspaceState.relativePathOf(node.getSourceFile());
182 return relativePath
183 ? [
184 {
185 relativePath,
186 start: node.getStart(),
187 length: node.getWidth(),
188 isDefinition: entry.isDefinition() ?? false,
189 },
190 ]
191 : [];
192 }),
193 );
194 const byLocation = new Map<string, SemanticReferenceLocation>();
195 for (const location of locations) {
196 byLocation.set(
197 `${location.relativePath}:${location.start}:${location.length}:${location.isDefinition}`,
198 location,
199 );
200 }
201 return [...byLocation.values()];
202 }

204 private async resolveCallTarget(identity: SymbolIdentity): Promise<CallTargetResolution> {
205 const definitions = await this.findDefinitions(identity);
206 if (definitions.length === 0) return { outcome: "not-found" };
207 const implementations = definitions.filter((definition) =>
208 definition.kind.nativeLabel.endsWith("-implementation"),
209 );
210 if (implementations.length > 1) {
211 return { outcome: "ambiguous", candidates: implementations };
212 }
213 if (implementations.length === 1) {
214 return { outcome: "resolved", target: implementations[0]! };
215 }
216 return { outcome: "resolved", target: definitions[0]! };
217 }

219 private nodeAtSemanticLocation(
220 location: SemanticNodeLocation,
221 sourceFile: SourceFile | undefined,
222 ): Node | undefined {
223 let node =
224 sourceFile?.getDescendantAtPos(location.start) ??
225 this.workspaceState.nodeAt(location.relativePath, location.start);
226 while (node) {
227 if (node.getStart() === location.start && node.getKind() === location.kind) return node;
228 node = node.getParent();
229 }
230 return undefined;
231 }

240 private static lineText(sourceFile: SourceFile, zeroBasedLine: number): string {
241 const fullText = sourceFile.getFullText();
242 const lineStarts = sourceFile.compilerNode.getLineStarts();
243 const start = lineStarts[zeroBasedLine] ?? 0;
244 const end =
245 zeroBasedLine + 1 < lineStarts.length ? lineStarts[zeroBasedLine + 1]! : fullText.length;
246 return fullText.slice(start, end).replace(/\r?\n$/, "");
247 }
