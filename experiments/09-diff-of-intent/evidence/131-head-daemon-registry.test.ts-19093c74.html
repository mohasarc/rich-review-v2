749class ControllerTransport {
750 readonly live = new Set<string>();
751 readonly killed: string[] = [];
752 onStop: (() => void) | undefined;
753 onKill: (() => void) | undefined;
754
755 constructor(private readonly registry: DaemonRegistry) {}
756
757 async request(_endpoint: string, request: DaemonRequest): Promise<DaemonResponse> {
758 const records = this.registry.list();
759 const record = records.find((candidate) => candidate.instanceId === request.instanceId);
760 if (record === undefined || !this.live.has(request.instanceId)) throw new Error("unreachable");
761 if (request.kind === "identify") {
762 return {
763 kind: "identity",
764 instanceId: record.instanceId,
765 processToken: record.processToken,
766 pid: record.pid,
767 startedAt: record.startedAt,
768 };
769 }
770 if (request.kind === "ping") {
771 return {
772 kind: "pong",
773 protocolVersion: DAEMON_PROTOCOL_VERSION,
774 instanceId: record.instanceId,
775 symnavVersion: record.symnavVersion,
776 startedAt: record.startedAt,
777 fileCount: record.fileCount ?? 0,
778 memoryBytes: 1234,
779 ...(record.lastNavigationAt === undefined
780 ? {}
781 : { lastNavigationAt: record.lastNavigationAt }),
782 };
783 }
784 if (request.kind === "stop") {
785 this.onStop?.();
786 return { kind: "stopped", instanceId: record.instanceId };
787 }
788 if (request.kind === "kill") {
789 this.onKill?.();
790 this.killed.push(record.instanceId);
791 return {
792 kind: "killing",
793 instanceId: record.instanceId,
794 processToken: record.processToken,
795 };
796 }
797 throw new Error(`Unsupported controller request ${request.kind}`);
798 }
799
800 async removeUnavailableEndpoint(): Promise<boolean> {
801 return true;
802 }
803}

851function waitForProcess(child: ChildProcess): Promise<void> {
852 if (child.exitCode !== null) {
853 return child.exitCode === 0
854 ? Promise.resolve()
855 : Promise.reject(new Error(`Registry cleaner exited with code ${String(child.exitCode)}`));
856 }
857 if (child.signalCode !== null) {
858 return Promise.reject(
859 new Error(`Registry cleaner exited with signal ${String(child.signalCode)}`),
860 );
861 }
862 return new Promise((resolve, reject) => {
863 child.once("error", reject);
864 child.once("exit", (code) => {
865 if (code === 0) resolve();
866 else reject(new Error(`Registry cleaner exited with code ${String(code)}`));
867 });
868 });
869}

871async function waitUntil(predicate: () => boolean): Promise<void> {
872 const deadline = Date.now() + 5_000;
873 while (Date.now() <= deadline) {
874 if (predicate()) return;
875 await new Promise((resolve) => setTimeout(resolve, 2));
876 }
877 throw new Error("Timed out waiting for registry race");
878}
