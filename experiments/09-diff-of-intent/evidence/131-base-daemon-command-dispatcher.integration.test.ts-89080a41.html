112class StartupBarrier {
113 private readonly waiting: Promise<void>;
114 private releaseWaiting!: () => void;

116 constructor() {

131class BarrierDaemonLauncher implements DaemonProcessLauncher {
132 readonly symnavVersion = "0.1.0";
133 readonly memoryCapBytes = 1024;
134 launchCount = 0;
135 private readonly pid = 12_345;
136
137 constructor(
138 private readonly registry: DaemonRegistry,
139 private readonly identity: DaemonWorkspaceIdentity,
140 private readonly processTerminator: LiveTestProcessTerminator,
141 private readonly barrier: StartupBarrier,
142 ) {}
143
144 async launch(
145 _identity: DaemonWorkspaceIdentity,
146 instanceId: string,
147 _processToken: string,
148 ): Promise<DaemonProcess> {
149 this.launchCount += 1;
150 this.processTerminator.alive.add(this.pid);
151 void this.barrier.wait().then(() => this.publishReady(instanceId));
152 return {
153 pid: this.pid,
154 exited: new Promise<never>(() => {}),
155 terminate: async () => {
156 this.processTerminator.alive.delete(this.pid);
157 },
158 };
159 }
160
161 private publishReady(instanceId: string): void {
162 const starting = this.registry.readStoredInstance(this.identity, instanceId);
163 if (starting?.state !== "starting") return;
164 const ready: DaemonRecord = {
165 ...starting,
166 state: "ready" as const,
167 readyAt: 2,
168 fileCount: REFERENCE_WORKSPACE_FILE_COUNT,
169 };
170 this.registry.writeIfStartupOwner(this.identity, ready);
171 this.registry.removeStartupLockIfProcess(this.identity, ready);
172 }
173}

187class RegistryDaemonTransport {
188 constructor(
189 private readonly registry: DaemonRegistry,
190 private readonly identity: DaemonWorkspaceIdentity,
191 ) {}
192
193 async request(
194 _endpoint: string,
195 request: DaemonLifecycleRequest,
196 ): Promise<DaemonLifecycleResponse> {
197 const record = this.registry.readStoredInstance(this.identity, request.instanceId);
198 if (record === undefined) throw new Error("Missing daemon record");
199 if (request.kind === "identify") {
200 return {
201 kind: "identity",
202 instanceId: record.instanceId,
203 processToken: record.processToken,
204 pid: record.pid,
205 startedAt: record.startedAt,
206 };
207 }
208 if (request.kind === "ping") {
209 return {
210 kind: "pong",
211 protocolVersion: record.protocolVersion,
212 instanceId: record.instanceId,
213 symnavVersion: record.symnavVersion,
214 state: record.state,
215 startedAt: record.startedAt,
216 ...(record.fileCount === undefined ? {} : { fileCount: record.fileCount }),
217 };
218 }
219 throw new Error(`Unexpected startup request ${request.kind}`);
220 }
221}
