 278   private async waitForStartupAuthorization(): Promise<{
 279     readonly lease: DaemonStartupLease;
 280     readonly record: DaemonRecord;
 281   }> {
 282     const deadline = this.now() + this.policy.values.startup.coordinationGraceMs;
 283     while (this.now() <= deadline) {
 284       const record = this.options.registry.readInstance(
 285         this.options.identity,
 286         this.options.instanceId,
 287       );
 288       if (
 289         record?.state === "starting" &&
 290         (record.pid === 0 || record.pid === process.pid) &&
 291         record.processToken === this.options.processToken
 292       ) {
 293         const lease = this.options.registry.claimStartupForDaemon(
 294           this.options.identity,
 295           this.options.instanceId,
 296           this.options.processToken,
 297           process.pid,
 298         );
 299         if (lease === undefined) {
 300           await this.pause();
 301           continue;
 302         }
 303         this.options.registry.writeStartingIfStartupOwner(this.options.identity, {
 304           ...record,
 305           pid: process.pid,
 306         });
 307         const adoptedRecord = this.options.registry.readInstance(
 308           this.options.identity,
 309           this.options.instanceId,
 310         );
 311         if (adoptedRecord?.pid !== process.pid) {
 312           lease.release();
 313           await this.pause();
 314           continue;
 315         }
 316         if (
 317           this.options.registry.startupOwnerMatchesProcess(this.options.identity, adoptedRecord)
 318         ) {
 319           return { lease, record: adoptedRecord };
 320         }
 321         lease.release();
 322       }
 323       await this.pause();
 324     }
 325     throw new Error("Daemon process did not receive startup authorization");
 326   }

 344   private async handle(
 345     request: DaemonRequest,
 346     send: DaemonServerSend,
 347   ): Promise<DaemonResponse | void> {
 348     if (request.kind === "identify") return this.identify(request);
 349     if (request.kind === "terminate" || request.kind === "kill") {
 350       return this.terminate(request);
 351     }
 352     if (
 353       request.protocolVersion !== DAEMON_PROTOCOL_VERSION ||
 354       request.instanceId !== this.options.instanceId
 355     ) {
 356       throw new Error("Daemon request does not match protocol or instance");
 357     }
 358     if (request.kind === "ping") return this.pong();
 359     if (request.kind === "execute") return this.acceptExecution(request, send);
 360     if (
 361       request.kind === "execution-status" ||
 362       request.kind === "result-fetch" ||
 363       request.kind === "result-ack"
 364     ) {
 365       if (request.processToken !== this.options.processToken) {
 366         throw new Error("Daemon execution request does not match process instance");
 367       }
 368     }
 369     if (request.kind === "result-fetch") {
 370       await this.deliverySession.fetch(request, send);
 371       return;
 372     }
 373     if (request.kind === "result-ack") {
 374       return this.deliverySession.acknowledge(request);
 375     }
 376     if (request.kind === "execution-status") {
 377       return {
 378         kind: "execution-status",
 379         instanceId: this.options.instanceId,
 380         processToken: this.options.processToken,
 381         requestId: request.requestId,
 382         status: this.acceptedRequests.status(request.requestId),
 383       };
 384     }
 385     this.beginGracefulShutdown();
 386     await this.requestQueue.drain();
 387     await this.deliverySession.waitForCompletionAcknowledgements();
 388     setTimeout(() => void this.shutdown("graceful"), 0);
 389     return { kind: "stopped", instanceId: this.options.instanceId };
 390   }

 459   private async acceptExecution(
 460     request: Extract<DaemonRequest, { kind: "execute" }>,
 461     send: DaemonServerSend,
 462   ): Promise<DaemonResponse | void> {
 463     const decision = this.decideAdmission(request);
 464     if (decision.kind === "disconnect") {
 465       throw new Error("Daemon execution request does not match process instance");
 466     }
 467     if (decision.kind === "reject") return this.rejection(request, decision.code);
 468     const existing = this.acceptedRequests.entryFor(request.requestId);
 469     const entry = this.acceptedRequests.accept(
 470       request.requestId,
 471       request.commandName,
 472       request.request,
 473     );
 474     if (entry.state.state === "queued") {
 475       this.acceptances.set(request.requestId, {
 476         acceptedAt: entry.state.acceptedAt,
 477         queuePosition: entry.state.queuePosition,
 478       });
 479     }
 480     const acceptance = this.acceptances.get(request.requestId);
 481     if (acceptance === undefined) throw new Error("Accepted request is missing admission metadata");
 482     if (existing === undefined) {
 483       this.lastNavigationAt = this.now();
 484       this.lifetime.navigationAccepted();
 485       const trace = this.deliverySession.beginAcceptedTrace(
 486         request.requestId,
 487         request.commandName,
 488         entry.state.state === "queued" ? entry.state.queuePosition : acceptance.queuePosition,
 489         this.resourceSupervisor.snapshot.generation,
 490       );
 491       void this.executeAccepted(request, trace);
 492     }
 493     await this.deliverySession.attach(
 494       {
 495         requestId: request.requestId,
 496         ...acceptance,
 497       },
 498       send,
 499     );
 500   }

 538   private async executeAccepted(
 539     request: Extract<DaemonRequest, { kind: "execute" }>,
 540     trace: DaemonOperationTrace,
 541   ): Promise<void> {
 542     let completion: DaemonCompletionWriter | undefined;
 543     try {
 544       await this.requestQueue.enqueue(
 545         {
 546           requestId: request.requestId,
 547           command: request.commandName,
 548           acceptedAt: this.clock.monotonicNowMs(),
 549         },
 550         async () => {
 551           trace.turnStarted(this.resourceSupervisor.snapshot.generation);
 552           try {
 553             completion = await this.deliverySession.createCompletion(request.requestId);
 554             this.acceptedRequests.markRunning(request.requestId, this.now());
 555             const response = await this.workerManager.execute(
 556               request.requestId,
 557               { commandName: request.commandName, request: request.request },
 558               completion,
 559             );
 560             this.resourceSupervisor.workerHeapReported(
 561               response.generation,
 562               response.resources.workerHeapUsedBytes,
 563               response.resources.workerHeapLimitBytes,
 564               response.resources.peakWorkerHeapUsedBytes,
 565             );
 566             trace.workerCompleted(
 567               {
 568                 freshnessMs: response.durations.freshnessMs,
 569                 navigationMs: response.durations.navigationMs,
 570                 renderMs: response.durations.renderMs,
 571                 workerOutputMs: response.durations.outputMs,
 572               },
 573               response.refresh,
 574             );
 575             await completion.finish(response.result.exitCode);
 576             const workspaceDeleted = await this.recordCompletion();
 577             trace.executionTerminated("completed");
 578             this.acceptedRequests.complete(request.requestId, request.requestId, this.now());
 579             await this.deliverySession.trackedCompletion(request.requestId);
 580             if (workspaceDeleted) {
 581               await this.deliverySession.waitForCompletionAcknowledgements();
 582               setTimeout(() => void this.shutdown("workspace-deleted", true), 0);
 583             }
 584           } finally {
 585             this.scheduleTurnCompleteResourceSample();
 586           }
 587         },
 588       );
 589     } catch (error) {
 590       trace.executionTerminated("failed");
 591       this.logger.record({
 592         kind: "failure",
 593         operation: "request",
 594         failureCode: "internal",
 595         errorName: DaemonLogger.errorName(error),
 596       });
 597       const code = DaemonExecutionFailures.classify({
 598         resourceInterrupted: this.resourceInterruptedRequests.delete(request.requestId),
 599         responseCapacityExceeded: error instanceof CompletionSpoolCapacityError,
 600         workerExited: error instanceof DaemonNavigationWorkerExitedError,
 601         ...(this.shutdownFailureCode === undefined
 602           ? {}
 603           : { shutdownFailureCode: this.shutdownFailureCode }),
 604         shutdownStarted: this.shutdownStarted,
 605       });
 606       await completion?.dispose().catch((cleanupError) => {
 607         this.logger.record({
 608           kind: "failure",
 609           operation: "completion-cleanup",
 610           failureCode: "internal",
 611           errorName: DaemonLogger.errorName(cleanupError),
 612         });
 613       });
 614       this.acceptedRequests.fail(request.requestId, code, this.now());
 615     } finally {
 616       if (this.requestQueue.isIdle) this.lifetime.queueBecameIdle();
 617     }
 618   }

 646       await access(path);

 648     } catch {
 649       return false;
 650     }
 651   }

 674   private async initiateResourceDrain(): Promise<void> {
 675     await this.deliverySession.waitForCompletionAcknowledgements();
 676     void this.shutdown("resource", true).catch((error) => {
 677       this.logger.record({
 678         kind: "failure",
 679         operation: "resource-drain",
 680         failureCode: "controlled-resource",
 681         errorName: DaemonLogger.errorName(error),
 682       });
 683     });
 684   }

 686   private async shutdown(
 687     reason: "graceful" | "idle" | "resource" | "workspace-deleted",
 688     force = false,
 689   ): Promise<void> {
 690     this.shutdownFailureCode ??= reason === "resource" ? "controlled-resource" : "stopping";
 691     if (force) this.forceWorkerShutdown();
 692     if (this.shutdownOperation !== undefined) return this.shutdownOperation;
 693     this.shutdownStarted = true;
 694     this.shutdownOperation = this.completeShutdown(reason, force);
 695     return this.shutdownOperation;
 696   }
 697 
 698   private async completeShutdown(
 699     reason: "graceful" | "idle" | "resource" | "workspace-deleted",
 700     force: boolean,
 701   ): Promise<void> {
 702     this.lifetime.stop();
 703     this.resourceSupervisor.stop();
 704     if (force) await this.forceWorkerShutdown();
 705     else await this.gracefullyShutdownWorker();
 706     this.operationObserver.shutdown({ kind: "shutdown", reason, force });
 707     this.logger.record({ kind: "stop", reason });
 708     try {
 709       await this.server?.close();
 710     } catch (error) {
 711       this.logger.record({
 712         kind: "failure",
 713         operation: "transport-close",
 714         failureCode: "operation-failed",
 715         errorName: DaemonLogger.errorName(error),
 716       });
 717     }
 718     await this.deliverySession.cleanupInstance();
 719     this.deliverySession.completeRetainedTraces();
 720     await this.logger.close();
 721     this.exit(0);
 722   }
 723 
 724   private async gracefullyShutdownWorker(): Promise<void> {
 725     await this.requestQueue.drain();
 726     const gracefulClose = this.workerManager.close();
 727     await Promise.race([gracefulClose, this.forceEscalated.then(() => this.forceWorkerShutdown())]);
 728   }
 729 
 730   private forceWorkerShutdown(): Promise<void> {
 731     if (this.forcedWorkerShutdown !== undefined) return this.forcedWorkerShutdown;
 732     this.requestQueue.close();
 733     this.forcedWorkerShutdown = this.workerManager
 734       .terminate()
 735       .then(() => this.requestQueue.drain());
 736     this.resolveForceEscalated();
 737     return this.forcedWorkerShutdown;
 738   }

 740   private pause(): Promise<void> {
 741     return new Promise((resolve) =>
 742       setTimeout(resolve, this.policy.values.startup.authorizationPollIntervalMs),
 743     );
 744   }

 746   private markActiveResourceInterruption(cause: DaemonWorkerReplacementCause): void {
 747     if (cause === "worker-exit") return;
 748     const activeRequest = this.requestQueue.snapshot.active;
 749     if (activeRequest !== undefined) this.resourceInterruptedRequests.add(activeRequest.requestId);
 750   }

 752   private async recoverWorkerExit(
 753     workerExit: import("./daemon-navigation-worker.js").DaemonNavigationWorkerExit,
 754   ): Promise<void> {
 755     if (this.shutdownStarted) return;
 756     this.logger.record({
 757       kind: "failure",
 758       operation: "worker-exit",
 759       failureCode: "worker-exit",
 760       errorName: DaemonLogger.errorName(
 761         workerExit.errorName === undefined ? undefined : { name: workerExit.errorName },
 762       ),
 763     });
 764     try {
 765       await this.resourceSupervisor.recover(workerExit);
 766     } catch (error) {
 767       this.logger.record({
 768         kind: "failure",
 769         operation: "worker-replacement",
 770         failureCode: "controlled-resource",
 771         errorName: DaemonLogger.errorName(error),
 772       });
 773       throw error;
 774     }
 775   }
