  49 interface DaemonWorkerGeneration {
  50   readonly worker: DaemonNavigationWorker;
  51   ready: Promise<DaemonWorkerReadyReport> | undefined;
  52 }

  54 export class DaemonWorkerGenerationManager {
  55   private currentGeneration: DaemonWorkerGeneration;
  56   private startOperation: Promise<DaemonWorkerReadyReport> | undefined;
  57   private replacementOperation: Promise<DaemonWorkerReadyReport> | undefined;
  58   private recoveryOperation: Promise<void> | undefined;
  59   private closeOperation: Promise<void> | undefined;
  60   private terminationOperation: Promise<void> | undefined;
  61   private protocolReadyGeneration: DaemonWorkerGeneration | undefined;
  62   private readinessActivated = false;
  63   private workerReady = false;
  64   private fileCount: number | undefined;
  65   private stopping = false;
  66 
  67   constructor(private readonly options: DaemonWorkerGenerationManagerOptions) {
  68     const worker = options.initialWorker ?? options.createWorker(1);
  69     this.currentGeneration = this.createGeneration(worker);
  70   }
  71 
  72   get snapshot(): DaemonWorkerGenerationSnapshot {
  73     return Object.freeze({
  74       generation: this.currentGeneration.worker.generation,
  75       ready: this.workerReady,
  76       ...(this.fileCount === undefined ? {} : { fileCount: this.fileCount }),
  77     });
  78   }
  79 
  80   start(): Promise<DaemonWorkerReadyReport> {
  81     if (this.startOperation !== undefined) return this.startOperation;
  82     const generation = this.currentGeneration;
  83     generation.ready = this.startGeneration(generation);
  84     this.startOperation = this.waitForReadyGeneration(generation).then((report) => {
  85       this.publishProtocolReady(generation, report);
  86       return report;
  87     });
  88     return this.startOperation;
  89   }
  90 
  91   activateReadiness(): void {
  92     if (this.protocolReadyGeneration !== this.currentGeneration) {
  93       throw new Error("Navigation worker protocol readiness is unavailable");
  94     }
  95     this.readinessActivated = true;
  96     this.workerReady = true;
  97   }
  98 
  99   execute(
 100     requestId: string,
 101     request: DaemonWorkerExecuteRequest,
 102     output: DaemonOutputSink,
 103   ): Promise<DaemonWorkerExecutionReport> {
 104     const generation = this.currentGeneration;
 105     const ready = generation.ready ?? this.start();
 106     return ready
 107       .then(() =>
 108         generation.worker.execute(requestId, request.commandName, request.request, output),
 109       )
 110       .then((response): DaemonWorkerExecutionReport => {
 111         if (
 112           response.kind !== "result" ||
 113           response.requestId !== requestId ||
 114           response.generation !== generation.worker.generation
 115         ) {
 116           throw new Error("Navigation worker returned an uncorrelated result");
 117         }
 118         return response;
 119       });
 120   }
 121 
 122   replace(cause: DaemonWorkerReplacementCause): Promise<DaemonWorkerReadyReport> {
 123     if (this.replacementOperation !== undefined) return this.replacementOperation;
 124     const operation = this.replaceGeneration(cause);
 125     this.replacementOperation = operation;
 126     void operation.then(
 127       () => this.clearReplacement(operation),
 128       () => this.clearReplacement(operation),
 129     );
 130     return operation;
 131   }
 132 
 133   async releaseTransientResources(): Promise<DaemonWorkerResourceReport> {
 134     const response = await this.currentGeneration.worker.releaseTransientResources();
 135     if (response.kind !== "heap") {
 136       throw new Error("Navigation worker did not report heap usage");
 137     }
 138     this.options.onDiagnostic({
 139       kind: "resources-released",
 140       workerGeneration: response.generation,
 141       workerHeapUsedBytes: response.usedHeapBytes,
 142       workerHeapLimitBytes: response.heapLimitBytes,
 143     });
 144     return response;
 145   }
 146 
 147   close(): Promise<void> {
 148     if (this.closeOperation !== undefined) return this.closeOperation;
 149     this.stopping = true;
 150     this.workerReady = false;
 151     this.closeOperation = this.currentGeneration.worker.drainAndClose();
 152     return this.closeOperation;
 153   }
 154 
 155   terminate(): Promise<void> {
 156     if (this.terminationOperation !== undefined) return this.terminationOperation;
 157     this.stopping = true;
 158     this.workerReady = false;
 159     this.terminationOperation = this.currentGeneration.worker.terminate();
 160     return this.terminationOperation;
 161   }
 162 
 163   private async replaceGeneration(
 164     cause: DaemonWorkerReplacementCause,
 165   ): Promise<DaemonWorkerReadyReport> {
 166     const previous = this.currentGeneration;
 167     if (cause !== "worker-exit") this.options.onActiveResourceInterruption(cause);
 168     this.workerReady = false;
 169     this.protocolReadyGeneration = undefined;
 170     const nextWorker = this.options.createWorker(previous.worker.generation + 1);
 171     if (nextWorker.generation !== previous.worker.generation + 1) {
 172       throw new Error("Navigation worker factory returned the wrong generation");
 173     }
 174     const next = this.createGeneration(nextWorker);
 175     this.currentGeneration = next;
 176     next.ready = this.startGeneration(next);
 177     await previous.worker.terminate().catch(() => undefined);
 178     const report = await next.ready;
 179     this.publishProtocolReady(next, report);
 180     this.options.onDiagnostic({
 181       kind: "worker-replaced",
 182       cause,
 183       previousWorkerGeneration: previous.worker.generation,
 184       workerGeneration: next.worker.generation,
 185       fileCount: report.fileCount,
 186       ...report.startupDurations,
 187     });
 188     return report;
 189   }
 190 
 191   private startGeneration(generation: DaemonWorkerGeneration): Promise<DaemonWorkerReadyReport> {
 192     return generation.worker
 193       .start(this.options.workspaceRoot)
 194       .then((response): DaemonWorkerReadyReport => {
 195         if (response.kind !== "ready" || response.generation !== generation.worker.generation) {
 196           throw new Error("Navigation worker did not become ready");
 197         }
 198         return response;
 199       });
 200   }
 201 
 202   private async waitForReadyGeneration(
 203     generation: DaemonWorkerGeneration,
 204   ): Promise<DaemonWorkerReadyReport> {
 205     try {
 206       if (generation.ready === undefined) throw new Error("Navigation worker is not starting");
 207       return await generation.ready;
 208     } catch (error) {
 209       if (!(error instanceof DaemonNavigationWorkerExitedError)) throw error;
 210       await generation.worker.exited;
 211       const recovery = this.recoveryOperation;
 212       if (recovery === undefined) throw error;
 213       await recovery;
 214       const replacement = this.currentGeneration;
 215       if (replacement === generation || replacement.ready === undefined) throw error;
 216       return replacement.ready;
 217     }
 218   }
 219 
 220   private createGeneration(worker: DaemonNavigationWorker): DaemonWorkerGeneration {
 221     const generation: DaemonWorkerGeneration = { worker, ready: undefined };
 222     void worker.exited.then((exit) => this.observeExit(generation, exit));
 223     return generation;
 224   }
 225 
 226   private observeExit(generation: DaemonWorkerGeneration, exit: DaemonNavigationWorkerExit): void {
 227     if (this.stopping || generation !== this.currentGeneration) return;
 228     const recovery = this.options.exitRecovery.recover(exit);
 229     this.recoveryOperation = recovery;
 230     void recovery.catch(() => undefined);
 231   }
 232 
 233   private publishProtocolReady(
 234     generation: DaemonWorkerGeneration,
 235     report: DaemonWorkerReadyReport,
 236   ): void {
 237     if (generation !== this.currentGeneration) return;
 238     this.protocolReadyGeneration = generation;
 239     this.fileCount = report.fileCount;
 240     if (this.readinessActivated) this.workerReady = true;
 241   }
 242 
 243   private clearReplacement(operation: Promise<DaemonWorkerReadyReport>): void {
 244     if (this.replacementOperation === operation) this.replacementOperation = undefined;
 245   }
