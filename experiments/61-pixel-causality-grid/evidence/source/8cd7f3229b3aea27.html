 123   async attach(attachment: DaemonDeliveryAttachment, send: DaemonServerSend): Promise<void> {
 124     const entry = this.options.journal.entryFor(attachment.requestId);
 125     if (entry === undefined) throw new Error("Accepted request result is missing");
 126     const traceWasDisconnected = this.traceWasDisconnected(attachment.requestId);
 127     const disconnectTraceConnection = this.attachOperationTraceConnection(
 128       attachment.requestId,
 129       send,
 130     );
 131     if (traceWasDisconnected) this.reattachOperationTrace(attachment.requestId);
 132     try {
 133       await this.deliver(send, {
 134         kind: "accepted",
 135         ...this.options.coordinates,
 136         ...attachment,
 137       });
 138       if (entry.state.state === "completed") {
 139         await this.deliverStoredCompletion(attachment.requestId, send);
 140         return;
 141       }
 142       if (entry.state.state === "failed") {
 143         await this.deliver(send, this.failedFrame(attachment.requestId, entry.state.code));
 144         this.completeOperationTrace(attachment.requestId, "delivered");
 145         return;
 146       }
 147     } catch (error) {
 148       disconnectTraceConnection();
 149       throw error;
 150     }
 151     let unsubscribe: (() => void) | undefined;
 152     unsubscribe = this.options.journal.subscribe(attachment.requestId, (updated) => {
 153       if (updated.state.state === "completed") {
 154         this.trackCompletionDelivery(attachment.requestId, send, disconnectTraceConnection);
 155         unsubscribe?.();
 156       } else if (updated.state.state === "failed") {
 157         void this.deliver(send, this.failedFrame(attachment.requestId, updated.state.code))
 158           .then(() => this.completeOperationTrace(attachment.requestId, "delivered"))
 159           .catch((error) =>
 160             this.recordDeliveryFailure(attachment.requestId, error, disconnectTraceConnection),
 161           );
 162         unsubscribe?.();
 163       }
 164     });
 165   }

 171   async fetch(
 172     request: AuthenticatedDaemonResultFetchRequest,
 173     send: DaemonServerSend,
 174   ): Promise<void> {
 175     let disconnectTraceConnection: (() => void) | undefined;
 176     if (this.options.journal.entryFor(request.requestId)?.state.state === "completed") {
 177       const traceWasDisconnected = this.traceWasDisconnected(request.requestId);
 178       disconnectTraceConnection = this.attachOperationTraceConnection(request.requestId, send);
 179       if (traceWasDisconnected) this.reattachOperationTrace(request.requestId);
 180     }
 181     try {
 182       await this.deliverStoredCompletion(request.requestId, send, request.offset);
 183     } catch (error) {
 184       disconnectTraceConnection?.();
 185       throw error;
 186     }
 187   }

 189   async acknowledge(
 190     request: AuthenticatedDaemonResultAcknowledgement,
 191   ): Promise<DaemonResultAcknowledged> {
 192     const spool = await this.options.spoolStore.open(request.requestId);
 193     if (spool === undefined) throw new Error("Accepted request completion is unavailable");
 194     if (spool.completedManifest?.transferId !== request.transferId) {
 195       throw new Error("Result acknowledgement does not match completion transfer");
 196     }
 197     await spool.acknowledge().catch((error) => {
 198       this.recordFailure("completion-cleanup", error);
 199     });
 200     this.options.journal.acknowledge(request.requestId);
 201     this.completeOperationTrace(request.requestId, "delivered");
 202     return {
 203       kind: "result-acknowledged",
 204       ...this.options.coordinates,
 205       requestId: request.requestId,
 206       transferId: request.transferId,
 207     };
 208   }
 209 
 210   completeRetainedTraces(): void {
 211     for (const requestId of this.operationTraces.keys()) {
 212       this.completeOperationTrace(requestId, "disconnected");
 213     }
 214   }

 216   async waitForCompletionAcknowledgements(): Promise<void> {
 217     const acknowledgementDeadline =
 218       this.options.clock.wallNowMs() +
 219       this.options.policy.shutdown.resourceDrainAcknowledgementGraceMs;
 220     while (
 221       this.options.journal.hasUnacknowledgedCompletions &&
 222       this.options.clock.wallNowMs() < acknowledgementDeadline
 223     ) {
 224       await new Promise((resolve) =>
 225         setTimeout(
 226           resolve,
 227           this.options.policy.shutdown.resourceDrainAcknowledgementPollIntervalMs,
 228         ),
 229       );
 230     }
 231   }

 256   private async deliverStoredCompletion(
 257     requestId: string,
 258     send: DaemonServerSend,
 259     offset = 0,
 260   ): Promise<void> {
 261     const spool = await this.options.spoolStore.open(requestId);
 262     if (spool === undefined) throw new Error("Accepted request result is missing");
 263     try {
 264       await this.deliverCompletion(requestId, spool, offset, send);
 265     } catch (error) {
 266       if (!(error instanceof CompletionSpoolReadError)) throw error;
 267       this.recordFailure("completion-delivery", error);
 268       await spool.dispose().catch((cleanupError) => {
 269         this.recordFailure("completion-cleanup", cleanupError);
 270       });
 271       this.options.journal.invalidateCompletion(
 272         requestId,
 273         "internal",
 274         this.options.clock.wallNowMs(),
 275       );
 276       await this.deliver(send, this.failedFrame(requestId, "internal"));
 277       this.completeOperationTrace(requestId, "failed");
 278     }
 279   }

 281   private trackCompletionDelivery(
 282     requestId: string,
 283     send: DaemonServerSend,
 284     disconnectTraceConnection: () => void,
 285   ): void {
 286     const delivery = this.deliverStoredCompletion(requestId, send).catch((error) =>
 287       this.recordDeliveryFailure(requestId, error, disconnectTraceConnection),
 288     );
 289     this.completionDeliveries.set(requestId, delivery);
 290     void delivery.finally(() => {
 291       if (this.completionDeliveries.get(requestId) === delivery) {
 292         this.completionDeliveries.delete(requestId);
 293       }
 294     });
 295   }

 297   private async deliverCompletion(
 298     requestId: string,
 299     spool: CompletionSpool,
 300     offset: number,
 301     send: DaemonServerSend,
 302   ): Promise<void> {
 303     const completedManifest = spool.completedManifest;
 304     if (completedManifest === undefined) throw new Error("Completion manifest is missing");
 305     await this.deliver(send, {
 306       kind: "result-manifest",
 307       ...this.options.coordinates,
 308       requestId,
 309       manifest: completedManifest,
 310     });
 311     for await (const record of spool.read(offset)) {
 312       await this.deliver(send, {
 313         transferId: completedManifest.transferId,
 314         requestId,
 315         offset: record.sequence,
 316         sequence: record.sequence,
 317         stream: record.stream,
 318         bytes: record.bytes,
 319       });
 320     }
 321     await this.deliver(send, {
 322       kind: "result-end",
 323       ...this.options.coordinates,
 324       requestId,
 325       transferId: completedManifest.transferId,
 326       rawBytes: completedManifest.rawBytes,
 327       recordCount: completedManifest.recordCount,
 328       sha256: completedManifest.sha256,
 329     });
 330     this.terminateOperationDelivery(requestId, "delivered");
 331   }

 366   private completeOperationTrace(requestId: string, outcome: DaemonDeliveryOutcome): void {
 367     const expiration = this.operationTraceExpirations.get(requestId);
 368     if (expiration !== undefined) clearTimeout(expiration);
 369     this.operationTraceExpirations.delete(requestId);
 370     this.operationTraceConnections.delete(requestId);
 371     this.terminateOperationDelivery(requestId, outcome);
 372     this.operationTraces.delete(requestId);
 373   }
 374 
 375   private terminateOperationDelivery(requestId: string, outcome: DaemonDeliveryOutcome): void {
 376     if (!this.options.journal.terminateDelivery(requestId)) return;
 377     const trace = this.operationTraces.get(requestId);
 378     if (trace === undefined) this.options.observer.deliveryTerminated(requestId, outcome, 0);
 379     else trace.deliveryTerminated(outcome);
 380   }
 381 
 382   private disconnectOperationTrace(requestId: string): void {
 383     if (this.operationTraceExpirations.has(requestId)) return;
 384     const trace = this.operationTraces.get(requestId);
 385     if (trace === undefined) return;
 386     trace.clientDisconnected();
 387     const expiration = setTimeout(
 388       () => this.expireOperationTrace(requestId),
 389       this.options.policy.diagnostics.disconnectedTraceRetentionMs,
 390     );
 391     expiration.unref();
 392     this.operationTraceExpirations.set(requestId, expiration);
 393     this.enforceOperationTraceCapacity();
 394   }
 395 
 396   private attachOperationTraceConnection(requestId: string, send: DaemonServerSend): () => void {
 397     const connectionCount = this.operationTraceConnections.get(requestId) ?? 0;
 398     if (connectionCount === 0) {
 399       const expiration = this.operationTraceExpirations.get(requestId);
 400       if (expiration !== undefined) clearTimeout(expiration);
 401       this.operationTraceExpirations.delete(requestId);
 402     }
 403     this.operationTraceConnections.set(requestId, connectionCount + 1);
 404     let connectionClosed = false;
 405     const disconnect = (): void => {
 406       if (connectionClosed) return;
 407       connectionClosed = true;
 408       const remainingConnections = (this.operationTraceConnections.get(requestId) ?? 1) - 1;
 409       if (remainingConnections > 0) {
 410         this.operationTraceConnections.set(requestId, remainingConnections);
 411         return;
 412       }
 413       this.operationTraceConnections.delete(requestId);
 414       this.disconnectOperationTrace(requestId);
 415     };
 416     send.onClose(disconnect);
 417     return disconnect;
 418   }
 419 
 420   private traceWasDisconnected(requestId: string): boolean {
 421     if (this.operationTraceExpirations.has(requestId)) return true;
 422     return (
 423       !this.operationTraces.has(requestId) && !this.options.journal.isDeliveryTerminated(requestId)
 424     );
 425   }
 426 
 427   private reattachOperationTrace(requestId: string): void {
 428     if (this.options.journal.isDeliveryTerminated(requestId)) return;
 429     const trace = this.operationTraces.get(requestId);
 430     if (trace === undefined) this.options.observer.reattached(requestId);
 431     else trace.reattached();
 432   }
 433 
 434   private expireOperationTrace(requestId: string): void {
 435     this.operationTraceExpirations.delete(requestId);
 436     this.operationTraceConnections.delete(requestId);
 437     if (!this.operationTraces.delete(requestId)) return;
 438     this.options.observer.traceExpired(requestId);
 439   }
 440 
 441   private enforceOperationTraceCapacity(): void {
 442     const capacity = Math.max(1, this.options.policy.diagnostics.maximumDisconnectedTraces);
 443     while (this.operationTraceExpirations.size > capacity) {
 444       const oldestRequestId = this.operationTraceExpirations.keys().next().value as
 445         | string
 446         | undefined;
 447       if (oldestRequestId === undefined) return;
 448       const expiration = this.operationTraceExpirations.get(oldestRequestId);
 449       if (expiration !== undefined) clearTimeout(expiration);
 450       this.expireOperationTrace(oldestRequestId);
 451     }
 452   }

 494 class ObservedDaemonCompletionWriter implements DaemonCompletionWriter {
 495   constructor(
 496     private readonly spool: CompletionSpool,
 497     private readonly trace: DaemonOperationTrace | undefined,
 498     private readonly clock: Pick<DaemonClock, "monotonicNowMs">,
 499   ) {}
 500 
 501   append(record: DaemonSequencedOutputRecord): Promise<void> {
 502     return this.spool.append(record);
 503   }
 504 
 505   async finish(exitCode: number): Promise<CompletionSpoolManifest> {
 506     const startedAt = this.clock.monotonicNowMs();
 507     const manifest = await this.spool.finish(exitCode);
 508     const durationMs = Math.max(0, this.clock.monotonicNowMs() - startedAt);
 509     this.trace?.spooled(manifest, durationMs);
 510     return manifest;
 511   }
 512 
 513   dispose(): Promise<void> {
 514     return this.spool.dispose();
 515   }
