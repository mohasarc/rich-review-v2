  25 export class AcceptedRequestCorruptionError extends Error {

  27     super(`Accepted request identifier ${requestId} has a different payload`);

  32 export class AcceptedRequestLedger {
  33   private readonly entries = new Map<string, AcceptedRequestEntry>();
  34   private readonly subscribers = new Map<string, Set<AcceptedRequestSubscriber>>();
  35   private readonly acknowledged = new Set<string>();
  36 
  37   constructor(private readonly now: () => number = Date.now) {}
  38 
  39   get size(): number {
  40     return this.entries.size;
  41   }
  42 
  43   get hasUnacknowledgedCompletions(): boolean {
  44     for (const entry of this.entries.values()) {
  45       if (entry.state.state === "completed" && !this.acknowledged.has(entry.requestId)) return true;
  46     }
  47     return false;
  48   }
  49 
  50   accept(requestId: string, request: CliExecutionRequest): AcceptedRequestEntry {
  51     const requestFingerprint = AcceptedRequestLedger.fingerprint(request);
  52     const existing = this.entries.get(requestId);
  53     if (existing !== undefined) {
  54       if (existing.requestFingerprint !== requestFingerprint) {
  55         throw new AcceptedRequestCorruptionError(requestId);
  56       }
  57       return existing;
  58     }
  59     const entry: AcceptedRequestEntry = {
  60       requestId,
  61       requestFingerprint,
  62       request,
  63       deliveryTerminated: false,
  64       state: {
  65         state: "queued",
  66         acceptedAt: this.now(),
  67         queuePosition: this.nonterminalCount,
  68       },
  69     };
  70     this.entries.set(requestId, entry);
  71     return entry;
  72   }
  73 
  74   markRunning(requestId: string, startedAt: number): AcceptedRequestEntry {
  75     const entry = this.transitionable(requestId);
  76     if (entry.state.state !== "queued") {
  77       throw new Error(`Accepted request ${requestId} is already ${entry.state.state}`);
  78     }
  79     return this.publish({
  80       ...entry,
  81       state: { state: "running", acceptedAt: entry.state.acceptedAt, startedAt },
  82     });
  83   }
  84 
  85   complete(requestId: string, resultId: string, completedAt: number): AcceptedRequestEntry {
  86     const entry = this.transitionable(requestId);
  87     return this.publish({
  88       ...entry,
  89       state: { state: "completed", completedAt, resultId },
  90     });
  91   }
  92 
  93   fail(
  94     requestId: string,
  95     code: DaemonExecutionFailureCode,
  96     completedAt: number,
  97   ): AcceptedRequestEntry {
  98     const entry = this.transitionable(requestId);
  99     return this.publish({
 100       ...entry,
 101       state: { state: "failed", completedAt, code },
 102     });
 103   }
 104 
 105   invalidateCompletion(
 106     requestId: string,
 107     code: DaemonExecutionFailureCode,
 108     completedAt: number,
 109   ): AcceptedRequestEntry {
 110     const entry = this.entry(requestId);
 111     if (entry.state.state === "failed" && entry.state.code === code) return entry;
 112     if (entry.state.state !== "completed") {
 113       throw new Error(`Accepted request ${requestId} is not completed`);
 114     }
 115     return this.publish({
 116       ...entry,
 117       state: { state: "failed", completedAt, code },
 118     });
 119   }
 120 
 121   status(requestId: string): DaemonExecutionStatus {
 122     const state = this.entries.get(requestId)?.state;
 123     if (state === undefined) return { state: "unknown" };
 124     if (state.state === "queued") {
 125       return { state: "queued", queuePosition: state.queuePosition };
 126     }
 127     if (state.state === "running") return { state: "running", startedAt: state.startedAt };
 128     if (state.state === "completed") return { state: "completed" };
 129     return { state: "failed", code: state.code };
 130   }
 131 
 132   acknowledge(requestId: string): void {
 133     const entry = this.entry(requestId);
 134     if (entry.state.state !== "completed" && entry.state.state !== "failed") {
 135       throw new Error(`Accepted request ${requestId} is not terminal`);
 136     }
 137     this.acknowledged.add(requestId);
 138   }
 139 
 140   isAcknowledged(requestId: string): boolean {
 141     return this.acknowledged.has(requestId);
 142   }
 143 
 144   terminateDelivery(requestId: string): boolean {
 145     const entry = this.entry(requestId);
 146     if (entry.deliveryTerminated) return false;
 147     this.entries.set(requestId, { ...entry, deliveryTerminated: true });
 148     return true;
 149   }
 150 
 151   isDeliveryTerminated(requestId: string): boolean {
 152     return this.entry(requestId).deliveryTerminated;
 153   }
 154 
 155   subscribe(requestId: string, subscriber: AcceptedRequestSubscriber): () => void {
 156     const entry = this.entry(requestId);
 157     let requestSubscribers = this.subscribers.get(requestId);
 158     if (requestSubscribers === undefined) {
 159       requestSubscribers = new Set();
 160       this.subscribers.set(requestId, requestSubscribers);
 161     }
 162     requestSubscribers.add(subscriber);
 163     subscriber(entry);
 164     return () => {
 165       requestSubscribers?.delete(subscriber);
 166       if (requestSubscribers?.size === 0) this.subscribers.delete(requestId);
 167     };
 168   }
 169 
 170   entryFor(requestId: string): AcceptedRequestEntry | undefined {
 171     return this.entries.get(requestId);
 172   }
 173 
 174   private get nonterminalCount(): number {
 175     let count = 0;
 176     for (const entry of this.entries.values()) {
 177       if (entry.state.state === "queued" || entry.state.state === "running") count += 1;
 178     }
 179     return count;
 180   }
 181 
 182   private transitionable(requestId: string): AcceptedRequestEntry {
 183     const entry = this.entry(requestId);
 184     if (entry.state.state === "completed" || entry.state.state === "failed") {
 185       throw new Error(`Accepted request ${requestId} is already ${entry.state.state}`);
 186     }
 187     return entry;
 188   }
 189 
 190   private entry(requestId: string): AcceptedRequestEntry {
 191     const entry = this.entries.get(requestId);
 192     if (entry === undefined) throw new Error(`Accepted request ${requestId} was not accepted`);
 193     return entry;
 194   }
 195 
 196   private publish(entry: AcceptedRequestEntry): AcceptedRequestEntry {
 197     this.entries.set(entry.requestId, entry);
 198     for (const subscriber of this.subscribers.get(entry.requestId) ?? []) subscriber(entry);
 199     return entry;
 200   }

 202   private static fingerprint(request: CliExecutionRequest): string {
 203     const canonical = AcceptedRequestLedger.canonicalValue(request);
 204     return createHash("sha256").update(JSON.stringify(canonical)).digest("hex");
 205   }
 206 
 207   private static canonicalValue(value: unknown): unknown {
 208     if (Array.isArray(value))
 209       return value.map((item) => AcceptedRequestLedger.canonicalValue(item));
 210     if (typeof value !== "object" || value === null) return value;
 211     return Object.fromEntries(
 212       Object.entries(value)
 213         .filter(([, item]) => item !== undefined)
 214         .sort(([left], [right]) => left.localeCompare(right))
 215         .map(([key, item]) => [key, AcceptedRequestLedger.canonicalValue(item)]),
 216     );
 217   }
