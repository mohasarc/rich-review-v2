 269   private async waitForStartupAuthorization(): Promise<{
 270     readonly lease: DaemonStartupLease;
 271     readonly record: DaemonRecord;
 272   }> {
 273     const deadline = this.now() + DAEMON_STARTUP_TIMEOUT_MS;
 274     while (this.now() <= deadline) {
 275       const record = this.options.registry.readInstance(
 276         this.options.identity,
 277         this.options.instanceId,
 278       );
 279       if (
 280         record?.state === "starting" &&
 281         (record.pid === 0 || record.pid === process.pid) &&
 282         record.processToken === this.options.processToken
 283       ) {
 284         const lease = this.options.registry.claimStartupForDaemon(
 285           this.options.identity,
 286           this.options.instanceId,
 287           this.options.processToken,
 288           process.pid,
 289         );
 290         if (lease === undefined) {
 291           await this.pause();
 292           continue;
 293         }
 294         this.options.registry.writeStartingIfStartupOwner(this.options.identity, {
 295           ...record,
 296           pid: process.pid,
 297         });
 298         const adoptedRecord = this.options.registry.readInstance(
 299           this.options.identity,
 300           this.options.instanceId,
 301         );
 302         if (adoptedRecord?.pid !== process.pid) {
 303           lease.release();
 304           await this.pause();
 305           continue;
 306         }
 307         if (
 308           this.options.registry.startupOwnerMatchesProcess(this.options.identity, adoptedRecord)
 309         ) {
 310           return { lease, record: adoptedRecord };
 311         }
 312         lease.release();
 313       }
 314       await this.pause();
 315     }
 316     throw new Error("Daemon process did not receive startup authorization");
 317   }

 335   private async handle(
 336     request: DaemonRequest,
 337     send: DaemonServerSend,
 338   ): Promise<DaemonResponse | void> {
 339     if (request.kind === "identify") return this.identify(request);
 340     if (request.kind === "terminate" || request.kind === "kill") {
 341       return this.terminate(request);
 342     }
 343     if (
 344       request.protocolVersion !== DAEMON_PROTOCOL_VERSION ||
 345       request.instanceId !== this.options.instanceId
 346     ) {
 347       throw new Error("Daemon request does not match protocol or instance");
 348     }
 349     if (request.kind === "ping") return this.pong();
 350     if (
 351       request.kind === "execute" ||
 352       request.kind === "execution-status" ||
 353       request.kind === "result-fetch" ||
 354       request.kind === "result-ack"
 355     ) {
 356       if (request.processToken !== this.options.processToken) {
 357         throw new Error("Daemon execution request does not match process instance");
 358       }
 359     }
 360     if (request.kind === "execute") return this.acceptExecution(request, send);
 361     if (request.kind === "result-fetch") {
 362       let disconnectTraceConnection: (() => void) | undefined;
 363       if (this.acceptedRequests.entryFor(request.requestId)?.state.state === "completed") {
 364         const traceWasDisconnected = !this.operationTraceConnections.has(request.requestId);
 365         disconnectTraceConnection = this.attachOperationTraceConnection(request.requestId, send);
 366         if (traceWasDisconnected) this.reattachOperationTrace(request.requestId);
 367       }
 368       try {
 369         await this.deliverStoredCompletion(request.requestId, send, request.offset);
 370       } catch (error) {
 371         disconnectTraceConnection?.();
 372         throw error;
 373       }
 374       return;
 375     }
 376     if (request.kind === "result-ack") {
 377       const spool = await this.completionSpools.open(request.requestId);
 378       if (spool === undefined) throw new Error("Accepted request completion is unavailable");
 379       if (spool.completedManifest?.transferId !== request.transferId) {
 380         throw new Error("Result acknowledgement does not match completion transfer");
 381       }
 382       await spool.acknowledge().catch((error) => {
 383         this.logger.record({
 384           kind: "failure",
 385           operation: "completion-cleanup",
 386           failureCode: "internal",
 387           errorName: DaemonLogger.errorName(error),
 388         });
 389       });
 390       this.acceptedRequests.acknowledge(request.requestId);
 391       this.completeOperationTrace(request.requestId, "delivered");
 392       return {
 393         kind: "result-acknowledged",
 394         instanceId: this.options.instanceId,
 395         processToken: this.options.processToken,
 396         requestId: request.requestId,
 397         transferId: request.transferId,
 398       };
 399     }
 400     if (request.kind === "execution-status") {
 401       return {
 402         kind: "execution-status",
 403         instanceId: this.options.instanceId,
 404         processToken: this.options.processToken,
 405         requestId: request.requestId,
 406         status: this.acceptedRequests.status(request.requestId),
 407       };
 408     }
 409     this.beginGracefulShutdown();
 410     await this.requestQueue.drain();
 411     await this.waitForCompletionAcknowledgements();
 412     setTimeout(() => void this.shutdown("graceful"), 0);
 413     return { kind: "stopped", instanceId: this.options.instanceId };
 414   }

 485   private async acceptExecution(
 486     request: Extract<DaemonRequest, { kind: "execute" }>,
 487     send: DaemonServerSend,
 488   ): Promise<DaemonResponse | void> {
 489     void this.resourceSupervisor.sample("admission").catch((error) => {
 490       this.logger.record({
 491         kind: "failure",
 492         operation: "resource-sample",
 493         failureCode: "operation-failed",
 494         errorName: DaemonLogger.errorName(error),
 495       });
 496     });
 497     if (!this.workerReady) return this.rejection(request, "not-ready", true);
 498     if (this.resourceSupervisor.snapshot.admissionPaused) {
 499       return this.rejection(request, "resource-pressure", true);
 500     }
 501     if (this.requestQueue.state !== "accepting") {
 502       return this.rejection(request, "draining", true);
 503     }
 504     const existing = this.acceptedRequests.entryFor(request.requestId);
 505     let entry;
 506     try {
 507       entry = this.acceptedRequests.accept(request.requestId, request.request);
 508     } catch (error) {
 509       if (error instanceof AcceptedRequestCorruptionError) {
 510         return this.rejection(request, "incompatible", false);
 511       }
 512       throw error;
 513     }
 514     if (entry.state.state === "queued") {
 515       this.acceptances.set(request.requestId, {
 516         acceptedAt: entry.state.acceptedAt,
 517         queuePosition: entry.state.queuePosition,
 518       });
 519       if (existing === undefined) {
 520         const trace = this.operationObserver.start(
 521           request.requestId,
 522           WorkspaceDaemon.commandName(request.request.argv),
 523         );
 524         this.operationTraces.set(request.requestId, trace);
 525         trace.accepted(entry.state.queuePosition, this.resourceSupervisor.snapshot.generation);
 526       }
 527     }
 528     const acceptance = this.acceptances.get(request.requestId);
 529     if (acceptance === undefined) throw new Error("Accepted request is missing admission metadata");
 530     const traceWasDisconnected = !this.operationTraceConnections.has(request.requestId);
 531     const disconnectTraceConnection = this.attachOperationTraceConnection(request.requestId, send);
 532     if (existing === undefined) {
 533       this.lastNavigationAt = this.now();
 534       this.lifetime.navigationAccepted();
 535       void this.executeAccepted(request);
 536     } else if (traceWasDisconnected) {
 537       this.reattachOperationTrace(request.requestId);
 538     }
 539     try {
 540       await this.deliver(send, {
 541         kind: "accepted",
 542         instanceId: this.options.instanceId,
 543         processToken: this.options.processToken,
 544         requestId: request.requestId,
 545         ...acceptance,
 546       });
 547     } catch (error) {
 548       disconnectTraceConnection();
 549       throw error;
 550     }
 551     if (entry.state.state === "completed") {
 552       try {
 553         await this.deliverStoredCompletion(request.requestId, send);
 554       } catch (error) {
 555         disconnectTraceConnection();
 556         throw error;
 557       }
 558       return;
 559     }
 560     if (entry.state.state === "failed") {
 561       try {
 562         await this.deliver(send, this.failedFrame(request.requestId, entry.state.code));
 563       } catch (error) {
 564         disconnectTraceConnection();
 565         throw error;
 566       }
 567       this.completeOperationTrace(request.requestId, "delivered");
 568       return;
 569     }
 570     let unsubscribe: (() => void) | undefined;
 571     unsubscribe = this.acceptedRequests.subscribe(request.requestId, (updated) => {
 572       if (updated.state.state === "completed") {
 573         this.trackCompletionDelivery(request.requestId, send, disconnectTraceConnection);
 574         unsubscribe?.();
 575       } else if (updated.state.state === "failed") {
 576         void this.deliver(send, this.failedFrame(request.requestId, updated.state.code))
 577           .then(() => this.completeOperationTrace(request.requestId, "delivered"))
 578           .catch((error) =>
 579             this.recordDeliveryFailure(request.requestId, error, disconnectTraceConnection),
 580           );
 581         unsubscribe?.();
 582       }
 583     });
 584   }

 586   private async executeAccepted(
 587     request: Extract<DaemonRequest, { kind: "execute" }>,
 588   ): Promise<void> {
 589     let spool: CompletionSpool | undefined;
 590     try {
 591       await this.requestQueue.enqueue(
 592         {
 593           requestId: request.requestId,
 594           command: WorkspaceDaemon.commandName(request.request.argv),
 595           acceptedAt: this.clock.monotonicNowMs(),
 596         },
 597         async () => {
 598           this.operationTraces
 599             .get(request.requestId)
 600             ?.turnStarted(this.resourceSupervisor.snapshot.generation);
 601           try {
 602             spool = await this.completionSpools.create(request.requestId);
 603             this.acceptedRequests.markRunning(request.requestId, this.now());
 604             const generation = this.workerGeneration;
 605             if (generation === undefined) throw new Error("Navigation worker is unavailable");
 606             const ready = await generation.ready;
 607             if (ready.kind !== "ready") throw new Error("Navigation worker did not become ready");
 608             const response = await generation.worker.execute(
 609               request.requestId,
 610               request.request,
 611               spool,
 612             );
 613             if (response.kind !== "result" || response.requestId !== request.requestId) {
 614               throw new Error("Navigation worker returned an uncorrelated result");
 615             }
 616             this.resourceSupervisor.workerHeapReported(
 617               response.generation,
 618               response.resources.workerHeapUsedBytes,
 619               response.resources.workerHeapLimitBytes,
 620               response.resources.peakWorkerHeapUsedBytes,
 621             );
 622             this.operationTraces.get(request.requestId)?.workerCompleted(
 623               {
 624                 freshnessMs: response.durations.freshnessMs,
 625                 navigationMs: response.durations.navigationMs,
 626                 renderMs: response.durations.renderMs,
 627                 workerOutputMs: response.durations.outputMs,
 628               },
 629               response.refresh,
 630             );
 631             const spoolStartedAt = this.clock.monotonicNowMs();
 632             const manifest = await spool.finish(response.result.exitCode);
 633             this.operationTraces
 634               .get(request.requestId)
 635               ?.spooled(manifest, Math.max(0, this.clock.monotonicNowMs() - spoolStartedAt));
 636             const workspaceDeleted = await this.recordCompletion();
 637             this.operationTraces.get(request.requestId)?.executionTerminated("completed");
 638             this.acceptedRequests.complete(request.requestId, request.requestId, this.now());
 639             await this.completionDeliveries.get(request.requestId);
 640             if (workspaceDeleted) {
 641               await this.waitForCompletionAcknowledgements();
 642               setTimeout(() => void this.shutdown("workspace-deleted", true), 0);
 643             }
 644           } finally {
 645             this.scheduleTurnCompleteResourceSample();
 646           }
 647         },
 648       );
 649     } catch (error) {
 650       this.operationTraces.get(request.requestId)?.executionTerminated("failed");
 651       this.logger.record({
 652         kind: "failure",
 653         operation: "request",
 654         failureCode: "internal",
 655         errorName: DaemonLogger.errorName(error),
 656       });
 657       const code =
 658         (this.resourceInterruptedRequests.delete(request.requestId)
 659           ? "controlled-resource"
 660           : undefined) ??
 661         (error instanceof CompletionSpoolCapacityError
 662           ? "response-capacity"
 663           : error instanceof DaemonNavigationWorkerExitedError
 664             ? this.shutdownFailureCode === "stopping"
 665               ? "stopping"
 666               : "worker-exit"
 667             : (this.shutdownFailureCode ?? (this.shutdownStarted ? "stopping" : "internal")));
 668       await spool?.dispose().catch((cleanupError) => {
 669         this.logger.record({
 670           kind: "failure",
 671           operation: "completion-cleanup",
 672           failureCode: "internal",
 673           errorName: DaemonLogger.errorName(cleanupError),
 674         });
 675       });
 676       this.acceptedRequests.fail(request.requestId, code, this.now());
 677     } finally {
 678       if (this.requestQueue.isIdle) this.lifetime.queueBecameIdle();
 679     }
 680   }

 683     void this.requestQueue

 692       })
 693       .finally(() => {

 695       });
 696   }

 735   private async deliverStoredCompletion(
 736     requestId: string,
 737     send: DaemonServerSend,
 738     offset = 0,
 739   ): Promise<void> {
 740     const spool = await this.completionSpools.open(requestId);
 741     if (spool === undefined) throw new Error("Accepted request result is missing");
 742     try {
 743       await this.deliverCompletion(requestId, spool, offset, send);
 744     } catch (error) {
 745       if (!(error instanceof CompletionSpoolReadError)) throw error;
 746       this.logger.record({
 747         kind: "failure",
 748         operation: "completion-delivery",
 749         failureCode: "internal",
 750         errorName: DaemonLogger.errorName(error),
 751       });
 752       await spool.dispose().catch((cleanupError) => {
 753         this.logger.record({
 754           kind: "failure",
 755           operation: "completion-cleanup",
 756           failureCode: "internal",
 757           errorName: DaemonLogger.errorName(cleanupError),
 758         });
 759       });
 760       this.acceptedRequests.invalidateCompletion(requestId, "internal", this.now());
 761       await this.deliver(send, this.failedFrame(requestId, "internal"));
 762       this.completeOperationTrace(requestId, "failed");
 763     }
 764   }

 766   private trackCompletionDelivery(
 767     requestId: string,
 768     send: DaemonServerSend,
 769     disconnectTraceConnection: () => void,
 770   ): void {

 772       this.recordDeliveryFailure(requestId, error, disconnectTraceConnection),

 777         this.completionDeliveries.delete(requestId);
 778       }
 779     });

 782   private async deliverCompletion(
 783     requestId: string,
 784     spool: CompletionSpool,
 785     offset: number,
 786     send: DaemonServerSend,
 787   ): Promise<void> {
 788     const completedManifest = spool.completedManifest;
 789     if (completedManifest === undefined) throw new Error("Completion manifest is missing");
 790     await this.deliver(send, {
 791       kind: "result-manifest",
 792       instanceId: this.options.instanceId,
 793       processToken: this.options.processToken,
 794       requestId,
 795       manifest: completedManifest,
 796     });
 797     for await (const record of spool.read(offset)) {
 798       await this.deliver(send, {
 799         transferId: completedManifest.transferId,
 800         requestId,
 801         offset: record.sequence,
 802         sequence: record.sequence,
 803         stream: record.stream,
 804         bytes: record.bytes,
 805       });
 806     }
 807     await this.deliver(send, {
 808       kind: "result-end",
 809       instanceId: this.options.instanceId,
 810       processToken: this.options.processToken,
 811       requestId,
 812       transferId: completedManifest.transferId,
 813       rawBytes: completedManifest.rawBytes,
 814       recordCount: completedManifest.recordCount,
 815       sha256: completedManifest.sha256,
 816     });
 817     this.terminateOperationDelivery(requestId, "delivered");
 818   }

 856   private disconnectOperationTrace(requestId: string): void {
 857     if (this.operationTraceExpirations.has(requestId)) return;
 858     const trace = this.operationTraces.get(requestId);
 859     if (trace === undefined) return;
 860     trace.clientDisconnected();
 861     const expiration = setTimeout(
 862       () => this.expireOperationTrace(requestId),
 863       this.options.operationTraceRetentionMs ?? DEFAULT_OPERATION_TRACE_RETENTION_MS,
 864     );
 865     expiration.unref();
 866     this.operationTraceExpirations.set(requestId, expiration);
 867     this.enforceOperationTraceCapacity();
 868   }

 870   private attachOperationTraceConnection(requestId: string, send: DaemonServerSend): () => void {
 871     const connectionCount = this.operationTraceConnections.get(requestId) ?? 0;
 872     if (connectionCount === 0) {
 873       const expiration = this.operationTraceExpirations.get(requestId);
 874       if (expiration !== undefined) clearTimeout(expiration);
 875       this.operationTraceExpirations.delete(requestId);
 876     }
 877     this.operationTraceConnections.set(requestId, connectionCount + 1);
 878     let connectionClosed = false;
 879     const disconnect = (): void => {
 880       if (connectionClosed) return;
 881       connectionClosed = true;
 882       const remainingConnections = (this.operationTraceConnections.get(requestId) ?? 1) - 1;
 883       if (remainingConnections > 0) {
 884         this.operationTraceConnections.set(requestId, remainingConnections);
 885         return;
 886       }
 887       this.operationTraceConnections.delete(requestId);
 888       this.disconnectOperationTrace(requestId);
 889     };
 890     send.onClose(disconnect);
 891     return disconnect;
 892   }

 896     const trace = this.operationTraces.get(requestId);

 902     this.operationTraceExpirations.delete(requestId);

 908   private enforceOperationTraceCapacity(): void {
 909     const capacity = Math.max(
 910       1,
 911       this.options.maximumRetainedOperationTraces ?? DEFAULT_MAXIMUM_RETAINED_OPERATION_TRACES,
 912     );
 913     while (this.operationTraceExpirations.size > capacity) {
 914       const oldestRequestId = this.operationTraceExpirations.keys().next().value as
 915         | string
 916         | undefined;
 917       if (oldestRequestId === undefined) return;
 918       const expiration = this.operationTraceExpirations.get(oldestRequestId);
 919       if (expiration !== undefined) clearTimeout(expiration);
 920       this.expireOperationTrace(oldestRequestId);
 921     }

 968   private async completeShutdown(
 969     reason: "graceful" | "idle" | "resource" | "workspace-deleted",
 970     force: boolean,
 971   ): Promise<void> {
 972     this.lifetime.stop();
 973     this.resourceSupervisor.stop();
 974     if (force) await this.forceWorkerShutdown();
 975     else await this.gracefullyShutdownWorker();
 976     this.operationObserver.shutdown({ kind: "shutdown", reason, force });
 977     this.logger.record({ kind: "stop", reason });
 978     try {
 979       await this.server?.close();
 980     } catch (error) {
 981       this.logger.record({
 982         kind: "failure",
 983         operation: "transport-close",
 984         failureCode: "operation-failed",
 985         errorName: DaemonLogger.errorName(error),
 986       });
 987     }
 988     await this.completionSpools.cleanupInstance(this.options.instanceId).catch((error) => {
 989       this.logger.record({
 990         kind: "failure",
 991         operation: "completion-cleanup",
 992         failureCode: "internal",
 993         errorName: DaemonLogger.errorName(error),
 994       });
 995     });
 996     this.completeRetainedOperationTraces();
 997     await this.logger.close();
 998     this.exit(0);
 999   }

1082   private async replaceNavigationWorker(cause: DaemonWorkerReplacementCause): Promise<number> {
1083     const current = this.workerGeneration;
1084     if (current === undefined) throw new Error("Navigation worker generation is unavailable");
1085     const activeRequest = this.requestQueue.snapshot.active;
1086     if (activeRequest !== undefined && cause !== "worker-exit") {
1087       this.resourceInterruptedRequests.add(activeRequest.requestId);
1088     }
1089     const nextWorker = this.createNavigationWorker(current.id + 1);
1090     this.workerReady = false;
1091     const next = this.startWorkerGeneration(nextWorker);
1092     await current.worker.terminate().catch(() => undefined);
1093     const response = await next.ready;
1094     if (response.kind !== "ready") throw new Error("Replacement navigation worker did not start");
1095     this.fileCount = response.fileCount;
1096     this.workerReady = true;
1097     this.operationObserver.worker({
1098       kind: "worker-replaced",
1099       cause,
1100       previousWorkerGeneration: current.id,
1101       workerGeneration: next.id,
1102       fileCount: response.fileCount,
1103       ...response.startupDurations,
1104     });
1105     this.logger.record({ kind: "freshness", ...response.refresh });
1106     return next.id;
1107   }

1131   private activitySnapshot(): DaemonActivitySnapshot {
1132     const queue = this.requestQueue.snapshot;
1133     const resources = this.resourceSupervisor.snapshot;
1134     const now = this.clock.monotonicNowMs();
1135     const lifecycle: DaemonActivitySnapshot["lifecycle"] =
1136       queue.state !== "accepting" || resources.state === "draining" || resources.state === "stopped"
1137         ? "draining"
1138         : resources.state === "replacing" || resources.state === "shedding"
1139           ? "recovering"
1140           : !this.workerReady
1141             ? "starting"
1142             : queue.active === undefined
1143               ? "ready"
1144               : "busy";
1145     const current =
1146       lifecycle !== "busy" || queue.active === undefined
1147         ? undefined
1148         : Object.freeze({
1149             requestId: queue.active.requestId,
1150             command: queue.active.command,
1151             elapsedMs: Math.max(0, now - queue.active.startedAt),
1152           });
1153     const recoveryDetail: DaemonActivitySnapshot["recoveryDetail"] =
1154       resources.state === "replacing"
1155         ? "worker-replacement"
1156         : resources.state === "shedding"
1157           ? "resource-pressure"
1158           : undefined;
1159     return Object.freeze({
1160       lifecycle,
1161       ...(recoveryDetail === undefined ? {} : { recoveryDetail }),
1162       pid: process.pid,
1163       startedAt: this.startedAt,
1164       startupElapsedMs: Math.max(0, now - this.startedMonotonicAt),
1165       ...(this.workerReady ? { fileCount: this.fileCount } : {}),
1166       processRssBytes: process.memoryUsage().rss,
1167       hardProcessRssBytes: this.resourcePolicy.record.hardProcessRssBytes,
1168       ...(resources.workerHeapUsedBytes === undefined
1169         ? {}
1170         : { workerHeapUsedBytes: resources.workerHeapUsedBytes }),
1171       workerGeneration: this.workerGeneration?.id ?? resources.generation,
1172       ...(current === undefined ? {} : { current }),
1173       queued: queue.queued,
1174       ...(this.lastCompletedMonotonicAt === undefined
1175         ? {}
1176         : { lastCompletedAgoMs: Math.max(0, now - this.lastCompletedMonotonicAt) }),
1177       spoolBytes: resources.spoolBytes,
1178     });
1179   }
