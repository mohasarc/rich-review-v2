  29 function readPackageVersion(): string {
  30   const here = dirname(fileURLToPath(import.meta.url));
  31   const raw = readFileSync(join(here, "..", "package.json"), "utf8");
  32   const parsed = JSON.parse(raw) as { version: string };
  33   return parsed.version;
  34 }

  45 export function createDefaultDependencies(
  46   canonicalStateDirectory: string,
  47   daemonPolicy: DaemonPolicy,
  48   daemonEnabled = true,
  49 ): ProgramDependencies {
  50   const fs = new NodeFileSystem();
  51   const clock: Clock = { now: () => Date.now() };
  52   const symnavVersion = readPackageVersion();
  53   let dependencies: ProgramDependencies;
  54   const executorFactory: DaemonExecutorFactory = async (options) => {
  55     const { createDaemonExecutorFromDependencies } = await import("./daemon-executor.js");
  56     return createDaemonExecutorFromDependencies(options, dependencies);
  57   };
  58   const daemonClient = new DaemonClient({
  59     stateDirectory: canonicalStateDirectory,
  60     productVersion: symnavVersion,
  61     daemonEnabled,
  62     executorFactory,
  63     executorModuleUrl: new URL("./daemon-executor.js", import.meta.url).href,
  64     readinessProbe: { commandName: "version", argv: ["--version"] },
  65     policy: daemonPolicy,
  66   });
  67   dependencies = {
  68     stateDirectory: canonicalStateDirectory,
  69     daemonPolicy,
  70     daemonClient,
  71     daemonEnabled,
  72     fs,
  73     backends: () => [new TypeScriptBackend(fs)],
  74     git: new NodeGitHistory(),
  75     recorder: new NodeUsageRecorder(
  76       new NodeTelemetryWritePort(usageLogPath(canonicalStateDirectory)),
  77       { next: () => randomUUID() },
  78     ),
  79     clock,
  80     telemetryEnabled: isTelemetryEnabled(process.env),
  81     identity: new NodeTelemetryIdentityProvider(canonicalStateDirectory, new NodeGitRemoteReader()),
  82     symnavVersion,
  83   };
  84   return dependencies;
  85 }

  87 export function buildProgram(
  88   context?: ProgramContext,
  89   dependencies?: ProgramDependencies,
  90 ): CommanderCommand {
  91   const ctx = context ?? createDefaultProgramContext();
  92   const deps =
  93     dependencies ??
  94     createDefaultDependencies(
  95       new StateDirectoryResolver(process.env).resolve(),
  96       DaemonPolicy.currentSystem(),
  97     );
  98   const program = new CommanderCommand();
  99   program
 100     .name("symnav")
 101     .version(readPackageVersion(), "-v, --version")
 102     .option("--cwd <dir>", "run as if symnav was started in <dir>")
 103     .configureOutput({
 104       writeOut: (s) => {
 105         ctx.stdout.write(s);
 106       },
 107       writeErr: (s) => {
 108         ctx.stderr.write(s);
 109       },
 110     })
 111     .exitOverride((err) => {
 112       ctx.exit(err.exitCode);
 113     })
 114     .action(() => {
 115       program.outputHelp({ error: true });
 116       ctx.exit(1);
 117     });
 118   registerOverviewCommand(program, ctx, deps);
 119   registerResolveCommand(program, ctx, deps);
 120   registerDefCommand(program, ctx, deps);
 121   registerRefsCommand(program, ctx, deps);
 122   registerContextCommand(program, ctx, deps);
 123   registerGraphCommand(program, ctx, deps);
 124   registerDaemonCommand(program, ctx, deps);
 125   registerStatsCommand(program, ctx, deps);
 126   return program;
