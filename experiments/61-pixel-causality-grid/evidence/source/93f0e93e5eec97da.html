 214         kind: "startup-completed",
 215         workerGeneration: response.generation,

 218       });

 221       this.workerReady = true;

 223         schemaVersion: DAEMON_RECORD_SCHEMA_VERSION,
 224         protocolVersion: DAEMON_PROTOCOL_VERSION,

 226         workspaceRoot: this.options.identity.workspaceRoot,
 227         workspaceKey: this.options.identity.workspaceKey,
 228         stateKey: this.options.identity.stateKey,
 229         identityKey: this.options.identity.identityKey,
 230         instanceId: this.options.instanceId,
 231         processToken: this.options.processToken,
 232         endpoint: this.options.identity.endpoint(this.options.instanceId),
 233         pid: process.pid,

 235         startedAt: startingRecord.startedAt,

 239       };
 240       if (!this.options.registry.writeIfStartupOwner(this.options.identity, readyRecord)) {
 241         throw new Error("Daemon startup ownership changed before readiness publication");
 242       }

 263   private async waitForStartupAuthorization(): Promise<{
 264     readonly lease: DaemonStartupLease;
 265     readonly record: DaemonRecord;
 266   }> {
 267     const deadline = this.now() + this.policy.values.startup.coordinationGraceMs;
 268     while (this.now() <= deadline) {
 269       const record = this.options.registry.readInstance(
 270         this.options.identity,
 271         this.options.instanceId,
 272       );
 273       if (
 274         record?.state === "starting" &&
 275         (record.pid === 0 || record.pid === process.pid) &&
 276         record.processToken === this.options.processToken
 277       ) {
 278         const lease = this.options.registry.claimStartupForDaemon(
 279           this.options.identity,
 280           this.options.instanceId,
 281           this.options.processToken,
 282           process.pid,
 283         );
 284         if (lease === undefined) {
 285           await this.pause();
 286           continue;
 287         }
 288         this.options.registry.writeStartingIfStartupOwner(this.options.identity, {
 289           ...record,
 290           pid: process.pid,
 291         });
 292         const adoptedRecord = this.options.registry.readInstance(
 293           this.options.identity,
 294           this.options.instanceId,
 295         );
 296         if (adoptedRecord?.pid !== process.pid) {
 297           lease.release();
 298           await this.pause();
 299           continue;
 300         }
 301         if (
 302           this.options.registry.startupOwnerMatchesProcess(this.options.identity, adoptedRecord)
 303         ) {
 304           return { lease, record: adoptedRecord };
 305         }
 306         lease.release();
 307       }
 308       await this.pause();
 309     }
 310     throw new Error("Daemon process did not receive startup authorization");
 311   }

 329   private async handle(
 330     request: DaemonRequest,
 331     send: DaemonServerSend,
 332   ): Promise<DaemonResponse | void> {
 333     if (request.kind === "identify") return this.identify(request);
 334     if (request.kind === "terminate" || request.kind === "kill") {
 335       return this.terminate(request);
 336     }
 337     if (
 338       request.protocolVersion !== DAEMON_PROTOCOL_VERSION ||
 339       request.instanceId !== this.options.instanceId
 340     ) {
 341       throw new Error("Daemon request does not match protocol or instance");
 342     }
 343     if (request.kind === "ping") return this.pong();
 344     if (request.kind === "execute") return this.acceptExecution(request, send);
 345     if (
 346       request.kind === "execution-status" ||
 347       request.kind === "result-fetch" ||
 348       request.kind === "result-ack"
 349     ) {
 350       if (request.processToken !== this.options.processToken) {
 351         throw new Error("Daemon execution request does not match process instance");
 352       }
 353     }
 354     if (request.kind === "result-fetch") {
 355       let disconnectTraceConnection: (() => void) | undefined;
 356       if (this.acceptedRequests.entryFor(request.requestId)?.state.state === "completed") {
 357         const traceWasDisconnected = !this.operationTraceConnections.has(request.requestId);
 358         disconnectTraceConnection = this.attachOperationTraceConnection(request.requestId, send);
 359         if (traceWasDisconnected) this.reattachOperationTrace(request.requestId);
 360       }
 361       try {
 362         await this.deliverStoredCompletion(request.requestId, send, request.offset);
 363       } catch (error) {
 364         disconnectTraceConnection?.();
 365         throw error;
 366       }
 367       return;
 368     }
 369     if (request.kind === "result-ack") {
 370       const spool = await this.completionSpools.open(request.requestId);
 371       if (spool === undefined) throw new Error("Accepted request completion is unavailable");
 372       if (spool.completedManifest?.transferId !== request.transferId) {
 373         throw new Error("Result acknowledgement does not match completion transfer");
 374       }
 375       await spool.acknowledge().catch((error) => {
 376         this.logger.record({
 377           kind: "failure",
 378           operation: "completion-cleanup",
 379           failureCode: "internal",
 380           errorName: DaemonLogger.errorName(error),
 381         });
 382       });
 383       this.acceptedRequests.acknowledge(request.requestId);
 384       this.completeOperationTrace(request.requestId, "delivered");
 385       return {
 386         kind: "result-acknowledged",
 387         instanceId: this.options.instanceId,
 388         processToken: this.options.processToken,
 389         requestId: request.requestId,
 390         transferId: request.transferId,
 391       };
 392     }
 393     if (request.kind === "execution-status") {
 394       return {
 395         kind: "execution-status",
 396         instanceId: this.options.instanceId,
 397         processToken: this.options.processToken,
 398         requestId: request.requestId,
 399         status: this.acceptedRequests.status(request.requestId),
 400       };
 401     }
 402     this.beginGracefulShutdown();
 403     await this.requestQueue.drain();
 404     await this.waitForCompletionAcknowledgements();
 405     setTimeout(() => void this.shutdown("graceful"), 0);
 406     return { kind: "stopped", instanceId: this.options.instanceId };
 407   }

 478   private async acceptExecution(
 479     request: Extract<DaemonRequest, { kind: "execute" }>,
 480     send: DaemonServerSend,
 481   ): Promise<DaemonResponse | void> {
 482     const decision = this.decideAdmission(request);
 483     if (decision.kind === "disconnect") {
 484       throw new Error("Daemon execution request does not match process instance");
 485     }
 486     if (decision.kind === "reject") return this.rejection(request, decision.code);
 487     const existing = this.acceptedRequests.entryFor(request.requestId);
 488     const entry = this.acceptedRequests.accept(
 489       request.requestId,
 490       request.commandName,
 491       request.request,
 492     );
 493     if (entry.state.state === "queued") {
 494       this.acceptances.set(request.requestId, {
 495         acceptedAt: entry.state.acceptedAt,
 496         queuePosition: entry.state.queuePosition,
 497       });
 498       if (existing === undefined) {
 499         const trace = this.operationObserver.start(request.requestId, request.commandName);
 500         this.operationTraces.set(request.requestId, trace);
 501         trace.accepted(entry.state.queuePosition, this.resourceSupervisor.snapshot.generation);
 502       }
 503     }
 504     const acceptance = this.acceptances.get(request.requestId);
 505     if (acceptance === undefined) throw new Error("Accepted request is missing admission metadata");
 506     const traceWasDisconnected = !this.operationTraceConnections.has(request.requestId);
 507     const disconnectTraceConnection = this.attachOperationTraceConnection(request.requestId, send);
 508     if (existing === undefined) {
 509       this.lastNavigationAt = this.now();
 510       this.lifetime.navigationAccepted();
 511       void this.executeAccepted(request);
 512     } else if (traceWasDisconnected) {
 513       this.reattachOperationTrace(request.requestId);
 514     }
 515     try {
 516       await this.deliver(send, {
 517         kind: "accepted",
 518         instanceId: this.options.instanceId,
 519         processToken: this.options.processToken,
 520         requestId: request.requestId,
 521         ...acceptance,
 522       });
 523     } catch (error) {
 524       disconnectTraceConnection();
 525       throw error;
 526     }
 527     if (entry.state.state === "completed") {
 528       try {
 529         await this.deliverStoredCompletion(request.requestId, send);
 530       } catch (error) {
 531         disconnectTraceConnection();
 532         throw error;
 533       }
 534       return;
 535     }
 536     if (entry.state.state === "failed") {
 537       try {
 538         await this.deliver(send, this.failedFrame(request.requestId, entry.state.code));
 539       } catch (error) {
 540         disconnectTraceConnection();
 541         throw error;
 542       }
 543       this.completeOperationTrace(request.requestId, "delivered");
 544       return;
 545     }
 546     let unsubscribe: (() => void) | undefined;
 547     unsubscribe = this.acceptedRequests.subscribe(request.requestId, (updated) => {
 548       if (updated.state.state === "completed") {
 549         this.trackCompletionDelivery(request.requestId, send, disconnectTraceConnection);
 550         unsubscribe?.();
 551       } else if (updated.state.state === "failed") {
 552         void this.deliver(send, this.failedFrame(request.requestId, updated.state.code))
 553           .then(() => this.completeOperationTrace(request.requestId, "delivered"))
 554           .catch((error) =>
 555             this.recordDeliveryFailure(request.requestId, error, disconnectTraceConnection),
 556           );
 557         unsubscribe?.();
 558       }
 559     });
 560   }

 598   private async executeAccepted(
 599     request: Extract<DaemonRequest, { kind: "execute" }>,
 600   ): Promise<void> {
 601     let spool: CompletionSpool | undefined;
 602     try {
 603       await this.requestQueue.enqueue(
 604         {
 605           requestId: request.requestId,
 606           command: request.commandName,
 607           acceptedAt: this.clock.monotonicNowMs(),
 608         },
 609         async () => {
 610           this.operationTraces
 611             .get(request.requestId)
 612             ?.turnStarted(this.resourceSupervisor.snapshot.generation);
 613           try {
 614             spool = await this.completionSpools.create(request.requestId);
 615             this.acceptedRequests.markRunning(request.requestId, this.now());
 616             const generation = this.workerGeneration;
 617             if (generation === undefined) throw new Error("Navigation worker is unavailable");
 618             const ready = await generation.ready;
 619             if (ready.kind !== "ready") throw new Error("Navigation worker did not become ready");
 620             const response = await generation.worker.execute(
 621               request.requestId,
 622               request.commandName,
 623               request.request,
 624               spool,
 625             );
 626             if (response.kind !== "result" || response.requestId !== request.requestId) {
 627               throw new Error("Navigation worker returned an uncorrelated result");
 628             }
 629             this.resourceSupervisor.workerHeapReported(
 630               response.generation,
 631               response.resources.workerHeapUsedBytes,
 632               response.resources.workerHeapLimitBytes,
 633               response.resources.peakWorkerHeapUsedBytes,
 634             );
 635             this.operationTraces.get(request.requestId)?.workerCompleted(
 636               {
 637                 freshnessMs: response.durations.freshnessMs,
 638                 navigationMs: response.durations.navigationMs,
 639                 renderMs: response.durations.renderMs,
 640                 workerOutputMs: response.durations.outputMs,
 641               },
 642               response.refresh,
 643             );
 644             const spoolStartedAt = this.clock.monotonicNowMs();
 645             const manifest = await spool.finish(response.result.exitCode);
 646             this.operationTraces
 647               .get(request.requestId)
 648               ?.spooled(manifest, Math.max(0, this.clock.monotonicNowMs() - spoolStartedAt));
 649             const workspaceDeleted = await this.recordCompletion();
 650             this.operationTraces.get(request.requestId)?.executionTerminated("completed");
 651             this.acceptedRequests.complete(request.requestId, request.requestId, this.now());
 652             await this.completionDeliveries.get(request.requestId);
 653             if (workspaceDeleted) {
 654               await this.waitForCompletionAcknowledgements();
 655               setTimeout(() => void this.shutdown("workspace-deleted", true), 0);
 656             }
 657           } finally {
 658             this.scheduleTurnCompleteResourceSample();
 659           }
 660         },
 661       );
 662     } catch (error) {
 663       this.operationTraces.get(request.requestId)?.executionTerminated("failed");
 664       this.logger.record({
 665         kind: "failure",
 666         operation: "request",
 667         failureCode: "internal",
 668         errorName: DaemonLogger.errorName(error),
 669       });
 670       const code = DaemonExecutionFailures.classify({
 671         resourceInterrupted: this.resourceInterruptedRequests.delete(request.requestId),
 672         responseCapacityExceeded: error instanceof CompletionSpoolCapacityError,
 673         workerExited: error instanceof DaemonNavigationWorkerExitedError,
 674         ...(this.shutdownFailureCode === undefined
 675           ? {}
 676           : { shutdownFailureCode: this.shutdownFailureCode }),
 677         shutdownStarted: this.shutdownStarted,
 678       });
 679       await spool?.dispose().catch((cleanupError) => {
 680         this.logger.record({
 681           kind: "failure",
 682           operation: "completion-cleanup",
 683           failureCode: "internal",
 684           errorName: DaemonLogger.errorName(cleanupError),
 685         });
 686       });
 687       this.acceptedRequests.fail(request.requestId, code, this.now());
 688     } finally {
 689       if (this.requestQueue.isIdle) this.lifetime.queueBecameIdle();
 690     }
 691   }

 694     void this.requestQueue

 703       })
 704       .finally(() => {
 705         if (this.requestQueue.isIdle) this.lifetime.queueBecameIdle();
 706       });
 707   }

 754   private async deliverStoredCompletion(
 755     requestId: string,
 756     send: DaemonServerSend,
 757     offset = 0,
 758   ): Promise<void> {
 759     const spool = await this.completionSpools.open(requestId);
 760     if (spool === undefined) throw new Error("Accepted request result is missing");
 761     try {
 762       await this.deliverCompletion(requestId, spool, offset, send);
 763     } catch (error) {
 764       if (!(error instanceof CompletionSpoolReadError)) throw error;
 765       this.logger.record({
 766         kind: "failure",
 767         operation: "completion-delivery",
 768         failureCode: "internal",
 769         errorName: DaemonLogger.errorName(error),
 770       });
 771       await spool.dispose().catch((cleanupError) => {
 772         this.logger.record({
 773           kind: "failure",
 774           operation: "completion-cleanup",
 775           failureCode: "internal",
 776           errorName: DaemonLogger.errorName(cleanupError),
 777         });
 778       });
 779       this.acceptedRequests.invalidateCompletion(requestId, "internal", this.now());
 780       await this.deliver(send, this.failedFrame(requestId, "internal"));
 781       this.completeOperationTrace(requestId, "failed");
 782     }
 783   }

 785   private trackCompletionDelivery(
 786     requestId: string,
 787     send: DaemonServerSend,
 788     disconnectTraceConnection: () => void,
 789   ): void {
 790     const delivery = this.deliverStoredCompletion(requestId, send).catch((error) =>
 791       this.recordDeliveryFailure(requestId, error, disconnectTraceConnection),
 792     );
 793     this.completionDeliveries.set(requestId, delivery);
 794     void delivery.finally(() => {
 795       if (this.completionDeliveries.get(requestId) === delivery) {
 796         this.completionDeliveries.delete(requestId);
 797       }
 798     });
 799   }

 801   private async deliverCompletion(
 802     requestId: string,
 803     spool: CompletionSpool,
 804     offset: number,
 805     send: DaemonServerSend,
 806   ): Promise<void> {
 807     const completedManifest = spool.completedManifest;
 808     if (completedManifest === undefined) throw new Error("Completion manifest is missing");
 809     await this.deliver(send, {
 810       kind: "result-manifest",
 811       instanceId: this.options.instanceId,
 812       processToken: this.options.processToken,
 813       requestId,
 814       manifest: completedManifest,
 815     });
 816     for await (const record of spool.read(offset)) {
 817       await this.deliver(send, {
 818         transferId: completedManifest.transferId,
 819         requestId,
 820         offset: record.sequence,
 821         sequence: record.sequence,
 822         stream: record.stream,
 823         bytes: record.bytes,
 824       });
 825     }
 826     await this.deliver(send, {
 827       kind: "result-end",
 828       instanceId: this.options.instanceId,
 829       processToken: this.options.processToken,
 830       requestId,
 831       transferId: completedManifest.transferId,
 832       rawBytes: completedManifest.rawBytes,
 833       recordCount: completedManifest.recordCount,
 834       sha256: completedManifest.sha256,
 835     });
 836     this.terminateOperationDelivery(requestId, "delivered");
 837   }

 853   private completeOperationTrace(requestId: string, outcome: DaemonDeliveryOutcome): void {
 854     const expiration = this.operationTraceExpirations.get(requestId);
 855     if (expiration !== undefined) clearTimeout(expiration);
 856     this.operationTraceExpirations.delete(requestId);
 857     this.operationTraceConnections.delete(requestId);
 858     this.terminateOperationDelivery(requestId, outcome);
 859     this.operationTraces.delete(requestId);
 860   }

 862   private terminateOperationDelivery(requestId: string, outcome: DaemonDeliveryOutcome): void {
 863     if (!this.acceptedRequests.terminateDelivery(requestId)) return;
 864     const trace = this.operationTraces.get(requestId);
 865     if (trace === undefined) this.operationObserver.deliveryTerminated(requestId, outcome, 0);
 866     else trace.deliveryTerminated(outcome);
 867   }

 869   private completeRetainedOperationTraces(): void {
 870     for (const requestId of this.operationTraces.keys()) {
 871       this.completeOperationTrace(requestId, "disconnected");
 872     }
 873   }

 875   private disconnectOperationTrace(requestId: string): void {
 876     if (this.operationTraceExpirations.has(requestId)) return;
 877     const trace = this.operationTraces.get(requestId);
 878     if (trace === undefined) return;
 879     trace.clientDisconnected();
 880     const expiration = setTimeout(
 881       () => this.expireOperationTrace(requestId),
 882       this.policy.values.diagnostics.disconnectedTraceRetentionMs,
 883     );
 884     expiration.unref();
 885     this.operationTraceExpirations.set(requestId, expiration);
 886     this.enforceOperationTraceCapacity();
 887   }

 889   private attachOperationTraceConnection(requestId: string, send: DaemonServerSend): () => void {
 890     const connectionCount = this.operationTraceConnections.get(requestId) ?? 0;
 891     if (connectionCount === 0) {
 892       const expiration = this.operationTraceExpirations.get(requestId);
 893       if (expiration !== undefined) clearTimeout(expiration);
 894       this.operationTraceExpirations.delete(requestId);
 895     }
 896     this.operationTraceConnections.set(requestId, connectionCount + 1);
 897     let connectionClosed = false;
 898     const disconnect = (): void => {
 899       if (connectionClosed) return;
 900       connectionClosed = true;
 901       const remainingConnections = (this.operationTraceConnections.get(requestId) ?? 1) - 1;
 902       if (remainingConnections > 0) {
 903         this.operationTraceConnections.set(requestId, remainingConnections);
 904         return;
 905       }
 906       this.operationTraceConnections.delete(requestId);
 907       this.disconnectOperationTrace(requestId);
 908     };
 909     send.onClose(disconnect);
 910     return disconnect;
 911   }

 913   private reattachOperationTrace(requestId: string): void {
 914     if (this.acceptedRequests.isDeliveryTerminated(requestId)) return;
 915     const trace = this.operationTraces.get(requestId);
 916     if (trace === undefined) this.operationObserver.reattached(requestId);
 917     else trace.reattached();
 918   }

 920   private expireOperationTrace(requestId: string): void {
 921     this.operationTraceExpirations.delete(requestId);
 922     this.operationTraceConnections.delete(requestId);
 923     if (!this.operationTraces.delete(requestId)) return;
 924     this.operationObserver.traceExpired(requestId);
 925   }

 927   private enforceOperationTraceCapacity(): void {
 928     const capacity = Math.max(1, this.policy.values.diagnostics.maximumDisconnectedTraces);
 929     while (this.operationTraceExpirations.size > capacity) {
 930       const oldestRequestId = this.operationTraceExpirations.keys().next().value as
 931         | string
 932         | undefined;
 933       if (oldestRequestId === undefined) return;
 934       const expiration = this.operationTraceExpirations.get(oldestRequestId);
 935       if (expiration !== undefined) clearTimeout(expiration);
 936       this.expireOperationTrace(oldestRequestId);
 937     }
 938   }

 975   private async shutdown(
 976     reason: "graceful" | "idle" | "resource" | "workspace-deleted",
 977     force = false,
 978   ): Promise<void> {
 979     this.shutdownFailureCode ??= reason === "resource" ? "controlled-resource" : "stopping";
 980     if (force) this.forceWorkerShutdown();
 981     if (this.shutdownOperation !== undefined) return this.shutdownOperation;
 982     this.shutdownStarted = true;
 983     this.shutdownOperation = this.completeShutdown(reason, force);
 984     return this.shutdownOperation;
 985   }

 987   private async completeShutdown(
 988     reason: "graceful" | "idle" | "resource" | "workspace-deleted",
 989     force: boolean,
 990   ): Promise<void> {
 991     this.lifetime.stop();
 992     this.resourceSupervisor.stop();
 993     if (force) await this.forceWorkerShutdown();
 994     else await this.gracefullyShutdownWorker();
 995     this.operationObserver.shutdown({ kind: "shutdown", reason, force });
 996     this.logger.record({ kind: "stop", reason });
 997     try {
 998       await this.server?.close();
 999     } catch (error) {
1000       this.logger.record({
1001         kind: "failure",
1002         operation: "transport-close",
1003         failureCode: "operation-failed",
1004         errorName: DaemonLogger.errorName(error),
1005       });
1006     }
1007     await this.completionSpools.cleanupInstance(this.options.instanceId).catch((error) => {
1008       this.logger.record({
1009         kind: "failure",
1010         operation: "completion-cleanup",
1011         failureCode: "internal",
1012         errorName: DaemonLogger.errorName(error),
1013       });
1014     });
1015     this.completeRetainedOperationTraces();
1016     await this.logger.close();
1017     this.exit(0);
1018   }

1021     await this.requestQueue.drain();

1023     await Promise.race([gracefulClose, this.forceEscalated.then(() => this.forceWorkerShutdown())]);

1087     generation: DaemonWorkerGeneration,
1088   ): Promise<DaemonNavigationWorkerResponse> {

1103   private async replaceNavigationWorker(cause: DaemonWorkerReplacementCause): Promise<number> {
1104     const current = this.workerGeneration;
1105     if (current === undefined) throw new Error("Navigation worker generation is unavailable");
1106     const activeRequest = this.requestQueue.snapshot.active;
1107     if (activeRequest !== undefined && cause !== "worker-exit") {
1108       this.resourceInterruptedRequests.add(activeRequest.requestId);
1109     }
1110     const nextWorker = this.createNavigationWorker(current.id + 1);
1111     this.workerReady = false;
1112     const next = this.startWorkerGeneration(nextWorker);
1113     await current.worker.terminate().catch(() => undefined);
1114     const response = await next.ready;
1115     if (response.kind !== "ready") throw new Error("Replacement navigation worker did not start");
1116     this.fileCount = response.fileCount;
1117     this.workerReady = true;
1118     this.operationObserver.worker({
1119       kind: "worker-replaced",
1120       cause,
1121       previousWorkerGeneration: current.id,
1122       workerGeneration: next.id,
1123       fileCount: response.fileCount,
1124       ...response.startupDurations,
1125     });
1126     this.logger.record({ kind: "freshness", ...response.refresh });
1127     return next.id;
1128   }

1152   private activitySnapshot(): DaemonActivitySnapshot {
1153     const queue = this.requestQueue.snapshot;
1154     const resources = this.resourceSupervisor.snapshot;
1155     const now = this.clock.monotonicNowMs();
1156     const lifecycle: DaemonActivitySnapshot["lifecycle"] =
1157       queue.state !== "accepting" || resources.state === "draining" || resources.state === "stopped"
1158         ? "draining"
1159         : resources.state === "replacing" || resources.state === "shedding"
1160           ? "recovering"
1161           : !this.workerReady
1162             ? "starting"
1163             : queue.active === undefined
1164               ? "ready"
1165               : "busy";
1166     const current =
1167       lifecycle !== "busy" || queue.active === undefined
1168         ? undefined
1169         : Object.freeze({
1170             requestId: queue.active.requestId,
1171             command: queue.active.command,
1172             elapsedMs: Math.max(0, now - queue.active.startedAt),
1173           });
1174     const recoveryDetail: DaemonActivitySnapshot["recoveryDetail"] =
1175       resources.state === "replacing"
1176         ? "worker-replacement"
1177         : resources.state === "shedding"
1178           ? "resource-pressure"
1179           : undefined;
1180     return Object.freeze({
1181       lifecycle,
1182       ...(recoveryDetail === undefined ? {} : { recoveryDetail }),
1183       pid: process.pid,
1184       startedAt: this.startedAt,
1185       startupElapsedMs: Math.max(0, now - this.startedMonotonicAt),
1186       ...(this.workerReady ? { fileCount: this.fileCount } : {}),
1187       processRssBytes: process.memoryUsage().rss,
1188       hardProcessRssBytes: this.resourcePolicy.hardProcessRssBytes,
1189       ...(resources.workerHeapUsedBytes === undefined
1190         ? {}
1191         : { workerHeapUsedBytes: resources.workerHeapUsedBytes }),
1192       workerGeneration: this.workerGeneration?.id ?? resources.generation,
1193       ...(current === undefined ? {} : { current }),
1194       queued: queue.queued,
1195       ...(this.lastCompletedMonotonicAt === undefined
1196         ? {}
1197         : { lastCompletedAgoMs: Math.max(0, now - this.lastCompletedMonotonicAt) }),
1198       spoolBytes: resources.spoolBytes,
1199     });
1200   }
