  61   enqueue<T>(metadata: WorkspaceQueuedRequest, execute: () => Promise<T>): Promise<T> {
  62     if (this.currentState !== "accepting") {
  63       return Promise.reject(new Error(`Workspace request queue is ${this.currentState}`));
  64     }
  65     this.admitted.push(Object.freeze({ ...metadata }));
  66     const result = new Promise<T>((resolve, reject) => {
  67       this.requestOperations.push(async () => {
  68         const admitted = this.admitted.shift();
  69         if (admitted === undefined || admitted.requestId !== metadata.requestId) {
  70           reject(new Error("Workspace request queue admission order changed"));
  71           return;
  72         }
  73         this.activeRequest = Object.freeze({ ...admitted, startedAt: this.now() });
  74         try {
  75           resolve(await execute());
  76         } catch (error) {
  77           reject(error);
  78         } finally {
  79           this.activeRequest = undefined;
  80         }
  81       });
  82     });
  83     void this.run();
  84     return result;
  85   }

  87   scheduleAtTurnBoundary(operation: () => Promise<void>): Promise<void> {
  88     if (this.scheduledBoundary !== undefined) return this.scheduledBoundary.completion;
  89     let resolve!: () => void;
  90     let reject!: (error: unknown) => void;
  91     const completion = new Promise<void>((complete, fail) => {
  92       resolve = complete;
  93       reject = fail;
  94     });

  97     return completion;
  98   }

 100   async drain(): Promise<void> {
 101     if (this.currentState !== "closed") this.currentState = "draining";
 102     await this.waitForIdle();
 103     this.currentState = "closed";
 104   }

 106   close(): void {
 107     this.currentState = "closed";

 110   private async run(): Promise<void> {
 111     if (this.running) return;
 112     this.running = true;
 113     try {
 114       while (this.scheduledBoundary !== undefined || this.requestOperations.length > 0) {
 115         if (this.scheduledBoundary !== undefined) {
 116           await this.runScheduledBoundary(this.scheduledBoundary);
 117           continue;
 118         }
 119         await this.requestOperations.shift()?.();
 120       }
 121     } finally {
 122       this.running = false;
 123       for (const resolve of this.idleWaiters.splice(0)) resolve();
 124       if (this.scheduledBoundary !== undefined || this.requestOperations.length > 0) {
 125         void this.run();
 126       }
 127     }
 128   }
 129 
 130   private async runScheduledBoundary(boundary: ScheduledBoundary): Promise<void> {
 131     try {
 132       await boundary.operation();
 133       boundary.resolve();
 134     } catch (error) {
 135       boundary.reject(error);
 136     } finally {
 137       if (this.scheduledBoundary === boundary) this.scheduledBoundary = undefined;
 138     }
 139   }
 140 
 141   private waitForIdle(): Promise<void> {
 142     if (this.isIdle) return Promise.resolve();
 143     return new Promise((resolve) => this.idleWaiters.push(resolve));
 144   }
